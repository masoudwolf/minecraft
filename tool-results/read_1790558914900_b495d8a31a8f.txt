    99→// ─── Model builders ──────────────────────────────────────────────────────────
   100→function partMat(tex: THREE.CanvasTexture): THREE.MeshLambertMaterial {
   101→  return new THREE.MeshLambertMaterial({ map: tex });
   102→}
   103→
   104→function boxPart(w: number, h: number, d: number, mat: THREE.MeshLambertMaterial, tag: string): THREE.Mesh {
   105→  const geo = new THREE.BoxGeometry(w, h, d);
   106→  const mesh = new THREE.Mesh(geo, mat);
   107→  mesh.userData.part = tag;
   108→  return mesh;
   109→}
   110→
   111→/**
   112→ * Head box with Minecraft-accurate texturing: the face texture is applied ONLY
   113→ * to the front (+Z — the model's forward axis); the other 5 faces get the
   114→ * featureless skin. Both materials are pushed onto extraMats so per-instance
   115→ * cloning and world-light tinting cover the whole head.
   116→ */
   117→function headPart(w: number, h: number, d: number, skins: { head: THREE.CanvasTexture; headPlain: THREE.CanvasTexture }, extraMats: THREE.MeshLambertMaterial[], tag = 'head'): THREE.Mesh {
   118→  const face = partMat(skins.head);
   119→  const plain = partMat(skins.headPlain);
   120→  extraMats.push(face, plain);
   121→  const mesh = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), [plain, plain, plain, plain, face, plain]);
   122→  mesh.userData.part = tag;
   123→  return mesh;
   124→}
   125→
   126→function makeShadow(radius: number, scene: THREE.Scene): THREE.Mesh {
   127→  const geo = new THREE.CircleGeometry(radius, 12);
   128→  const mat = new THREE.MeshBasicMaterial({ color: 0x000000, transparent: true, opacity: 0.28, depthWrite: false });
   129→  const mesh = new THREE.Mesh(geo, mat);
   130→  mesh.rotation.x = -Math.PI / 2;
   131→  mesh.renderOrder = 1;
   132→  scene.add(mesh);
   133→  return mesh;
   134→}

   136→function quadruped(skins: ReturnType<typeof getMobSkins>, opts: {
   137→  bodyW: number; bodyH: number; bodyD: number; bodyY: number;
   138→  legW: number; legH: number;
   139→  headS: number; headY: number; headZ: number;
   140→  shadowR: number;
   141→}): MobParts {
   142→  const mats = [partMat(skins.head), partMat(skins.body), partMat(skins.limb)];
   143→  const extra: THREE.MeshLambertMaterial[] = [];
   144→  const group = new THREE.Group();
   145→  const body = boxPart(opts.bodyW, opts.bodyH, opts.bodyD, mats[1], 'body');
   146→  body.position.y = opts.bodyY;
   147→  group.add(body);
   148→  const head = headPart(opts.headS, opts.headS, opts.headS, skins, extra);
   149→  head.position.set(0, opts.headY, opts.headZ);
   150→  group.add(head);
   151→  const legs: THREE.Mesh[] = [];
   152→  const lx = opts.bodyW / 2 - opts.legW / 2;
   153→  const lz = opts.bodyD / 2 - opts.legW / 2;
   154→  for (const [sx, sz] of [[-1, -1], [1, -1], [-1, 1], [1, 1]]) {
   155→    const leg = boxPart(opts.legW, opts.legH, opts.legW, mats[2], 'limb');
   156→    leg.position.set(sx * lx, opts.bodyY - opts.bodyH / 2 - opts.legH / 2, sz * lz);
   157→    group.add(leg);
   158→    legs.push(leg);
   159→  }
   160→  return { group, head, legs, arms: [], materials: [...mats, ...extra], shadow: null as unknown as THREE.Mesh };
   161→}

   163→function humanoid(skins: ReturnType<typeof getMobSkins>, thin = false): MobParts {
   164→  const lw = thin ? 0.16 : 0.25;
   165→  const aw = thin ? 0.14 : 0.22;
   166→  const mats = [partMat(skins.head), partMat(skins.body), partMat(skins.limb)];
   167→  const extra: THREE.MeshLambertMaterial[] = [];
   168→  const group = new THREE.Group();
   169→  const body = boxPart(0.5, 0.72, 0.26, mats[1], 'body');
   170→  body.position.y = 1.1;
   171→  group.add(body);
   172→  const head = headPart(0.5, 0.5, 0.5, skins, extra);
   173→  head.position.y = 1.72;
   174→  group.add(head);
   175→  const legs: THREE.Mesh[] = [];
   176→  for (const sx of [-1, 1]) {
   177→    const leg = boxPart(lw, 0.74, lw, mats[2], 'limb');
   178→    leg.position.set(sx * 0.125, 0.37, 0);
   179→    group.add(leg);
   180→    legs.push(leg);
   181→  }
   182→  const arms: THREE.Mesh[] = [];
   183→  for (const sx of [-1, 1]) {
   184→    const pivot = new THREE.Group();
   185→    pivot.position.set(sx * (0.25 + aw / 2), 1.42, 0);
   186→    const arm = boxPart(aw, 0.7, aw, mats[2], 'limb');
   187→    arm.position.y = -0.32;
   188→    pivot.add(arm);
   189→    group.add(pivot);
   190→    arms.push(arm);
   191→    (arm as unknown as { pivot: THREE.Group }).pivot = pivot;
   192→  }
   193→  return { group, head, legs, arms, materials: [...mats, ...extra], shadow: null as unknown as THREE.Mesh };
   194→}

   432→function rayAABB(ox: number, oy: number, oz: number, dx: number, dy: number, dz: number, e: AABBEntity, maxDist: number): number | null {
   433→  const half = e.width / 2;
   434→  const minX = e.x - half, maxX = e.x + half;
   435→  const minY = e.y, maxY = e.y + e.height;
   436→  const minZ = e.z - half, maxZ = e.z + half;
   437→  let tmin = 0;
   438→  let tmax = maxDist;
   439→  for (const [o, d, min, max] of [[ox, dx, minX, maxX], [oy, dy, minY, maxY], [oz, dz, minZ, maxZ]] as const) {
   440→    if (Math.abs(d) < 1e-9) {
   441→      if (o < min || o > max) return null;
   442→    } else {
   443→      let t1 = (min - o) / d;
   444→      let t2 = (max - o) / d;
   445→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp; }
   446→      tmin = Math.max(tmin, t1);
   447→      tmax = Math.min(tmax, t2);
   448→      if (tmin > tmax) return null;
   449→    }
   450→  }
   451→  return tmin;
   452→}

   558→  private spawn(type: MobType, x: number, y: number, z: number, variant = ''): Mob | null {
   559→    const def = MOB_DEFS[type];
   560→    const skinKey = variant && type === 'sheep' ? `sheep:${variant}` : type;
   561→    const skins = getMobSkins(skinKey);
   562→    const parts = def.builder(skins);

   564→    // use [plain×4, face] material arrays)
   565→    const cloned = parts.materials.map((mm) => mm.clone());
   566→    const mapping = new Map<THREE.Material, THREE.Material>();
   567→    parts.materials.forEach((mm, i2) => mapping.set(mm, cloned[i2]));
   568→    parts.group.traverse((obj) => {

   572→        mesh.material = mesh.material.map((mm) => mapping.get(mm) ?? mm);
   573→      } else if (mapping.has(mesh.material as THREE.Material)) {
   574→        mesh.material = mapping.get(mesh.material) as THREE.Material;
   575→      }
   576→    });
   577→    parts.materials = cloned;

   582→      x, y, z, vx: 0, vy: 0, vz: 0,

   585→      group: parts.group, parts,

   587→      yaw: Math.random() * Math.PI * 2,
   588→      targetYaw: 0,
   589→      state: 'idle',
   590→      stateTimer: 1 + Math.random() * 3,
   591→      walkPhase: 0,
   592→      hurtT: 0, attackCd: 0, ambientCd: 3 + Math.random() * 6,

   594→      fuse: -1, dead: false, deathT: 0,

   599→      lightF: 1, lastAppliedF: 1,

   602→    parts.shadow = shadow;

   607→  private trySpawnMob(playerX: number, playerY: number, playerZ: number, sunLevel: number, passiveCount: number, hostileCount: number): void {
   608→    const angle = Math.random() * Math.PI * 2;
   609→    const dist = 16 + Math.random() * 26;
   610→    const x = Math.floor(playerX + Math.cos(angle) * dist) + 0.5;
   611→    const z = Math.floor(playerZ + Math.sin(angle) * dist) + 0.5;
   612→    const cx = Math.floor(x);
   613→    const cz = Math.floor(z);
   614→    // find surface
   615→    let sy = -1;
   616→    for (let y = 95; y > 2; y--) {
   617→      const id = this.world.getBlock(cx, y, cz);
   618→      if (id !== 0 && !isWaterId(id)) {
   619→        sy = y;
   620→        break;
   621→      }
   622→    }
   623→    if (sy < 3) return;
   624→    const above = this.world.getBlock(cx, sy + 1, cz);
   625→    const above2 = this.world.getBlock(cx, sy + 2, cz);
   626→    if (above !== 0 || above2 !== 0) return;
   627→    const groundId = this.world.getBlock(cx, sy, cz);
   628→    if (isWaterId(groundId)) return;
   629→
   630→    const l = this.world.getLight(cx, sy + 1, cz);
   631→    if (l < 0) return;
   632→    const skyL = l >> 4;
   633→    const blkL = l & 15;
   634→    const effLight = Math.max(blkL, skyL * sunLevel);
   635→
   636→    // villagers live on village grounds (planks / cobblestone); mooshrooms on mycelium
   637→    if (groundId === BLOCK.PLANKS || groundId === BLOCK.COBBLESTONE) {
   638→      // iron golem: at most one patrolling each village chunk roll
   639→      const golemCount = this.mobs.filter((m2) => m2.type === 'golem').length;
   640→      if (golemCount < 1 && Math.random() < 0.22 && skyL >= 9 && sunLevel > 0.55) {
   641→        this.spawn('golem', x, sy + 1, z);
   642→        return;
   643→      }
   644→      const villagerCount = this.mobs.filter((m) => m.type === 'villager').length;
   645→      if (villagerCount < 5 && skyL >= 9 && sunLevel > 0.55) {
   646→        const herd = 1 + Math.floor(Math.random() * 2);
   647→        for (let i = 0; i < herd; i++) this.spawn('villager', x + (Math.random() - 0.5) * 2, sy + 1, z + (Math.random() - 0.5) * 2);
   648→      }
   649→      return;
   650→    }
   651→    if (groundId === BLOCK.MYCELIUM) {
   652→      const passiveNow = passiveCount + this.mobs.filter((m) => m.type === 'mooshroom').length;
   653→      if (passiveNow < 12 && skyL >= 9 && sunLevel > 0.55) {
   654→        const herd = 1 + Math.floor(Math.random() * 2);
   655→        for (let i = 0; i < herd; i++) this.spawn('mooshroom', x + (Math.random() - 0.5) * 3, sy + 1, z + (Math.random() - 0.5) * 3);
   656→      }
   657→      return;
   658→    }
   659→
   660→    const wantHostile = hostileCount < 12 && (effLight < 6);
   661→    const wantPassive = passiveCount < 10 && skyL >= 9 && sunLevel > 0.55 && (groundId === BLOCK.GRASS || groundId === BLOCK.SNOW_GRASS);
   662→
   663→    if (wantHostile && (!wantPassive || Math.random() < 0.65)) {
   664→      const roll = Math.random();
   665→      const type: MobType = roll < 0.36 ? 'zombie' : roll < 0.64 ? 'skeleton' : roll < 0.84 ? 'creeper' : roll < 0.96 ? 'spider' : 'enderman';
   666→      this.spawn(type, x, sy + 1, z);
   667→    } else if (wantPassive) {
   668→      const roll = Math.random();
   669→      const type: MobType = roll < 0.3 ? 'pig' : roll < 0.55 ? 'sheep' : roll < 0.8 ? 'cow' : 'chicken';
   670→      // spawn small herd for passive
   671→      const herd = 1 + Math.floor(Math.random() * 3);
   672→      for (let i = 0; i < herd; i++) {
   673→        this.spawn(type, x + (Math.random() - 0.5) * 3, sy + 1, z + (Math.random() - 0.5) * 3, type === 'sheep' ? pickSheepVariant() : '');
   674→      }
   675→    }
   676→  }

   678→  hurtMob(mob: Mob, dmg: number, kx: number, kz: number, cb?: MobCallbacks): boolean {
   679→    if (mob.dead) return false;
   680→    mob.health -= dmg;
   681→    mob.hurtT = 0.4;
   682→    const len = Math.hypot(kx, kz) || 1;
   683→    mob.vx += (kx / len) * 7;
   684→    mob.vz += (kz / len) * 7;
   685→    mob.vy = Math.max(mob.vy, 4.4);
   686→    audio.mobHurt(mob.def.sound);
   687→    if (mob.def.hostile) {
   688→      mob.state = 'chase';
   689→      if (mob.type === 'spider' || mob.type === 'enderman') mob.provoked = true;
   690→    } else if (mob.type === 'golem') {
   691→      // golems never flee — they stoically keep patrolling
   692→      mob.state = 'idle';
   693→    } else {
   694→      mob.state = 'flee';
   695→      mob.stateTimer = 4;
   696→      mob.wanderX = mob.x + (mob.x - kx) * 10;
   697→      mob.wanderZ = mob.z + (mob.z - kz) * 10;
   698→    }
   699→    // endermen often warp away when hurt (MC behavior)
   700→    const ecb = cb ?? this.lastCb;
   701→    if (mob.type === 'enderman' && !mob.dead && ecb && Math.random() < 0.55) {
   702→      this.teleportNear(mob, mob.x, mob.z, 7, 15, ecb);
   703→    }
   704→    if (mob.health <= 0) {
   705→      mob.dead = true;
   706→      mob.deathT = 0.45;
   707→      if (mob.type === 'creeper' && mob.fuse >= 0) mob.fuse = -1;
   708→      return true;
   709→    }
   710→    return false;
   711→  }

   713→  /** enderman teleport: find a valid surface spot within [minR,maxR] of (cx,cz) */
   714→  private teleportNear(m: Mob, cx: number, cz: number, minR: number, maxR: number, cb: MobCallbacks): boolean {
   715→    const ox = m.x, oy = m.y, oz = m.z;
   716→    for (let i = 0; i < 14; i++) {
   717→      const a = Math.random() * Math.PI * 2;
   718→      const r = minR + Math.random() * (maxR - minR);
   719→      const x = Math.floor(cx + Math.cos(a) * r);
   720→      const z = Math.floor(cz + Math.sin(a) * r);
   721→      let sy = -1;
   722→      for (let y = 95; y > 2; y--) {
   723→        const id = this.world.getBlock(x, y, z);
   724→        if (id !== 0 && !isWaterId(id)) { sy = y; break; }
   725→      }
   726→      if (sy < 3) continue;
   727→      // head room for full height
   728→      const top = Math.ceil(m.def.height);
   729→      let clear = true;
   730→      for (let yy = sy + 1; yy <= sy + top; yy++) {
   731→        if (this.world.getBlock(x, yy, z) !== 0) { clear = false; break; }
   732→      }
   733→      if (!clear) continue;
   734→      m.x = x + 0.5; m.y = sy + 1; m.z = z + 0.5;
   735→      m.vx = 0; m.vy = 0; m.vz = 0;
   736→      m.state = 'idle'; m.stateTimer = 1;
   737→      cb.teleportParticles(ox, oy + m.height * 0.4, oz);
   738→      cb.teleportParticles(m.x, m.y + m.height * 0.4, m.z);
   739→      audio.enderTeleport();
   740→      return true;
   741→    }
   742→    return false;
   743→  }

   745→  /** effective light (0..15) at a mob's position */
   746→  private lightAt(m: Mob, sunLevel: number): number {
   747→    const l = this.world.getLight(Math.floor(m.x), Math.floor(m.y + 0.5), Math.floor(m.z));
   748→    if (l < 0) return 15;
   749→    return Math.max(l & 15, (l >> 4) * sunLevel);
   750→  }

   752→  /** creeper explosion */
   753→  private explode(x: number, y: number, z: number, cb: MobCallbacks): void {
   754→    audio.boom();
   755→    const R = 2.6;
   756→    for (let bx = Math.floor(x - R); bx <= Math.floor(x + R); bx++)
   757→      for (let by = Math.floor(y - R); by <= Math.floor(y + R); by++)
   758→        for (let bz = Math.floor(z - R); bz <= Math.floor(z + R); bz++) {
   759→          const d = Math.hypot(bx + 0.5 - x, by + 0.5 - y, bz + 0.5 - z);
   760→          if (d > R) continue;
   761→          const id = this.world.getBlock(bx, by, bz);
   762→          if (id !== 0 && id !== BLOCK.BEDROCK && id !== BLOCK.WATER) {
   763→            // chain reaction: TNT in the blast primes instead of vanishing
   764→            if (id === BLOCK.TNT) {
   765→              cb.igniteTnt?.(bx, by, bz);
   766→              continue;
   767→            }
   768→            this.world.setBlock(bx, by, bz, 0);
   769→          }
   770→        }
   771→    // damage nearby mobs (chain) + player
   772→    for (const m of this.mobs) {
   773→      if (m.dead) continue;
   774→      const d = Math.hypot(m.x - x, m.y - y, m.z - z);
   775→      if (d < R * 2 && m.type !== 'creeper') {
   776→        this.hurtMob(m, Math.max(1, Math.round(14 * (1 - d / (R * 2)))), m.x - x, m.z - z, cb);
   777→      }
   778→    }
   779→    const pd = Math.hypot(cb.playerX - x, cb.playerY - y, cb.playerZ - z);
   780→    if (pd < R * 2) {
   781→      cb.damagePlayer(Math.max(1, Math.round(16 * (1 - pd / (R * 2)))), x, z);
   782→    }
   783→    cb.explodeParticles(x, y, z);
   784→  }

   786→  update(dt: number, player: AABBEntity & { eyeY(): number }, sunLevel: number, cb: MobCallbacks): void {
   787→    this.time += dt;
   788→    this.lastCb = cb;
   789→    // ── spawning ──
   790→    this.spawnTimer -= dt;
   791→    if (this.spawnTimer <= 0) {
   792→      this.spawnTimer = 1.6;
   793→      let passive = 0, hostile = 0;
   794→      for (const m of this.mobs) {
   795→        if (m.def.hostile) hostile++; else passive++;
   796→      }
   797→      if (this.mobs.length < 22 && Math.random() < 0.5) {
   798→        this.trySpawnMob(player.x, player.y, player.z, sunLevel, passive, hostile);
   799→      }
   800→    }
   801→
   802→    // ── mobs ──
   803→    for (let i = this.mobs.length - 1; i >= 0; i--) {
   804→      const m = this.mobs[i];
   805→      const distToPlayer = Math.hypot(player.x - m.x, player.y - m.y, player.z - m.z);
   806→
   807→      // despawn far / invalid: NaN or below-world positions never heal on their
   808→      // own and NaN breaks the distance check (NaN > 64 === false), so mobs that
   809→      // glitch out would otherwise persist forever
   810→      if (
   811→        distToPlayer > 64 ||
   812→        !Number.isFinite(m.x) || !Number.isFinite(m.y) || !Number.isFinite(m.z) ||
   813→        m.y < -20
   814→      ) {
   815→        this.scene.remove(m.group);
   816→        this.scene.remove(m.parts.shadow);
   817→        this.mobs.splice(i, 1);
   818→        continue;
   819→      }
   820→
   821→      // death animation
   822→      if (m.dead) {
   823→        m.deathT -= dt;
   824→        m.group.rotation.z = Math.min(Math.PI / 2, m.group.rotation.z + dt * 6);
   825→        this.setTint(m, 1, 0.35, 0.35);
   826→        if (m.deathT <= 0) {
   827→          for (const drop of m.def.drops) {
   828→            const n = drop.min + Math.floor(Math.random() * (drop.max - drop.min + 1));
   829→            // sheep drop wool matching their color variant
   830→            const dropId = m.type === 'sheep' && drop.id === BLOCK.WOOL ? sheepWoolId(m.variant) : drop.id;
   831→            for (let d = 0; d < n; d++) {
   832→              cb.spawnDrop(dropId, m.x, m.y + 0.4, m.z);
   833→            }
   834→          }
   835→          cb.spawnXP(m.x, m.y + 0.4, m.z, m.def.hostile ? 5 : 1 + Math.floor(Math.random() * 3));
   836→          cb.deathParticles(m.x, m.y + m.height / 2, m.z);
   837→          this.scene.remove(m.group);
   838→          this.scene.remove(m.parts.shadow);
   839→          this.mobs.splice(i, 1);
   840→        }
   841→        continue;
   842→      }
   843→
   844→      // hurt tint decay
   845→      if (m.hurtT > 0) {
   846→        m.hurtT -= dt;
   847→        this.setTint(m, 1, 0.35, 0.35);
   848→        if (m.hurtT <= 0) this.setTint(m, 1, 1, 1);
   849→      }
   850→
   851→      // ambient sound
   852→      m.ambientCd -= dt;
   853→      if (m.ambientCd <= 0 && distToPlayer < 18) {
   854→        m.ambientCd = 5 + Math.random() * 9;
   855→        audio.mobAmbient(m.def.sound, distToPlayer);
   856→      }
   857→
   858→      // ── AI ──
   859→      m.attackCd -= dt;
   860→      m.stateTimer -= dt;
   861→      let moveSpeed = 0;
   862→      let wantX = 0, wantZ = 0;
   863→
   864→      // creative players are invisible to hostile AI: treat hostiles as passive wanderers
   865→      const hostileActive = m.def.hostile && !cb.playerCreative;
   866→
   867→      if (m.type === 'golem') {
   868→        // ── iron golem: patrol, and charge the nearest hostile mob ──
   869→        let prey: Mob | null = null;
   870→        let preyDist = 11;

  1291→              const shooterDist = Math.hypot(a.x - cb.playerX, a.y - cb.playerY, a.z - cb.playerZ);
  1292→              this.hurtMob(m, a.dmg, (a.vx / klen) * 4.5, (a.vz / klen) * 4.5, cb);

  1300→          if (done) break;
  1301→        } else {
  1302→          // player hit
  1303→          const px = player.x, py = player.y + player.height * 0.5, pz = player.z;
