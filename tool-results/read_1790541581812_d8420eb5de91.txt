    64→// ─── Model builders ──────────────────────────────────────────────────────────
    65→function partMat(tex: THREE.CanvasTexture): THREE.MeshLambertMaterial {
    66→  return new THREE.MeshLambertMaterial({ map: tex });
    67→}
    68→
    69→function boxPart(w: number, h: number, d: number, mat: THREE.MeshLambertMaterial, tag: string): THREE.Mesh {
    70→  const geo = new THREE.BoxGeometry(w, h, d);
    71→  const mesh = new THREE.Mesh(geo, mat);
    72→  mesh.userData.part = tag;
    73→  return mesh;
    74→}

    76→function makeShadow(radius: number, scene: THREE.Scene): THREE.Mesh {
    77→  const geo = new THREE.CircleGeometry(radius, 12);
    78→  const mat = new THREE.MeshBasicMaterial({ color: 0x000000, transparent: true, opacity: 0.28, depthWrite: false });
    79→  const mesh = new THREE.Mesh(geo, mat);
    80→  mesh.rotation.x = -Math.PI / 2;
    81→  mesh.renderOrder = 1;
    82→  scene.add(mesh);
    83→  return mesh;
    84→}

    86→function quadruped(skins: ReturnType<typeof getMobSkins>, opts: {
    87→  bodyW: number; bodyH: number; bodyD: number; bodyY: number;
    88→  legW: number; legH: number;
    89→  headS: number; headY: number; headZ: number;
    90→  shadowR: number;
    91→}): MobParts {
    92→  const mats = [partMat(skins.head), partMat(skins.body), partMat(skins.limb)];
    93→  const group = new THREE.Group();
    94→  const body = boxPart(opts.bodyW, opts.bodyH, opts.bodyD, mats[1], 'body');
    95→  body.position.y = opts.bodyY;
    96→  group.add(body);
    97→  const head = boxPart(opts.headS, opts.headS, opts.headS, mats[0], 'head');
    98→  head.position.set(0, opts.headY, opts.headZ);
    99→  group.add(head);
   100→  const legs: THREE.Mesh[] = [];
   101→  const lx = opts.bodyW / 2 - opts.legW / 2;
   102→  const lz = opts.bodyD / 2 - opts.legW / 2;
   103→  for (const [sx, sz] of [[-1, -1], [1, -1], [-1, 1], [1, 1]]) {
   104→    const leg = boxPart(opts.legW, opts.legH, opts.legW, mats[2], 'limb');
   105→    leg.position.set(sx * lx, opts.bodyY - opts.bodyH / 2 - opts.legH / 2, sz * lz);
   106→    group.add(leg);
   107→    legs.push(leg);
   108→  }
   109→  return { group, head, legs, arms: [], materials: mats, shadow: null as unknown as THREE.Mesh };
   110→}

   112→function humanoid(skins: ReturnType<typeof getMobSkins>, thin = false): MobParts {
   113→  const lw = thin ? 0.16 : 0.25;
   114→  const aw = thin ? 0.14 : 0.22;
   115→  const mats = [partMat(skins.head), partMat(skins.body), partMat(skins.limb)];
   116→  const group = new THREE.Group();
   117→  const body = boxPart(0.5, 0.72, 0.26, mats[1], 'body');
   118→  body.position.y = 1.1;
   119→  group.add(body);
   120→  const head = boxPart(0.5, 0.5, 0.5, mats[0], 'head');
   121→  head.position.y = 1.72;
   122→  group.add(head);
   123→  const legs: THREE.Mesh[] = [];
   124→  for (const sx of [-1, 1]) {
   125→    const leg = boxPart(lw, 0.74, lw, mats[2], 'limb');
   126→    leg.position.set(sx * 0.125, 0.37, 0);
   127→    group.add(leg);
   128→    legs.push(leg);
   129→  }
   130→  const arms: THREE.Mesh[] = [];
   131→  for (const sx of [-1, 1]) {
   132→    const pivot = new THREE.Group();
   133→    pivot.position.set(sx * (0.25 + aw / 2), 1.42, 0);
   134→    const arm = boxPart(aw, 0.7, aw, mats[2], 'limb');
   135→    arm.position.y = -0.32;
   136→    pivot.add(arm);
   137→    group.add(pivot);
   138→    arms.push(arm);
   139→    (arm as unknown as { pivot: THREE.Group }).pivot = pivot;
   140→  }
   141→  return { group, head, legs, arms, materials: mats, shadow: null as unknown as THREE.Mesh };
   142→}

   203→// ─── Ray vs AABB (slab method) ───────────────────────────────────────────────
   204→export interface MobHit { mob: Mob; dist: number }
   205→
   206→function rayAABB(ox: number, oy: number, oz: number, dx: number, dy: number, dz: number, e: AABBEntity, maxDist: number): number | null {
   207→  const half = e.width / 2;
   208→  const minX = e.x - half, maxX = e.x + half;
   209→  const minY = e.y, maxY = e.y + e.height;
   210→  const minZ = e.z - half, maxZ = e.z + half;
   211→  let tmin = 0;
   212→  let tmax = maxDist;
   213→  for (const [o, d, min, max] of [[ox, dx, minX, maxX], [oy, dy, minY, maxY], [oz, dz, minZ, maxZ]] as const) {
   214→    if (Math.abs(d) < 1e-9) {
   215→      if (o < min || o > max) return null;
   216→    } else {
   217→      let t1 = (min - o) / d;
   218→      let t2 = (max - o) / d;
   219→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp; }
   220→      tmin = Math.max(tmin, t1);
   221→      tmax = Math.min(tmax, t2);
   222→      if (tmin > tmax) return null;
   223→    }
   224→  }
   225→  return tmin;
   226→}

   243→export class MobManager {
   244→  mobs: Mob[] = [];
   245→  arrows: Arrow[] = [];
   246→  private scene: THREE.Scene;
   247→  private world: { getBlock(x: number, y: number, z: number): number; setBlock(x: number, y: number, z: number, id: number): void };
   248→  private spawnTimer = 0;
   249→  private arrowMat: THREE.MeshLambertMaterial;
   250→  private time = 0;
   251→
   252→  constructor(scene: THREE.Scene, world: { getBlock(x: number, y: number, z: number): number; setBlock(x: number, y: number, z: number, id: number): void }) {
   253→    this.scene = scene;
   254→    this.world = world;
   255→    this.arrowMat = new THREE.MeshLambertMaterial({ color: 0x8a6a4a });
   256→  }
   257→
   258→  get count(): number {
   259→    return this.mobs.length;
   260→  }
   261→
   262→  /** QA/testing helper: force-spawn a mob at position */
   263→  debugSpawn(type: MobType, x: number, y: number, z: number): Mob | null {
   264→    return this.spawn(type, x, y, z);
   265→  }
   266→
   267→  clear(): void {
   268→    for (const m of this.mobs) {
   269→      this.scene.remove(m.group);
   270→      this.scene.remove(m.parts.shadow);
   271→    }
   272→    this.mobs = [];
   273→    for (const a of this.arrows) this.scene.remove(a.mesh);
   274→    this.arrows = [];
   275→  }
   276→
   277→  /** raycast mobs for attacks; returns nearest hit within reach */
   278→  raycastMob(ox: number, oy: number, oz: number, dx: number, dy: number, dz: number, reach: number): MobHit | null {
   279→    let best: MobHit | null = null;
   280→    for (const m of this.mobs) {
   281→      if (m.dead) continue;
   282→      const t = rayAABB(ox, oy, oz, dx, dy, dz, m, reach);
   283→      if (t !== null && (!best || t < best.dist)) best = { mob: m, dist: t };
   284→    }
   285→    return best;
   286→  }

   288→  private spawn(type: MobType, x: number, y: number, z: number): Mob | null {
   289→    const def = MOB_DEFS[type];
   290→    const skins = getMobSkins(type);
   291→    const parts = def.builder(skins);
   292→    // per-instance material clones so hurt tint is individual
   293→    const cloned = parts.materials.map((mm) => mm.clone());
   294→    const mapping = new Map<THREE.Material, THREE.Material>();
   295→    parts.materials.forEach((mm, i2) => mapping.set(mm, cloned[i2]));
   296→    parts.group.traverse((obj) => {
   297→      const mesh = obj as THREE.Mesh;
   298→      if (mesh.isMesh && mapping.has(mesh.material as THREE.Material)) {
   299→        mesh.material = mapping.get(mesh.material) as THREE.Material;
   300→      }
   301→    });
   302→    parts.materials = cloned;
   303→    this.scene.add(parts.group);
   304→    const shadow = makeShadow(def.width * 0.55, this.scene);
   305→    const mob: Mob = {
   306→      type, def,
   307→      x, y, z, vx: 0, vy: 0, vz: 0,
   308→      width: def.width, height: def.height,
   309→      onGround: false, inWater: false,
   310→      group: parts.group, parts,
   311→      health: def.health,
   312→      yaw: Math.random() * Math.PI * 2,
   313→      targetYaw: 0,
   314→      state: 'idle',
   315→      stateTimer: 1 + Math.random() * 3,
   316→      walkPhase: 0,
   317→      hurtT: 0, attackCd: 0, ambientCd: 3 + Math.random() * 6,
   318→      burnTimer: 0, burning: false,
   319→      fuse: -1, dead: false, deathT: 0,
   320→      wanderX: x, wanderZ: z,
   321→    };
   322→    mob.targetYaw = mob.yaw;
   323→    parts.shadow = shadow;
   324→    this.mobs.push(mob);
   325→    return mob;
   326→  }

   328→  private trySpawnMob(playerX: number, playerY: number, playerZ: number, sunLevel: number, passiveCount: number, hostileCount: number): void {
   329→    const angle = Math.random() * Math.PI * 2;
   330→    const dist = 16 + Math.random() * 26;
   331→    const x = Math.floor(playerX + Math.cos(angle) * dist) + 0.5;
   332→    const z = Math.floor(playerZ + Math.sin(angle) * dist) + 0.5;
   333→    const cx = Math.floor(x);
   334→    const cz = Math.floor(z);
   335→    // find surface
   336→    let sy = -1;
   337→    for (let y = 95; y > 2; y--) {
   338→      const id = this.world.getBlock(cx, y, cz);
   339→      if (id !== 0 && !isWaterId(id)) {
   340→        sy = y;
   341→        break;
   342→      }
   343→    }
   344→    if (sy < 3) return;
   345→    const above = this.world.getBlock(cx, sy + 1, cz);
   346→    const above2 = this.world.getBlock(cx, sy + 2, cz);
   347→    if (above !== 0 || above2 !== 0) return;
   348→    const groundId = this.world.getBlock(cx, sy, cz);
   349→    if (isWaterId(groundId)) return;
   350→
   351→    const l = this.world.getLight(cx, sy + 1, cz);
   352→    if (l < 0) return;
   353→    const skyL = l >> 4;
   354→    const blkL = l & 15;
   355→    const effLight = Math.max(blkL, skyL * sunLevel);
   356→
   357→    const wantHostile = hostileCount < 12 && (effLight < 6);
   358→    const wantPassive = passiveCount < 10 && skyL >= 9 && sunLevel > 0.55 && (groundId === BLOCK.GRASS || groundId === BLOCK.SNOW_GRASS);
   359→
   360→    if (wantHostile && (!wantPassive || Math.random() < 0.65)) {
   361→      const roll = Math.random();
   362→      const type: MobType = roll < 0.45 ? 'zombie' : roll < 0.75 ? 'skeleton' : 'creeper';
   363→      this.spawn(type, x, sy + 1, z);
   364→    } else if (wantPassive) {
   365→      const roll = Math.random();
   366→      const type: MobType = roll < 0.3 ? 'pig' : roll < 0.55 ? 'sheep' : roll < 0.8 ? 'cow' : 'chicken';
   367→      // spawn small herd for passive
   368→      const herd = 1 + Math.floor(Math.random() * 3);
   369→      for (let i = 0; i < herd; i++) {
   370→        this.spawn(type, x + (Math.random() - 0.5) * 3, sy + 1, z + (Math.random() - 0.5) * 3);
   371→      }
   372→    }
   373→  }

   375→  hurtMob(mob: Mob, dmg: number, kx: number, kz: number): boolean {
   376→    if (mob.dead) return false;
   377→    mob.health -= dmg;
   378→    mob.hurtT = 0.4;
   379→    const len = Math.hypot(kx, kz) || 1;
   380→    mob.vx += (kx / len) * 7;
   381→    mob.vz += (kz / len) * 7;
   382→    mob.vy = Math.max(mob.vy, 4.4);
   383→    audio.mobHurt(mob.def.sound);
   384→    if (mob.def.hostile) {
   385→      mob.state = 'chase';
   386→    } else {
   387→      mob.state = 'flee';
   388→      mob.stateTimer = 4;
   389→      mob.wanderX = mob.x + (mob.x - kx) * 10;
   390→      mob.wanderZ = mob.z + (mob.z - kz) * 10;
   391→    }
   392→    if (mob.health <= 0) {
   393→      mob.dead = true;
   394→      mob.deathT = 0.45;
   395→      if (mob.type === 'creeper' && mob.fuse >= 0) mob.fuse = -1;
   396→      return true;
   397→    }
   398→    return false;
   399→  }

   401→  /** creeper explosion */
   402→  private explode(x: number, y: number, z: number, cb: MobCallbacks): void {
   403→    audio.boom();
   404→    const R = 2.6;
   405→    for (let bx = Math.floor(x - R); bx <= Math.floor(x + R); bx++)
   406→      for (let by = Math.floor(y - R); by <= Math.floor(y + R); by++)
   407→        for (let bz = Math.floor(z - R); bz <= Math.floor(z + R); bz++) {
   408→          const d = Math.hypot(bx + 0.5 - x, by + 0.5 - y, bz + 0.5 - z);
   409→          if (d > R) continue;
   410→          const id = this.world.getBlock(bx, by, bz);
   411→          if (id !== 0 && id !== BLOCK.BEDROCK && id !== BLOCK.WATER) {
   412→            this.world.setBlock(bx, by, bz, 0);
   413→          }
   414→        }
   415→    // damage nearby mobs (chain) + player
   416→    for (const m of this.mobs) {
   417→      if (m.dead) continue;
   418→      const d = Math.hypot(m.x - x, m.y - y, m.z - z);
   419→      if (d < R * 2 && m.type !== 'creeper') {
   420→        this.hurtMob(m, Math.max(1, Math.round(14 * (1 - d / (R * 2)))), m.x - x, m.z - z);
   421→      }
   422→    }
   423→    const pd = Math.hypot(cb.playerX - x, cb.playerY - y, cb.playerZ - z);
   424→    if (pd < R * 2) {
   425→      cb.damagePlayer(Math.max(1, Math.round(16 * (1 - pd / (R * 2)))), x, z);
   426→    }
   427→    cb.explodeParticles(x, y, z);
   428→  }

   430→  update(dt: number, player: AABBEntity & { eyeY(): number }, sunLevel: number, cb: MobCallbacks): void {
   431→    this.time += dt;
   432→    // ── spawning ──
   433→    this.spawnTimer -= dt;
   434→    if (this.spawnTimer <= 0) {
   435→      this.spawnTimer = 1.6;
   436→      let passive = 0, hostile = 0;
   437→      for (const m of this.mobs) {
   438→        if (m.def.hostile) hostile++; else passive++;
   439→      }
   440→      if (this.mobs.length < 22 && Math.random() < 0.5) {
   441→        this.trySpawnMob(player.x, player.y, player.z, sunLevel, passive, hostile);
   442→      }
   443→    }
   444→
   445→    // ── mobs ──
   446→    for (let i = this.mobs.length - 1; i >= 0; i--) {
   447→      const m = this.mobs[i];
   448→      const distToPlayer = Math.hypot(player.x - m.x, player.y - m.y, player.z - m.z);
   449→
   450→      // despawn far
   451→      if (distToPlayer > 64) {
   452→        this.scene.remove(m.group);
   453→        this.scene.remove(m.parts.shadow);
   454→        this.mobs.splice(i, 1);
   455→        continue;
   456→      }
   457→
   458→      // death animation
   459→      if (m.dead) {
   460→        m.deathT -= dt;
   461→        m.group.rotation.z = Math.min(Math.PI / 2, m.group.rotation.z + dt * 6);
   462→        tint(m.parts.materials, 1, 0.35, 0.35);
   463→        if (m.deathT <= 0) {
   464→          for (const drop of m.def.drops) {
   465→            const n = drop.min + Math.floor(Math.random() * (drop.max - drop.min + 1));
   466→            for (let d = 0; d < n; d++) {
   467→              cb.spawnDrop(drop.id, m.x, m.y + 0.4, m.z);
   468→            }
   469→          }
   470→          cb.spawnXP(m.x, m.y + 0.4, m.z, m.def.hostile ? 5 : 1 + Math.floor(Math.random() * 3));
   471→          cb.deathParticles(m.x, m.y + m.height / 2, m.z);
   472→          this.scene.remove(m.group);
   473→          this.scene.remove(m.parts.shadow);
   474→          this.mobs.splice(i, 1);
   475→        }
   476→        continue;
   477→      }
   478→
   479→      // hurt tint decay
   480→      if (m.hurtT > 0) {
   481→        m.hurtT -= dt;
   482→        tint(m.parts.materials, 1, 0.35, 0.35);
   483→        if (m.hurtT <= 0) tint(m.parts.materials, 1, 1, 1);
   484→      }
   485→
   486→      // ambient sound
   487→      m.ambientCd -= dt;
   488→      if (m.ambientCd <= 0 && distToPlayer < 18) {
   489→        m.ambientCd = 5 + Math.random() * 9;
   490→        audio.mobAmbient(m.def.sound, distToPlayer);
   491→      }
   492→
   493→      // ── AI ──
   494→      m.attackCd -= dt;
   495→      m.stateTimer -= dt;
   496→      let moveSpeed = 0;
   497→      let wantX = 0, wantZ = 0;
   498→
   499→      // creative players are invisible to hostile AI: treat hostiles as passive wanderers
   500→      const hostileActive = m.def.hostile && !cb.playerCreative;
   501→
   502→      if (!hostileActive) {
   503→        // passive
   504→        if (m.state === 'flee') {
   505→          const dx = m.x - player.x;
   506→          const dz = m.z - player.z;
   507→          const len = Math.hypot(dx, dz) || 1;
   508→          wantX = dx / len; wantZ = dz / len;
   509→          moveSpeed = m.def.speed * 1.7;
   510→          if (m.stateTimer <= 0) { m.state = 'idle'; m.stateTimer = 2 + Math.random() * 3; }
   511→        } else if (m.state === 'walk') {
   512→          const dx = m.wanderX - m.x;
   513→          const dz = m.wanderZ - m.z;
   514→          const len = Math.hypot(dx, dz);
   515→          if (len < 0.8 || m.stateTimer <= 0) {
   516→            m.state = 'idle';
   517→            m.stateTimer = 2 + Math.random() * 4;
   518→          } else {
   519→            wantX = dx / len; wantZ = dz / len;
   520→            moveSpeed = m.def.speed;
   521→          }
   522→        } else {
   523→          if (m.stateTimer <= 0) {
   524→            m.state = 'walk';
   525→            m.stateTimer = 4 + Math.random() * 5;
   526→            const a = Math.random() * Math.PI * 2;
   527→            const r = 3 + Math.random() * 6;
   528→            m.wanderX = m.x + Math.cos(a) * r;
   529→            m.wanderZ = m.z + Math.sin(a) * r;
   530→          }
   531→        }
   532→      } else if (m.type === 'zombie') {
   533→        // burn in sunlight
   534→        this.updateBurning(m, dt, sunLevel, cb);
   535→        if (distToPlayer < 24) {
   536→          m.state = 'chase';
   537→          const dx = player.x - m.x;
   538→          const dz = player.z - m.z;
   539→          const len = Math.hypot(dx, dz) || 1;
   540→          wantX = dx / len; wantZ = dz / len;
   541→          moveSpeed = m.def.speed;
   542→          if (distToPlayer < 1.7 && m.attackCd <= 0) {
   543→            m.attackCd = 1.1;
   544→            cb.damagePlayer(m.def.damage, m.x, m.z);
   545→            audio.zombieAttack();
   546→          }
   547→        } else {
   548→          wanderAI(m, dt);
   549→          if (m.state === 'walk') {
   550→            const dx = m.wanderX - m.x;
   551→            const dz = m.wanderZ - m.z;
   552→            const len = Math.hypot(dx, dz);
   553→            if (len < 0.8 || m.stateTimer <= 0) { m.state = 'idle'; m.stateTimer = 2 + Math.random() * 3; }
   554→            else { wantX = dx / len; wantZ = dz / len; moveSpeed = m.def.speed * 0.55; }
   555→          }
   556→        }
   557→      } else if (m.type === 'skeleton') {
   558→        this.updateBurning(m, dt, sunLevel, cb);
   559→        if (distToPlayer < 17) {
   560→          m.state = 'chase';
   561→          const dx = player.x - m.x;
   562→          const dz = player.z - m.z;
   563→          const len = Math.hypot(dx, dz) || 1;
   564→          // keep distance ~8
   565→          if (distToPlayer > 9.5) { wantX = dx / len; wantZ = dz / len; moveSpeed = m.def.speed; }
   566→          else if (distToPlayer < 6) { wantX = -dx / len; wantZ = -dz / len; moveSpeed = m.def.speed * 0.8; }
   567→          else {
   568→            // strafe
   569→            wantX = -dz / len * 0.6; wantZ = dx / len * 0.6;
   570→            moveSpeed = m.def.speed * 0.5;
   571→          }
   572→          if (m.attackCd <= 0 && distToPlayer < 15) {
   573→            m.attackCd = 2.2;
   574→            this.shootArrow(m, player);
   575→          }
   576→        } else {
   577→          wanderAI(m, dt);
   578→          if (m.state === 'walk') {
   579→            const dx = m.wanderX - m.x;
   580→            const dz = m.wanderZ - m.z;
   581→            const len = Math.hypot(dx, dz);
   582→            if (len < 0.8 || m.stateTimer <= 0) { m.state = 'idle'; m.stateTimer = 2 + Math.random() * 3; }
   583→            else { wantX = dx / len; wantZ = dz / len; moveSpeed = m.def.speed * 0.55; }
   584→          }
   585→        }
   586→      } else if (m.type === 'creeper') {
   587→        if (m.fuse >= 0) {
   588→          // fused: flash + wait
   589→          m.fuse += dt;
   590→          const flash = Math.sin(m.fuse * 22) > 0 ? 1 : 0;
   591→          tint(m.parts.materials, 1, 1 + flash * 1.2, 1 + flash * 1.2);
   592→          if (distToPlayer > 5) {
   593→            m.fuse = -1;
   594→            tint(m.parts.materials, 1, 1, 1);
   595→          } else if (m.fuse > 1.5) {
   596→            this.explode(m.x, m.y + 0.8, m.z, cb);
   597→            this.scene.remove(m.group);
   598→            this.scene.remove(m.parts.shadow);
   599→            this.mobs.splice(i, 1);
   600→            continue;
   601→          }
   602→        } else if (distToPlayer < 13) {
   603→          m.state = 'chase';
   604→          const dx = player.x - m.x;
   605→          const dz = player.z - m.z;
   606→          const len = Math.hypot(dx, dz) || 1;
   607→          wantX = dx / len; wantZ = dz / len;
   608→          moveSpeed = m.def.speed;
   609→          if (distToPlayer < 2.4) {
   610→            m.fuse = 0;
   611→            audio.fuseHiss();
   612→          }
   613→        } else {
   614→          wanderAI(m, dt);
   615→          if (m.state === 'walk') {
   616→            const dx = m.wanderX - m.x;
   617→            const dz = m.wanderZ - m.z;
   618→            const len = Math.hypot(dx, dz);
   619→            if (len < 0.8 || m.stateTimer <= 0) { m.state = 'idle'; m.stateTimer = 2 + Math.random() * 3; }
   620→            else { wantX = dx / len; wantZ = dz / len; moveSpeed = m.def.speed * 0.55; }
   621→          }
   622→        }
   623→      }
   624→
   625→      // ── movement + physics ──
   626→      const prevX = m.x, prevZ = m.z;
   627→      if (moveSpeed > 0) {
   628→        m.vx += (wantX * moveSpeed - m.vx) * Math.min(1, dt * 8);
   629→        m.vz += (wantZ * moveSpeed - m.vz) * Math.min(1, dt * 8);
   630→        m.targetYaw = Math.atan2(wantX, wantZ);
   631→      } else {
   632→        m.vx *= Math.pow(0.02, dt);
   633→        m.vz *= Math.pow(0.02, dt);
   634→      }
   635→      m.vy -= 32 * dt;
   636→      if (m.type === 'chicken') m.vy = Math.max(m.vy, -3.2);
   637→      moveEntity(this.world, m, dt);
   638→
   639→      // jump when blocked (or climb 1 block)
   640→      if (moveSpeed > 0 && m.onGround) {
   641→        const blockedXZ = Math.hypot(m.x - prevX, m.z - prevZ) < moveSpeed * dt * 0.3;
   642→        if (blockedXZ) {
   643→          m.vy = 8.4;
   644→        }
   645→      }
   646→      if (m.inWater) {
   647→        m.vy = Math.max(m.vy, 1.8); // swim up
   648→      }
   649→
   650→      // ── visuals ──
   651→      // smooth yaw
   652→      let dyaw = m.targetYaw - m.yaw;
   653→      while (dyaw > Math.PI) dyaw -= Math.PI * 2;
   654→      while (dyaw < -Math.PI) dyaw += Math.PI * 2;
   655→      m.yaw += dyaw * Math.min(1, dt * 8);
   656→      m.group.position.set(m.x, m.y, m.z);
   657→      m.group.rotation.y = m.yaw;
   658→
   659→      // walk animation
   660→      const hSpeed = Math.hypot(m.vx, m.vz);
   661→      m.walkPhase += hSpeed * dt * 3.2;
   662→      const swing = Math.sin(m.walkPhase * 2.4) * Math.min(1, hSpeed / m.def.speed) * 0.65;
   663→      const legs = m.parts.legs;
   664→      if (legs.length === 4) {
   665→        legs[0].rotation.x = swing;
   666→        legs[3].rotation.x = swing;
   667→        legs[1].rotation.x = -swing;
   668→        legs[2].rotation.x = -swing;
   669→      } else {
   670→        legs[0].rotation.x = swing;
   671→        if (legs[1]) legs[1].rotation.x = -swing;
   672→      }
   673→      // zombie arms forward
   674→      for (const arm of m.parts.arms) {
   675→        const pivot = (arm as unknown as { pivot?: THREE.Group }).pivot;
   676→        if (pivot) {
   677→          pivot.rotation.x = -Math.PI / 2 + Math.sin(m.walkPhase * 2.4) * 0.12;
   678→          pivot.rotation.z = Math.sin(m.walkPhase * 1.2) * 0.06;
   679→        }
   680→      }
   681→      // head bob
   682→      m.parts.head.rotation.y = Math.sin(this.time * 0.7 + m.walkPhase) * 0.14;
   683→
   684→      // shadow
   685→      m.parts.shadow.position.set(m.x, m.y + 0.03, m.z);
   686→      const shadowScale = m.onGround ? 1 : Math.max(0.4, 1 - Math.min(1, Math.abs(m.vy) * 0.06));
   687→      m.parts.shadow.scale.setScalar(shadowScale);
   688→
   689→      // burning visual
   690→      if (m.burning && Math.random() < dt * 12) {
   691→        cb.fireParticle(m.x + (Math.random() - 0.5) * 0.5, m.y + Math.random() * m.height, m.z + (Math.random() - 0.5) * 0.5);
   692→      }
   693→    }
   694→
   695→    // ── arrows ──
   696→    for (let i = this.arrows.length - 1; i >= 0; i--) {
   697→      const a = this.arrows[i];
   698→      a.life -= dt;
   699→      if (a.life <= 0 || a.stuck > 0.8) {
   700→        this.scene.remove(a.mesh);
   701→        this.arrows.splice(i, 1);
   702→        continue;
   703→      }
   704→      if (a.stuck > 0) {
   705→        a.stuck += dt;
   706→        continue;
   707→      }
   708→      a.vy -= 18 * dt;
   709→      const nx = a.x + a.vx * dt;
   710→      const ny = a.y + a.vy * dt;
   711→      const nz = a.z + a.vz * dt;
   712→      // block hit
   713→      const bid = this.world.getBlock(Math.floor(nx), Math.floor(ny), Math.floor(nz));
   714→      if (bid !== 0) {
   715→        a.stuck = 0.001;
   716→        continue;
   717→      }
   718→      a.x = nx; a.y = ny; a.z = nz;
   719→      // player hit
   720→      const px = player.x, py = player.y + player.height * 0.5, pz = player.z;
   721→      if (Math.hypot(a.x - px, a.y - py, a.z - pz) < 0.75) {
   722→        cb.damagePlayer(3, a.x - a.vx, a.z - a.vz);
   723→        a.life = 0;
   724→        continue;
   725→      }
   726→      a.mesh.position.set(a.x, a.y, a.z);
   727→      a.mesh.lookAt(a.x + a.vx, a.y + a.vy, a.z + a.vz);
   728→    }
   729→  }

   731→  private updateBurning(m: Mob, dt: number, sunLevel: number, _cb: MobCallbacks): void {
   732→    void _cb;
   733→    const lx = Math.floor(m.x);
   734→    const ly = Math.floor(m.y + 1);
   735→    const lz = Math.floor(m.z);
   736→    const light = this.world.getLight(lx, ly, lz);
   737→    const skyL = light < 0 ? 0 : light >> 4;
   738→    const inSunlight = skyL === 15 && sunLevel > 0.82;
   739→    if (inSunlight) {
   740→      m.burnTimer += dt;
   741→      m.burning = true;
   742→      if (m.burnTimer > 1) {
   743→        m.burnTimer = 0;
   744→        this.hurtMob(m, 1, 0, 0.01);
   745→      }
   746→    } else {
   747→      m.burning = false;
   748→      m.burnTimer = 0;
   749→    }
   750→  }

   752→  private shootArrow(m: Mob, player: AABBEntity & { eyeY(): number }): void {
   753→    const ox = m.x;
   754→    const oy = m.y + 1.5;
   755→    const oz = m.z;
   756→    const tx = player.x;
   757→    const ty = player.eyeY() - 0.2;
   758→    const tz = player.z;
   759→    const dx = tx - ox, dy = ty - oy, dz = tz - oz;
   760→    const dist = Math.hypot(dx, dy, dz);
   761→    const speed = 18;
   762→    const lead = dist / speed;
   763→    const vx = dx / dist * speed + (Math.random() - 0.5) * 1.2;
   764→    const vy = dy / dist * speed + lead * 9 + (Math.random() - 0.5) * 0.8;
   765→    const vz = dz / dist * speed + (Math.random() - 0.5) * 1.2;
   766→    const mesh = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.06, 0.55), this.arrowMat);
   767→    mesh.position.set(ox, oy, oz);
   768→    this.scene.add(mesh);
   769→    this.arrows.push({ x: ox, y: oy, z: oz, vx, vy, vz, life: 8, stuck: 0, mesh });
   770→    audio.bowShoot(dist);
   771→  }
   772→}

   774→function tint(mats: THREE.MeshLambertMaterial[], r: number, g: number, b: number): void {
   775→  for (const m of mats) m.color.setRGB(r, g, b);

   778→/** idle<->walk transitions for hostile mobs far from player */
   779→function wanderAI(m: Mob, _dt: number): void {
   780→  void _dt;
   781→  if (m.state === 'chase' || m.state === 'flee') {
   782→    m.state = 'idle';
   783→    m.stateTimer = 1 + Math.random() * 2;
   784→  }
   785→  if (m.state === 'idle' && m.stateTimer <= 0) {
   786→    m.state = 'walk';
   787→    m.stateTimer = 4 + Math.random() * 5;
   788→    const a = Math.random() * Math.PI * 2;
   789→    const r = 3 + Math.random() * 6;
   790→    m.wanderX = m.x + Math.cos(a) * r;
   791→    m.wanderZ = m.z + Math.sin(a) * r;
   792→  }
   793→}
