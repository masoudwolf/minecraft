   141→   141→  async newGame(): Promise<void> {
   142→   142→    const seed = Math.floor(Math.random() * 2147483647);
   143→   143→    this.setupWorld(seed, null);
   144→   144→    await this.preloadSpawn();

   146→   146→    useGameStore.getState().setHasSave(false);
   147→   147→    this.enterPlaying();

   150→   150→  async continueGame(): Promise<void> {
   151→   151→    const raw = localStorage.getItem(SAVE_KEY);
   152→   152→    const save: SaveData | null = raw ? JSON.parse(raw) : null;
   153→   153→    if (!save) { await this.newGame(); return; }
   154→   154→    this.setupWorld(save.seed, save);
   155→   155→    await this.preloadSpawn();
   156→   156→    this.enterPlaying();

   159→   159→  private setupWorld(seed: number, save: SaveData | null): void {
   160→   160→    // clear previous world if any
   161→   161→    if (this.world) {
   162→   162→      for (const key of Array.from(this.world.chunks.keys())) {
   163→   163→        const [cx, cz] = key.split(',').map(Number);
   164→   164→        this.world.unloadChunk(cx, cz);
   165→   165→      }
   166→   166→    }
   167→   167→    this.drops?.clear();
   168→   168→    this.mobs?.clear();
   169→   169→    this.world = new World(seed, save?.edits);
   170→   170→    this.scene.add(this.world.group);
   171→   171→    this.sky = new SkySystem(this.scene, seed);
   172→   172→    this.sky.time = save?.time ?? DAY_LENGTH * 0.3;
   173→   173→    this.drops = new DropManager(this.scene, this.world, getAtlas().texture);
   174→   174→    this.particles = new ParticleSystem(this.scene);
   175→   175→    this.mobs = new MobManager(this.scene, this.world);
   176→   176→    this.player = new Player(this.camera);
   177→   177→
   178→   178→    // spawn position
   179→   179→    let sx = 8, sz = 8;
   180→   180→    this.world.ensureChunk(0, 0);
   181→   181→    let sy = this.world.surfaceY(sx, sz);
   182→   182→    let attempts = 0;
   183→   183→    while ((sy <= 40 || sy === 0) && attempts < 24) {
   184→   184→      sx += 12; sz += 5; attempts++;
   185→   185→      this.world.ensureChunk(Math.floor(sx / CHUNK_SIZE), Math.floor(sz / CHUNK_SIZE));
   186→   186→      sy = this.world.surfaceY(sx, sz);
   187→   187→    }
   188→   188→    if (save) {
   189→   189→      this.player.entity.x = save.player.x;
   190→   190→      this.player.entity.y = save.player.y;
   191→   191→      this.player.entity.z = save.player.z;
   192→   192→      this.player.yaw = save.player.yaw;
   193→   193→      this.player.pitch = save.player.pitch;
   194→   194→      this.player.health = save.player.health;
   195→   195→      this.player.hotbar = save.player.hotbar;
   196→   196→      this.player.main = save.player.main && save.player.main.length === 27
   197→   197→        ? save.player.main
   198→   198→        : Array.from({ length: 27 }, () => ({ blockId: 0, count: 0 }));
   199→   199→      this.player.selected = save.player.selected;
   200→   200→      this.player.fallStartY = save.player.y;
   201→   201→    } else {
   202→   202→      this.player.entity.x = sx + 0.5;
   203→   203→      this.player.entity.y = sy + 1.2;
   204→   204→      this.player.entity.z = sz + 0.5;
   205→   205→      this.player.fallStartY = this.player.entity.y;
   206→   206→    }
   207→   207→    this.updateHandMesh(true);
   208→   208→    this.syncHUD(true);
   209→   209→    this.applySkyFog();
   210→   210→  }

   212→   212→  private async preloadSpawn(): Promise<void> {
   213→   213→    const store = useGameStore.getState();
   214→   214→    store.setScreen('loading');
   215→   215→    const pcx = Math.floor(this.player.entity.x / CHUNK_SIZE);
   216→   216→    const pcz = Math.floor(this.player.entity.z / CHUNK_SIZE);
   217→   217→    const dataR = 2;
   218→   218→    const jobs: [number, number][] = [];
   219→   219→    for (let dx = -dataR; dx <= dataR; dx++)
   220→   220→      for (let dz = -dataR; dz <= dataR; dz++)
   221→   221→        jobs.push([pcx + dx, pcz + dz]);
   222→   222→    jobs.sort((a, b) => (Math.abs(a[0] - pcx) + Math.abs(a[1] - pcz)) - (Math.abs(b[0] - pcx) + Math.abs(b[1] - pcz)));
   223→   223→
   224→   224→    for (let i = 0; i < jobs.length; i++) {
   225→   225→      this.world.ensureChunk(jobs[i][0], jobs[i][1]);
   226→   226→      if (i % 3 === 0) {
   227→   227→        const p = i / jobs.length;
   228→   228→        useGameStore.getState().setHud({ loadingProgress: p, loadingLabel: 'Generating world…' });
   229→   229→        await this.nextFrame();
   230→   230→      }
   231→   231→    }
   232→   232→    // mesh immediate area
   233→   233→    for (let dx = -1; dx <= 1; dx++) {
   234→   234→      for (let dz = -1; dz <= 1; dz++) {
   235→   235→        const c = this.world.getChunk(pcx + dx, pcz + dz);
   236→   236→        if (c) this.world.buildMesh(c);
   237→   237→      }
   238→   238→    }
   239→   239→    useGameStore.getState().setHud({ loadingProgress: 1, loadingLabel: 'Building terrain…' });
   240→   240→    await this.nextFrame();
   241→   241→  }

   271→   271→      seed: this.world.terrain.getSeed(),
   272→   272→      time: this.sky.time,
   273→   273→      player: {
   274→   274→        x: this.player.entity.x, y: this.player.entity.y, z: this.player.entity.z,
   275→   275→        yaw: this.player.yaw, pitch: this.player.pitch,

   277→   277→        hotbar: this.player.hotbar,
   278→   278→        main: this.player.main,
   279→   279→        selected: this.player.selected,

   297→   297→  private onKeyDown = (e: KeyboardEvent): void => {
   298→   298→    if (e.code === 'F3') { e.preventDefault(); useGameStore.getState().toggleDebug(); return; }
   299→   299→    const st = useGameStore.getState();
   300→   300→    if (st.inv.open && st.screen === 'playing') {
   301→   301→      if (e.code === 'KeyE' || e.code === 'Escape') { e.preventDefault(); this.closeInventory(); return; }
   302→   302→      if (e.code.startsWith('Digit')) {
   303→   303→        const n = parseInt(e.code.slice(5), 10);
   304→   304→        if (n >= 1 && n <= 9 && this.invHover) this.invHotbarSwap(this.invHover.area, this.invHover.idx, n - 1);
   305→   305→      }
   306→   306→      return;
   307→   307→    }
   308→   308→    if (st.screen !== 'playing') return;
   309→   309→    this.keys.add(e.code);
   310→   310→    if (e.code === 'KeyE') { e.preventDefault(); this.openInventory(false); return; }
   311→   311→    if (e.code.startsWith('Digit')) {
   312→   312→      const n = parseInt(e.code.slice(5), 10);
   313→   313→      if (n >= 1 && n <= 9) { this.player.selected = n - 1; this.syncHUD(); this.updateHandMesh(); }
   314→   314→    }
   315→   315→    if (e.code === 'KeyQ') this.dropSelected();
   316→   316→    if (e.code === 'Space') e.preventDefault();
   317→   317→  };

   323→   323→  private onMouseMove = (e: MouseEvent): void => {
   324→   324→    if (document.pointerLockElement !== this.canvas || useGameStore.getState().screen !== 'playing') return;
   325→   325→    const sens = 0.0022 * this.settings.sensitivity;
   326→   326→    const mx = Math.max(-60, Math.min(60, e.movementX));
   327→   327→    const my = Math.max(-60, Math.min(60, e.movementY));
   328→   328→    this.player.yaw -= mx * sens;
   329→   329→    this.player.pitch -= my * sens;
   330→   330→    this.player.pitch = Math.max(-Math.PI / 2 + 0.01, Math.min(Math.PI / 2 - 0.01, this.player.pitch));
   331→   331→  };

   336→   336→    audio.resume();
   337→   337→    if (e.button === 0) { this.mining = true; this.startSwing(); }
   338→   338→    else if (e.button === 2) { this.placeBlock(); }
   339→   339→    else if (e.button === 1) { e.preventDefault(); this.pickBlock(); }
   340→   340→  };

   395→   395→  private updateTarget(): void {
   396→   396→    const eye = new THREE.Vector3(this.player.x, this.player.eyeY(), this.player.z);
   397→   397→    const dir = this.player.forwardVector();
   398→   398→    this.target = raycast(this.world, eye.x, eye.y, eye.z, dir.x, dir.y, dir.z, 4.5);
   399→   399→    if (this.target) {
   400→   400→      this.highlight.visible = true;
   401→   401→      this.highlight.position.set(this.target.x + 0.5, this.target.y + 0.5, this.target.z + 0.5);
   402→   402→    } else {
   403→   403→      this.highlight.visible = false;
   404→   404→    }

   407→   407→  private mineTick(dt: number): void {
   408→   408→    const heldSlot = this.player.hotbar[this.player.selected];
   409→   409→    const heldTool = heldSlot && heldSlot.count > 0 && isItemId(heldSlot.blockId)
   410→   410→      ? getToolDef(heldSlot.blockId)
   411→   411→      : undefined;
   412→   412→
   413→   413→    // mob attack takes priority over mining
   414→   414→    if (this.mining && this.attackCooldown <= 0) {
   415→   415→      const eye = new THREE.Vector3(this.player.x, this.player.eyeY(), this.player.z);
   416→   416→      const dir = this.player.forwardVector();
   417→   417→      const hit = this.mobs.raycastMob(eye.x, eye.y, eye.z, dir.x, dir.y, dir.z, 3.4);
   418→   418→      if (hit) {
   419→   419→        this.attackCooldown = 0.42;
   420→   420→        this.startSwing();
   421→   421→        const kx = hit.mob.x - this.player.x;
   422→   422→        const kz = hit.mob.z - this.player.z;
   423→   423→        this.mobs.hurtMob(hit.mob, heldTool ? heldTool.dmg : 2, kx, kz);
   424→   424→        this.particles.hurt(hit.mob.x, hit.mob.y, hit.mob.z);
   425→   425→        this.damageTool(heldTool && heldTool.type !== 'sword' ? 2 : 1);
   426→   426→        this.crackMesh.visible = false;
   427→   427→        return;
   428→   428→      }
   429→   429→    }
   430→   430→    if (!this.mining || !this.target) { this.crackMesh.visible = false; return; }
   431→   431→    // target changed? reset progress
   432→   432→    const t = this.target;
   433→   433→    if (!this.mineTarget || this.mineTarget.x !== t.x || this.mineTarget.y !== t.y || this.mineTarget.z !== t.z) {
   434→   434→      this.mineTarget = t;
   435→   435→      this.mineProgress = 0;
   436→   436→    }
   437→   437→    const def = getBlockDef(t.id);
   438→   438→    if (!def) { this.crackMesh.visible = false; return; }
   439→   439→    const { time, harvest } = breakInfo(def, heldTool);
   440→   440→    if (!Number.isFinite(time)) { this.crackMesh.visible = false; return; }
   441→   441→    this.mineProgress += dt / time;
   442→   442→
   443→   443→    // dig sound + swing loop
   444→   444→    this.stepTimer += dt;
   445→   445→    if (this.stepTimer > 0.24) {
   446→   446→      this.stepTimer = 0;
   447→   447→      audio.dig((def.sound ?? 'stone') as MaterialSound);
   448→   448→      this.startSwing();
   449→   449→    }
   450→   450→
   451→   451→    if (this.mineProgress >= 1) {
   452→   452→      // break!
   453→   453→      const col = tileAvgColor(Array.isArray(def.tiles) ? def.tiles[2] : def.tiles);
   454→   454→      this.particles.burstBlockBreak(t.x, t.y, t.z, col);
   455→   455→      audio.breakBlock((def.sound ?? 'stone') as MaterialSound);
   456→   456→      const dropId = def.drop === undefined ? t.id : def.drop;
   457→   457→      if (harvest && dropId && dropId > 0) {
   458→   458→        this.drops.spawn(dropId, t.x + 0.5, t.y + 0.3, t.z + 0.5, 1);
   459→   459→      }
   460→   460→      this.damageTool(1);
   461→   461→      this.world.setBlock(t.x, t.y, t.z, BLOCK.AIR);
   462→   462→      this.mineProgress = 0;
   463→   463→      this.mineTarget = null;
   464→   464→      this.crackMesh.visible = false;
   465→   465→      return;
   466→   466→    }
   467→   467→    // crack overlay
   468→   468→    const stage = Math.min(9, Math.floor(this.mineProgress * 10));
   469→   469→    this.crackMesh.material = this.crackMats[stage];
   470→   470→    this.crackMesh.visible = true;
   471→   471→    this.crackMesh.position.set(t.x + 0.5, t.y + 0.5, t.z + 0.5);
   472→   472→  }

   474→   474→  private placeBlock(): void {
   475→   475→    if (!this.target || this.placeCooldown > 0) return;
   476→   476→    // right-click on crafting table opens the 3x3 grid (unless sneaking)
   477→   477→    const targetId = this.world.getBlock(this.target.x, this.target.y, this.target.z);
   478→   478→    if (targetId === BLOCK.CRAFTING_TABLE && !this.player.sneaking) {

   482→   482→    const slot = this.player.hotbar[this.player.selected];
   483→   483→    if (!slot || slot.blockId === 0 || slot.count <= 0) return;
   484→   484→    // eating food items
   485→   485→    if (isItemId(slot.blockId)) {
   486→   486→      const itemDef = getItemDef(slot.blockId);
   487→   487→      if (itemDef?.food && this.player.hunger < 19.6 && this.eatCooldown <= 0) {
   488→   488→        this.eatCooldown = 1.4;
   489→   489→        this.player.hunger = Math.min(20, this.player.hunger + itemDef.food);
   490→   490→        audio.eat();

   492→   492→        this.startSwing();
   493→   493→        slot.count--;
   494→   494→        if (slot.count <= 0) { slot.blockId = 0; slot.count = 0; }
   495→   495→        this.syncHUD();
   496→   496→        this.updateHandMesh();
   497→   497→      }
   498→   498→      return;
   499→   499→    }
   500→   500→    const bx = this.target.x + this.target.nx;
   501→   501→    const by = this.target.y + this.target.ny;
   502→   502→    const bz = this.target.z + this.target.nz;
   503→   503→    if (by < 0 || by >= WORLD_HEIGHT) return;
   504→   504→    const existing = this.world.getBlock(bx, by, bz);
   505→   505→    if (existing !== BLOCK.AIR && !isLiquid(existing)) return;
   506→   506→    // don't place inside player
   507→   507→    if (aabbIntersectsBlock(this.player.entity, bx, by, bz)) return;
   508→   508→    const def = getBlockDef(slot.blockId);
   509→   509→    this.world.setBlock(bx, by, bz, slot.blockId);
   510→   510→    audio.place((def?.sound ?? 'stone') as MaterialSound);
   511→   511→    this.placeCooldown = 0.22;
   512→   512→    this.startSwing();
   513→   513→    slot.count--;
   514→   514→    if (slot.count <= 0) { slot.blockId = 0; slot.count = 0; }
   515→   515→    this.syncHUD();
   516→   516→    this.updateHandMesh();
   517→   517→  }

   649→   649→  private clickSlot(list: HotbarSlot[], idx: number, button: 'left' | 'right'): void {
   650→   650→    const slot = list[idx];
   651→   651→    const cur = this.cursor;
   652→   652→    if (button === 'right') {
   653→   653→      if (!cur) {
   654→   654→        if (isEmptySlot(slot)) return;
   655→   655→        const half = Math.ceil(slot.count / 2);
   656→   656→        this.cursor = { ...slot, count: half };
   657→   657→        slot.count -= half;
   658→   658→        if (slot.count <= 0) list[idx] = emptySlot();
   659→   659→      } else {
   660→   660→        if (isEmptySlot(slot)) {
   661→   661→          list[idx] = { blockId: cur.blockId, count: 1, dur: cur.dur };
   662→   662→          cur.count--;
   663→   663→        } else if (slot.blockId === cur.blockId && slot.count < maxStack(slot.blockId)) {
   664→   664→          slot.count++;
   665→   665→          cur.count--;
   666→   666→        } else return;
   667→   667→        if (cur.count <= 0) this.cursor = null;
   668→   668→      }
   669→   669→      return;
   670→   670→    }
   671→   671→    // left click
   672→   672→    if (!cur) {
   673→   673→      if (isEmptySlot(slot)) return;
   674→   674→      this.cursor = { ...slot };
   675→   675→      list[idx] = emptySlot();
   676→   676→    } else if (isEmptySlot(slot)) {
   677→   677→      list[idx] = { ...cur };
   678→   678→      this.cursor = null;
   679→   679→    } else if (slot.blockId === cur.blockId && !isToolItem(slot.blockId)) {
   680→   680→      const max = maxStack(slot.blockId);
   681→   681→      const take = Math.min(cur.count, max - slot.count);
   682→   682→      slot.count += take;
   683→   683→      cur.count -= take;
   684→   684→      if (cur.count <= 0) this.cursor = null;
   685→   685→    } else {
   686→   686→      list[idx] = { ...cur };
   687→   687→      this.cursor = { ...slot };
   688→   688→    }
   689→   689→  }

   691→   691→  private shiftMove(area: 'hotbar' | 'main' | 'craft', list: HotbarSlot[], idx: number): void {
   692→   692→    const slot = list[idx];
   693→   693→    if (isEmptySlot(slot)) return;
   694→   694→    let left: number;
   695→   695→    if (area === 'hotbar') {
   696→   696→      left = addToSlots(this.player.main, slot.blockId, slot.count, slot.dur);
   697→   697→    } else {
   698→   698→      // main/craft -> hotbar first, then main (craft items go home)
   699→   699→      left = addToSlots(this.player.hotbar, slot.blockId, slot.count, slot.dur);
   700→   700→      if (left > 0 && area === 'main') left = addToSlots(this.player.main, slot.blockId, left, slot.dur);
   701→   701→    }
   702→   702→    if (left <= 0) list[idx] = emptySlot();
   703→   703→    else slot.count = left;
   704→   704→  }

   706→   706→  private updateCraftOut(): void {
   707→   707→    const grid = this.craftGrid;
   708→   708→    const size = this.invTable ? 3 : 2;
   709→   709→    const ids = grid.map((s) => (isEmptySlot(s) ? 0 : s.blockId));
   710→   710→    const res = matchRecipe(ids, size);
   711→   711→    this.craftOut = res
   712→   712→      ? { blockId: res.id, count: res.count, dur: freshDur(res.id) }
   713→   713→      : null;
   714→   714→  }

   716→   716→  private takeCraftOutput(shift: boolean): void {
   717→   717→    if (!this.craftOut) return;
   718→   718→    const grid = this.craftGrid;
   719→   719→    const consume = (): void => {
   720→   720→      for (let i = 0; i < grid.length; i++) {
   721→   721→        const s = grid[i];
   722→   722→        if (isEmptySlot(s)) continue;
   723→   723→        s.count--;
   724→   724→        if (s.count <= 0) grid[i] = emptySlot();
   725→   725→      }
   726→   726→    };
   727→   727→    if (!shift) {
   728→   728→      const out = this.craftOut;
   729→   729→      if (!this.cursor) {
   730→   730→        this.cursor = { ...out };
   731→   731→        consume();
   732→   732→      } else if (this.cursor.blockId === out.blockId && !isToolItem(out.blockId) && this.cursor.count + out.count <= maxStack(out.blockId)) {
   733→   733→        this.cursor.count += out.count;
   734→   734→        consume();
   735→   735→      } else return;
   736→   736→    } else {
   737→   737→      let guard = 0;
   738→   738→      while (guard++ < 64) {
   739→   739→        const size = this.invTable ? 3 : 2;
   740→   740→        const res = matchRecipe(grid.map((s) => (isEmptySlot(s) ? 0 : s.blockId)), size);
   741→   741→        if (!res) break;
   742→   742→        const left = this.addToInventory(res.id, res.count, freshDur(res.id));
   743→   743→        if (left > 0) {
   744→   744→          const p = this.player.entity;
   745→   745→          this.drops.spawn(res.id, p.x, p.y + 1, p.z, left);
   746→   746→        }
   747→   747→        consume();
   748→   748→        audio.pop();
   749→   749→      }
   750→   750→    }
   751→   751→    this.updateCraftOut();
   752→   752→    this.syncInventory(true);
   753→   753→    this.syncHUD(true);
   754→   754→  }

   843→   843→  private streamChunks(): void {
   844→   844→    const R = this.settings.renderDistance;
   845→   845→    const pcx = Math.floor(this.player.entity.x / CHUNK_SIZE);
   846→   846→    const pcz = Math.floor(this.player.entity.z / CHUNK_SIZE);
   847→   847→
   848→   848→    const t0 = performance.now();
   849→   849→    // generate data (spiral by distance), budget ~6ms
   850→   850→    outer:
   851→   851→    for (let r = 0; r <= R + 1; r++) {
   852→   852→      for (let dx = -r; dx <= r; dx++) {
   853→   853→        for (let dz = -r; dz <= r; dz++) {
   854→   854→          if (Math.max(Math.abs(dx), Math.abs(dz)) !== r) continue;
   855→   855→          const key = chunkKey(pcx + dx, pcz + dz);
   856→   856→          if (!this.world.chunks.has(key)) {
   857→   857→            this.world.ensureChunk(pcx + dx, pcz + dz);
   858→   858→            if (performance.now() - t0 > 6) break outer;
   859→   859→          }
   860→   860→        }
   861→   861→      }
   862→   862→    }
   863→   863→
   864→   864→    // mesh chunks with all 4 neighbors' data, budget ~7ms
   865→   865→    const t1 = performance.now();
   866→   866→    outer2:
   867→   867→    for (let r = 0; r <= R; r++) {
   868→   868→      for (let dx = -r; dx <= r; dx++) {
   869→   869→        for (let dz = -r; dz <= r; dz++) {
   870→   870→          if (Math.max(Math.abs(dx), Math.abs(dz)) !== r) continue;
   871→   871→          const chunk = this.world.getChunk(pcx + dx, pcz + dz);
   872→   872→          if (chunk && chunk.hasData && chunk.needsMesh) {
   873→   873→            const n = this.world.getChunk(pcx + dx + 1, pcz + dz);
   874→   874→            const s = this.world.getChunk(pcx + dx - 1, pcz + dz);
   875→   875→            const e = this.world.getChunk(pcx + dx, pcz + dz + 1);
   876→   876→            const w = this.world.getChunk(pcx + dx, pcz + dz - 1);
   877→   877→            if (n?.hasData && s?.hasData && e?.hasData && w?.hasData) {
   878→   878→              this.world.buildMesh(chunk);
   879→   879→              if (performance.now() - t1 > 7) break outer2;
   880→   880→            }
   881→   881→          }
   882→   882→        }
   883→   883→      }
   884→   884→    }
   885→   885→
   886→   886→    // unload far chunks occasionally
   887→   887→    if ((this.saveTimer % 5) < 0.016) {
   888→   888→      for (const key of Array.from(this.world.chunks.keys())) {
   889→   889→        const [cx, cz] = key.split(',').map(Number);
   890→   890→        if (Math.max(Math.abs(cx - pcx), Math.abs(cz - pcz)) > R + 3) {
   891→   891→          this.world.unloadChunk(cx, cz);
   892→   892→        }
   893→   893→      }
   894→   894→    }
   895→   895→  }

   969→   969→  private physicsStep(step: number): void {
   970→   970→    const p = this.player;
   971→   971→    // input direction
   972→   972→    let forward = 0, strafe = 0;
   973→   973→    if (this.keys.has('KeyW')) forward += 1;
   974→   974→    if (this.keys.has('KeyS')) forward -= 1;
   975→   975→    if (this.keys.has('KeyA')) strafe -= 1;
   976→   976→    if (this.keys.has('KeyD')) strafe += 1;
   977→   977→    const wishJump = this.keys.has('Space');
   978→   978→    const wishSneak = this.keys.has('ShiftLeft') || this.keys.has('ShiftRight');
   979→   979→    p.sprinting = (this.keys.has('ControlLeft') || this.keys.has('ControlRight')) && forward > 0;
   980→   980→
   981→   981→    const wasInWater = p.entity.inWater;
   982→   982→    const wasOnGround = p.entity.onGround;
   983→   983→    const prevVy = p.entity.vy;
   984→   984→
   985→   985→    p.moveInput({ forward, strafe }, this.world, step, wishJump, wishSneak);
   986→   986→
   987→   987→    // footsteps
   988→   988→    if (p.entity.onGround && p.stepDistance > 2.1) {
   989→   989→      p.stepDistance = 0;
   990→   990→      const below = this.world.getBlock(Math.floor(p.x), Math.floor(p.y - 0.4), Math.floor(p.z));
   991→   991→      const def = getBlockDef(below);
   992→   992→      if (def) audio.step((def.sound ?? 'stone') as MaterialSound);
   993→   993→    }
   994→   994→    // splash
   995→   995→    if (!wasInWater && p.entity.inWater && prevVy < -4) {
   996→   996→      audio.splash();
   997→   997→      this.particles.splash(p.x, p.y + 0.2, p.z);
   998→   998→    }
   999→   999→    // landing particles
  1000→  1000→    if (!wasOnGround && p.entity.onGround && this.prevVy < -7) {
  1001→  1001→      const below = this.world.getBlock(Math.floor(p.x), Math.floor(p.y - 0.4), Math.floor(p.z));
  1002→  1002→      const def = getBlockDef(below);
  1003→  1003→      if (def) {
  1004→  1004→        const col = tileAvgColor(Array.isArray(def.tiles) ? def.tiles[2] : def.tiles);
  1005→  1005→        this.particles.burstLand(p.x, p.y, p.z, col, Math.min(14, Math.floor(-this.prevVy / 2)));
  1006→  1006→      }
  1007→  1007→    }
  1008→  1008→    this.wasInWater = p.entity.inWater;
  1009→  1009→    this.wasOnGround = p.entity.onGround;
  1010→  1010→    this.prevVy = p.entity.vy;
  1011→  1011→
  1012→  1012→    // hurt cooldown decay
  1013→  1013→    if (p.hurtCooldown > 0) p.hurtCooldown -= step;
  1014→  1014→
  1015→  1015→    // ── hunger drain ──
  1016→  1016→    if (p.sprinting && (forward !== 0 || strafe !== 0)) {
  1017→  1017→      p.hunger = Math.max(0, p.hunger - 0.085 * step);
  1018→  1018→    } else if (forward !== 0 || strafe !== 0) {
  1019→  1019→      p.hunger = Math.max(0, p.hunger - 0.012 * step);
  1020→  1020→    } else {
  1021→  1021→      p.hunger = Math.max(0, p.hunger - 0.0015 * step);
  1022→  1022→    }
  1023→  1023→    if (wishJump && p.onGround) p.hunger = Math.max(0, p.hunger - 0.05);
  1024→  1024→
  1025→  1025→    // void damage
  1026→  1026→    if (p.y < -8) {
  1027→  1027→      p.damage(4);
  1028→  1028→      p.entity.vy = 0;
  1029→  1029→      p.entity.y = -8;
  1030→  1030→    }
  1031→  1031→
  1032→  1032→    // death
  1033→  1033→    if (p.dead) {
  1034→  1034→      useGameStore.getState().setScreen('dead');
  1035→  1035→      if (document.pointerLockElement === this.canvas) document.exitPointerLock();
  1036→  1036→    }
  1037→  1037→  }
