   108→// ─── Model builders (vanilla textures + MC box-UV mapping) ───────────────────
   109→/** Build a box mesh textured from a vanilla skin part via standard MC box UVs.
   110→ *  Cutout (alphaTest) so vanilla transparent texels (ribcage gaps, fleece face)
   111→ *  render as holes instead of black; DoubleSide so thin parts show both sides. */
   112→function boxPart(p: MobSkinPart, w: number, h: number, d: number, tag: string, opts?: BoxUVOptions): THREE.Mesh {
   113→  const geo = new THREE.BoxGeometry(w, h, d);
   114→  boxUV(geo, p.lay, p.texW, p.texH, opts);
   115→  const mesh = new THREE.Mesh(geo, new THREE.MeshLambertMaterial({ map: p.tex, alphaTest: 0.35, side: THREE.DoubleSide }));
   116→  mesh.userData.part = tag;
   117→  return mesh;
   118→}

   120→/** Collect a mesh's material(s) into the parts list (per-instance tint/light). */
   121→function collectMats(list: THREE.MeshLambertMaterial[], mesh: THREE.Mesh): void {
   122→  const m = mesh.material as THREE.MeshLambertMaterial | THREE.MeshLambertMaterial[];
   123→  if (Array.isArray(m)) list.push(...m);
   124→  else list.push(m);
   125→}

   127→/**
   128→ * Leg with a proper HIP PIVOT (Minecraft-style limb rigging): the pivot Group
   129→ * sits at the hip joint (top of the leg) and the leg box hangs BELOW it, so
   130→ * animating pivot.rotation.x swings the leg from the top — like MC. `furPart`
   131→ * optionally adds an inflated fleece box (sheep) that rides the swing; like
   132→ * vanilla, the fleece leg wraps only the UPPER HALF of the leg.
   133→ */
   134→function legPivot(p: MobSkinPart, w: number, h: number, d: number, hipX: number, hipY: number, hipZ: number, side = 0, furPart?: { part: MobSkinPart; inflate: number }): THREE.Group {
   135→  const pivot = new THREE.Group();
   136→  pivot.position.set(hipX, hipY, hipZ);
   137→  pivot.userData.side = side;
   138→  const leg = boxPart(p, w, h, d, 'limb');
   139→  leg.position.y = -h / 2; // hang from the hip
   140→  pivot.add(leg);
   141→  if (furPart) {
   142→    const f = furPart.inflate;
   143→    const furH = h * 0.5 + f * 2;
   144→    const fur = boxPart(furPart.part, w + f * 2, furH, d + f * 2, 'fur');
   145→    leg.add(fur);
   146→    fur.position.y = h / 2 - furH / 2 + f * 0.5; // top-aligned wrap
   147→  }
   148→  return pivot;
   149→}

   151→const solidTexCache = new Map<string, THREE.Texture>();
   152→/** Tiny solid-color texture — lets plain-colored parts (bow, stick arms) ride
   153→ *  the shared map+white-color material pipeline (world-light tinting safe). */
   154→function solidTex(hex: string): THREE.Texture {
   155→  const hit = solidTexCache.get(hex);
   156→  if (hit) return hit;
   157→  const c = document.createElement('canvas');
   158→  c.width = 2; c.height = 2;
   159→  const ctx = c.getContext('2d')!;
   160→  ctx.fillStyle = hex;
   161→  ctx.fillRect(0, 0, 2, 2);
   162→  const t = new THREE.CanvasTexture(c);
   163→  t.magFilter = THREE.NearestFilter;
   164→  t.minFilter = THREE.NearestFilter;
   165→  t.generateMipmaps = false;
   166→  t.colorSpace = THREE.SRGBColorSpace;
   167→  solidTexCache.set(hex, t);
   168→  return t;
   169→}

   171→function cutoutMat(hex: string): THREE.MeshLambertMaterial {
   172→  return new THREE.MeshLambertMaterial({ map: solidTex(hex), alphaTest: 0.35, side: THREE.DoubleSide });
   173→}

   175→/** Simple boxy Minecraft-style bow: curved limbs + string. Shoots toward the
   176→ *  belly side (-X of the group); attach with rotation.y = π/2 to shoot +Z. */
   177→function buildBow(): THREE.Group {
   178→  const g = new THREE.Group();
   179→  const wood = cutoutMat('#6b4d2a');
   180→  const woodLight = cutoutMat('#8a683c');
   181→  const str = cutoutMat('#e8e8e8');
   182→  const grip = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.2, 0.06), wood);
   183→  g.add(grip);
   184→  const upper = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.24, 0.05), woodLight);
   185→  upper.position.set(0.05, 0.2, 0);
   186→  upper.rotation.z = 0.5;
   187→  g.add(upper);
   188→  const lower = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.24, 0.05), woodLight);
   189→  lower.position.set(0.05, -0.2, 0);
   190→  lower.rotation.z = -0.5;
   191→  g.add(lower);
   192→  const string = new THREE.Mesh(new THREE.BoxGeometry(0.015, 0.5, 0.015), str);
   193→  string.position.set(0.1, 0, 0);
   194→  g.add(string);
   195→  return g;
   196→}

   198→function makeShadow(radius: number, scene: THREE.Scene): THREE.Mesh {
   199→  const geo = new THREE.CircleGeometry(radius, 12);
   200→  const mat = new THREE.MeshBasicMaterial({ color: 0x000000, transparent: true, opacity: 0.28, depthWrite: false });
   201→  const mesh = new THREE.Mesh(geo, mat);
   202→  mesh.rotation.x = -Math.PI / 2;
   203→  mesh.renderOrder = 1;
   204→  scene.add(mesh);
   205→  return mesh;
   206→}

   208→function quadruped(skins: MobSkins, opts: {
   209→  bodyW: number; bodyH: number; bodyD: number; bodyY: number;
   210→  legW: number; legH: number;
   211→  headS: number; headY: number; headZ: number;
   212→  shadowR: number;
   213→  fur?: { head: MobSkinPart; body: MobSkinPart; limb: MobSkinPart; inflate: number };
   214→}): MobParts {
   215→  const group = new THREE.Group();
   216→  const mats: THREE.MeshLambertMaterial[] = [];
   217→  const body = boxPart(skins.body, opts.bodyW, opts.bodyH, opts.bodyD, 'body');
   218→  body.position.y = opts.bodyY;
   219→  group.add(body);
   220→  if (opts.fur) {
   221→    // fleece layer as a CHILD of the body — rides along, inflates by ~MC 1.75u
   222→    const f = opts.fur.inflate;
   223→    const furBody = boxPart(opts.fur.body, opts.bodyW + f * 2, opts.bodyH + f * 2, opts.bodyD + f * 2, 'fur');
   224→    body.add(furBody);
   225→    furBody.position.set(0, 0, 0);
   226→  }
   227→  const head = boxPart(skins.head, opts.headS, opts.headS, opts.headS, 'head');
   228→  head.position.set(0, opts.headY, opts.headZ);
   229→  group.add(head);
   230→  if (opts.fur) {
   231→    const f = opts.fur.inflate * 0.6;
   232→    const furHead = boxPart(opts.fur.head, opts.headS + f * 2, opts.headS + f * 2, opts.headS + f * 2, 'fur',
   233→      // fleece FRONT maps to a transparent texel rect → the sheep's real face
   234→      // (eyes/muzzle) shows through the wool, like vanilla
   235→      { frontTransparentRect: [26, 1, 6, 6] });
   236→    head.add(furHead);
   237→    furHead.position.set(0, 0, 0);
   238→  }
   239→  const legs: THREE.Mesh[] = [];
   240→  const lx = opts.bodyW / 2 - opts.legW / 2;
   241→  const lz = opts.bodyD / 2 - opts.legW / 2;
   242→  // MC-style hip: the pivot sits INSIDE the body (leg top hidden), so the leg
   243→  // visibly swings from its top — no gap opens at the hip while walking.
   244→  const hipInset = opts.bodyH * 0.3;
   245→  const hipY = opts.bodyY - opts.bodyH / 2 + hipInset;
   246→  for (const [sx, sz] of [[-1, -1], [1, -1], [-1, 1], [1, 1]]) {
   247→    const pivot = legPivot(skins.limb, opts.legW, opts.legH + hipInset, opts.legW, sx * lx, hipY, sz * lz, sx, opts.fur ? { part: opts.fur.limb, inflate: opts.fur.inflate * 0.5 } : undefined);
   248→    group.add(pivot);
   249→    legs.push(pivot as unknown as THREE.Mesh);
   250→  }
   251→  for (const mesh of [body, head]) collectMats(mats, mesh);
   252→  for (const pv of legs) for (const child of (pv as unknown as THREE.Group).children) collectMats(mats, child as THREE.Mesh);
   253→  return { group, head, legs, arms: [], materials: mats, shadow: null as unknown as THREE.Mesh };
   254→}

   256→function humanoid(skins: MobSkins, thin = false): MobParts {
   257→  const lw = thin ? 0.16 : 0.25;
   258→  const aw = thin ? 0.14 : 0.22;
   259→  const group = new THREE.Group();
   260→  const mats: THREE.MeshLambertMaterial[] = [];
   261→  const body = boxPart(skins.body, 0.5, 0.72, 0.26, 'body');
   262→  body.position.y = 1.1;
   263→  group.add(body);
   264→  const head = boxPart(skins.head, 0.5, 0.5, 0.5, 'head');
   265→  head.position.y = 1.72;
   266→  group.add(head);
   267→  const legs: THREE.Mesh[] = [];
   268→  for (const sx of [-1, 1]) {
   269→    const pivot = legPivot(skins.limb, lw, 0.74, lw, sx * 0.125, 0.74, 0, sx);
   270→    group.add(pivot);
   271→    legs.push(pivot as unknown as THREE.Mesh);
   272→  }
   273→  const arms: THREE.Mesh[] = [];
   274→  const armPart = skins.limb2 ?? skins.limb;
   275→  for (const sx of [-1, 1]) {
   276→    const pivot = new THREE.Group();
   277→    pivot.position.set(sx * (0.25 + aw / 2), 1.42, 0);
   278→    const arm = boxPart(armPart, aw, 0.7, aw, 'limb');
   279→    arm.position.y = -0.32;
   280→    pivot.add(arm);
   281→    group.add(pivot);
   282→    arms.push(arm);
   283→    (arm as unknown as { limbPivot: THREE.Group }).limbPivot = pivot;
   284→  }
   285→  for (const mesh of [body, head]) collectMats(mats, mesh);
   286→  for (const pv of legs) for (const child of (pv as unknown as THREE.Group).children) collectMats(mats, child as THREE.Mesh);
   287→  for (const arm of arms) collectMats(mats, arm);
   288→  return { group, head, legs, arms, materials: mats, shadow: null as unknown as THREE.Mesh };
   289→}
