    10→   109→/** Build a box mesh textured from a vanilla skin part via standard MC box UVs.
    11→   110→ *  Cutout (alphaTest) so vanilla transparent texels (ribcage gaps, fleece face)
    12→   111→ *  render as holes instead of black; DoubleSide so thin parts show both sides. */
    13→   112→function boxPart(p: MobSkinPart, w: number, h: number, d: number, tag: string, opts?: BoxUVOptions): THREE.Mesh {
    14→   113→  const geo = new THREE.BoxGeometry(w, h, d);
    15→   114→  boxUV(geo, p.lay, p.texW, p.texH, opts);
    16→   115→  const mesh = new THREE.Mesh(geo, new THREE.MeshLambertMaterial({ map: p.tex, alphaTest: 0.35, side: THREE.DoubleSide }));
    17→   116→  mesh.userData.part = tag;
    18→   117→  return mesh;
    19→   118→}

    22→   121→function collectMats(list: THREE.MeshLambertMaterial[], mesh: THREE.Mesh): void {
    23→   122→  const m = mesh.material as THREE.MeshLambertMaterial | THREE.MeshLambertMaterial[];
    24→   123→  if (Array.isArray(m)) list.push(...m);
    25→   124→  else list.push(m);

    28→   127→/**
    29→   128→ * Leg with a proper HIP PIVOT (Minecraft-style limb rigging): the pivot Group
    30→   129→ * sits at the hip joint (top of the leg) and the leg box hangs BELOW it, so
    31→   130→ * animating pivot.rotation.x swings the leg from the top — like MC. `furPart`
    32→   131→ * optionally adds an inflated fleece box (sheep) that rides the swing; like
    33→   132→ * vanilla, the fleece leg wraps only the UPPER HALF of the leg.
    34→   133→ */
    35→   134→function legPivot(p: MobSkinPart, w: number, h: number, d: number, hipX: number, hipY: number, hipZ: number, side = 0, furPart?: { part: MobSkinPart; inflate: number }): THREE.Group {
    36→   135→  const pivot = new THREE.Group();
    37→   136→  pivot.position.set(hipX, hipY, hipZ);
    38→   137→  pivot.userData.side = side;
    39→   138→  const leg = boxPart(p, w, h, d, 'limb');
    40→   139→  leg.position.y = -h / 2; // hang from the hip
    41→   140→  pivot.add(leg);
    42→   141→  if (furPart) {
    43→   142→    const f = furPart.inflate;
    44→   143→    const furH = h * 0.5 + f * 2;
    45→   144→    const fur = boxPart(furPart.part, w + f * 2, furH, d + f * 2, 'fur');
    46→   145→    leg.add(fur);
    47→   146→    fur.position.y = h / 2 - furH / 2 + f * 0.5; // top-aligned wrap
    48→   147→  }
    49→   148→  return pivot;
    50→   149→}

    52→   151→const solidTexCache = new Map<string, THREE.Texture>();
    53→   152→/** Tiny solid-color texture — lets plain-colored parts (bow, stick arms) ride
    54→   153→ *  the shared map+white-color material pipeline (world-light tinting safe). */
    55→   154→function solidTex(hex: string): THREE.Texture {
    56→   155→  const hit = solidTexCache.get(hex);
    57→   156→  if (hit) return hit;
    58→   157→  const c = document.createElement('canvas');
    59→   158→  c.width = 2; c.height = 2;
    60→   159→  const ctx = c.getContext('2d')!;
    61→   160→  ctx.fillStyle = hex;
    62→   161→  ctx.fillRect(0, 0, 2, 2);
    63→   162→  const t = new THREE.CanvasTexture(c);
    64→   163→  t.magFilter = THREE.NearestFilter;
    65→   164→  t.minFilter = THREE.NearestFilter;
    66→   165→  t.generateMipmaps = false;
    67→   166→  t.colorSpace = THREE.SRGBColorSpace;
    68→   167→  solidTexCache.set(hex, t);
    69→   168→  return t;
    70→   169→}

    72→   171→function cutoutMat(hex: string): THREE.MeshLambertMaterial {
    73→   172→  return new THREE.MeshLambertMaterial({ map: solidTex(hex), alphaTest: 0.35, side: THREE.DoubleSide });

    76→   175→/** Simple boxy Minecraft-style bow: curved limbs + string. Shoots toward the
    77→   176→ *  belly side (-X of the group); attach with rotation.y = π/2 to shoot +Z. */
    78→   177→function buildBow(): THREE.Group {
    79→   178→  const g = new THREE.Group();
    80→   179→  const wood = cutoutMat('#6b4d2a');
    81→   180→  const woodLight = cutoutMat('#8a683c');
    82→   181→  const str = cutoutMat('#e8e8e8');
    83→   182→  const grip = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.2, 0.06), wood);
    84→   183→  g.add(grip);
    85→   184→  const upper = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.24, 0.05), woodLight);
    86→   185→  upper.position.set(0.05, 0.2, 0);
    87→   186→  upper.rotation.z = 0.5;
    88→   187→  g.add(upper);
    89→   188→  const lower = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.24, 0.05), woodLight);
    90→   189→  lower.position.set(0.05, -0.2, 0);
    91→   190→  lower.rotation.z = -0.5;
    92→   191→  g.add(lower);
    93→   192→  const string = new THREE.Mesh(new THREE.BoxGeometry(0.015, 0.5, 0.015), str);
    94→   193→  string.position.set(0.1, 0, 0);
    95→   194→  g.add(string);
    96→   195→  return g;
    97→   196→}

   109→   208→function quadruped(skins: MobSkins, opts: {
   110→   209→  bodyW: number; bodyH: number; bodyD: number; bodyY: number;
   111→   210→  legW: number; legH: number;
   112→   211→  headS: number; headY: number; headZ: number;
   113→   212→  shadowR: number;
   114→   213→  fur?: { head: MobSkinPart; body: MobSkinPart; limb: MobSkinPart; inflate: number };
   115→   214→}): MobParts {
   116→   215→  const group = new THREE.Group();
   117→   216→  const mats: THREE.MeshLambertMaterial[] = [];
   118→   217→  const body = boxPart(skins.body, opts.bodyW, opts.bodyH, opts.bodyD, 'body');
   119→   218→  body.position.y = opts.bodyY;
   120→   219→  group.add(body);
   121→   220→  if (opts.fur) {
   122→   221→    // fleece layer as a CHILD of the body — rides along, inflates by ~MC 1.75u
   123→   222→    const f = opts.fur.inflate;
   124→   223→    const furBody = boxPart(opts.fur.body, opts.bodyW + f * 2, opts.bodyH + f * 2, opts.bodyD + f * 2, 'fur');
   125→   224→    body.add(furBody);
   126→   225→    furBody.position.set(0, 0, 0);
   127→   226→  }
   128→   227→  const head = boxPart(skins.head, opts.headS, opts.headS, opts.headS, 'head');
   129→   228→  head.position.set(0, opts.headY, opts.headZ);
   130→   229→  group.add(head);
   131→   230→  if (opts.fur) {
   132→   231→    const f = opts.fur.inflate * 0.6;
   133→   232→    const furHead = boxPart(opts.fur.head, opts.headS + f * 2, opts.headS + f * 2, opts.headS + f * 2, 'fur',
   134→   233→      // fleece FRONT maps to a transparent texel rect → the sheep's real face
   135→   234→      // (eyes/muzzle) shows through the wool, like vanilla
   136→   235→      { frontTransparentRect: [26, 1, 6, 6] });
   137→   236→    head.add(furHead);
   138→   237→    furHead.position.set(0, 0, 0);
   139→   238→  }
   140→   239→  const legs: THREE.Mesh[] = [];
   141→   240→  const lx = opts.bodyW / 2 - opts.legW / 2;
   142→   241→  const lz = opts.bodyD / 2 - opts.legW / 2;
   143→   242→  // MC-style hip: the pivot sits INSIDE the body (leg top hidden), so the leg
   144→   243→  // visibly swings from its top — no gap opens at the hip while walking.
   145→   244→  const hipInset = opts.bodyH * 0.3;
   146→   245→  const hipY = opts.bodyY - opts.bodyH / 2 + hipInset;
   147→   246→  for (const [sx, sz] of [[-1, -1], [1, -1], [-1, 1], [1, 1]]) {
   148→   247→    const pivot = legPivot(skins.limb, opts.legW, opts.legH + hipInset, opts.legW, sx * lx, hipY, sz * lz, sx, opts.fur ? { part: opts.fur.limb, inflate: opts.fur.inflate * 0.5 } : undefined);
   149→   248→    group.add(pivot);
   150→   249→    legs.push(pivot as unknown as THREE.Mesh);
   151→   250→  }
   152→   251→  for (const mesh of [body, head]) collectMats(mats, mesh);
   153→   252→  for (const pv of legs) for (const child of (pv as unknown as THREE.Group).children) collectMats(mats, child as THREE.Mesh);
   154→   253→  return { group, head, legs, arms: [], materials: mats, shadow: null as unknown as THREE.Mesh };
   155→   254→}

   157→   256→function humanoid(skins: MobSkins, thin = false): MobParts {

   174→   273→  const arms: THREE.Mesh[] = [];

   187→   286→  for (const pv of legs) for (const child of (pv as unknown as THREE.Group).children) collectMats(mats, child as THREE.Mesh);
   188→   287→  for (const arm of arms) collectMats(mats, arm);
   189→   288→  return { group, head, legs, arms, materials: mats, shadow: null as unknown as THREE.Mesh };
   190→   289→}
