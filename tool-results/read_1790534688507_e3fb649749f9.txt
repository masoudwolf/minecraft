   140→  // ── game start / save / load ───────────────────────────────────────────────
   141→  async newGame(): Promise<void> {
   142→    const seed = Math.floor(Math.random() * 2147483647);
   143→    this.setupWorld(seed, null);
   144→    await this.preloadSpawn();
   145→    localStorage.removeItem(SAVE_KEY);
   146→    useGameStore.getState().setHasSave(false);
   147→    this.enterPlaying();
   148→  }

   150→  async continueGame(): Promise<void> {
   151→    const raw = localStorage.getItem(SAVE_KEY);
   152→    const save: SaveData | null = raw ? JSON.parse(raw) : null;
   153→    if (!save) { await this.newGame(); return; }
   154→    this.setupWorld(save.seed, save);
   155→    await this.preloadSpawn();
   156→    this.enterPlaying();
   157→  }

   159→  private setupWorld(seed: number, save: SaveData | null): void {
   160→    // clear previous world if any
   161→    if (this.world) {
   162→      for (const key of Array.from(this.world.chunks.keys())) {
   163→        const [cx, cz] = key.split(',').map(Number);
   164→        this.world.unloadChunk(cx, cz);
   165→      }
   166→    }
   167→    this.drops?.clear();
   168→    this.mobs?.clear();
   169→    this.world = new World(seed, save?.edits);
   170→    this.scene.add(this.world.group);
   171→    this.sky = new SkySystem(this.scene, seed);
   172→    this.sky.time = save?.time ?? DAY_LENGTH * 0.3;
   173→    this.drops = new DropManager(this.scene, this.world, getAtlas().texture);
   174→    this.particles = new ParticleSystem(this.scene);
   175→    this.mobs = new MobManager(this.scene, this.world);
   176→    this.player = new Player(this.camera);
   177→
   178→    // spawn position
   179→    let sx = 8, sz = 8;
   180→    this.world.ensureChunk(0, 0);
   181→    let sy = this.world.surfaceY(sx, sz);
   182→    let attempts = 0;
   183→    while ((sy <= 40 || sy === 0) && attempts < 24) {
   184→      sx += 12; sz += 5; attempts++;
   185→      this.world.ensureChunk(Math.floor(sx / CHUNK_SIZE), Math.floor(sz / CHUNK_SIZE));
   186→      sy = this.world.surfaceY(sx, sz);
   187→    }
   188→    if (save) {
   189→      this.player.entity.x = save.player.x;
   190→      this.player.entity.y = save.player.y;
   191→      this.player.entity.z = save.player.z;
   192→      this.player.yaw = save.player.yaw;
   193→      this.player.pitch = save.player.pitch;
   194→      this.player.health = save.player.health;
   195→      this.player.hotbar = save.player.hotbar;
   196→      this.player.main = save.player.main && save.player.main.length === 27
   197→        ? save.player.main
   198→        : Array.from({ length: 27 }, () => ({ blockId: 0, count: 0 }));
   199→      this.player.selected = save.player.selected;
   200→      this.player.fallStartY = save.player.y;
   201→    } else {
   202→      this.player.entity.x = sx + 0.5;
   203→      this.player.entity.y = sy + 1.2;
   204→      this.player.entity.z = sz + 0.5;
   205→      this.player.fallStartY = this.player.entity.y;
   206→    }
   207→    this.updateHandMesh(true);
   208→    this.syncHUD(true);
   209→    this.applySkyFog();
   210→  }

   212→  private async preloadSpawn(): Promise<void> {
   213→    const store = useGameStore.getState();
   214→    store.setScreen('loading');
   215→    const pcx = Math.floor(this.player.entity.x / CHUNK_SIZE);
   216→    const pcz = Math.floor(this.player.entity.z / CHUNK_SIZE);
   217→    const dataR = 2;
   218→    const jobs: [number, number][] = [];
   219→    for (let dx = -dataR; dx <= dataR; dx++)
   220→      for (let dz = -dataR; dz <= dataR; dz++)
   221→        jobs.push([pcx + dx, pcz + dz]);
   222→    jobs.sort((a, b) => (Math.abs(a[0] - pcx) + Math.abs(a[1] - pcz)) - (Math.abs(b[0] - pcx) + Math.abs(b[1] - pcz)));
   223→
   224→    for (let i = 0; i < jobs.length; i++) {
   225→      this.world.ensureChunk(jobs[i][0], jobs[i][1]);
   226→      if (i % 3 === 0) {
   227→        const p = i / jobs.length;
   228→        useGameStore.getState().setHud({ loadingProgress: p, loadingLabel: 'Generating world…' });
   229→        await this.nextFrame();
   230→      }
   231→    }
   232→    // mesh immediate area
   233→    for (let dx = -1; dx <= 1; dx++) {
   234→      for (let dz = -1; dz <= 1; dz++) {
   235→        const c = this.world.getChunk(pcx + dx, pcz + dz);
   236→        if (c) this.world.buildMesh(c);
   237→      }
   238→    }
   239→    useGameStore.getState().setHud({ loadingProgress: 1, loadingLabel: 'Building terrain…' });
   240→    await this.nextFrame();
   241→  }

   262→  saveGame(): void {
   263→    if (!this.world || !this.player) return;
   264→    const edits: Record<string, Record<number, number>> = {};
   265→    for (const [key, map] of this.world.edits) {
   266→      const obj: Record<number, number> = {};
   267→      for (const [idx, id] of map) obj[idx] = id;
   268→      edits[key] = obj;
   269→    }
   270→    const save: SaveData = {
   271→      seed: this.world.terrain.getSeed(),
   272→      time: this.sky.time,
   273→      player: {
   274→        x: this.player.entity.x, y: this.player.entity.y, z: this.player.entity.z,
   275→        yaw: this.player.yaw, pitch: this.player.pitch,
   276→        health: this.player.health,
   277→        hotbar: this.player.hotbar,
   278→        main: this.player.main,
   279→        selected: this.player.selected,
   280→      },
   281→      edits,
   282→    };
   283→    try {
   284→      localStorage.setItem(SAVE_KEY, JSON.stringify(save));
   285→      useGameStore.getState().setHasSave(true);
   286→    } catch { /* quota */ }
   287→  }

   297→  private onKeyDown = (e: KeyboardEvent): void => {
   298→    if (e.code === 'F3') { e.preventDefault(); useGameStore.getState().toggleDebug(); return; }
   299→    const st = useGameStore.getState();
   300→    if (st.inv.open && st.screen === 'playing') {
   301→      if (e.code === 'KeyE' || e.code === 'Escape') { e.preventDefault(); this.closeInventory(); return; }
   302→      if (e.code.startsWith('Digit')) {
   303→        const n = parseInt(e.code.slice(5), 10);
   304→        if (n >= 1 && n <= 9 && this.invHover) this.invHotbarSwap(this.invHover.area, this.invHover.idx, n - 1);
   305→      }
   306→      return;
   307→    }
   308→    if (st.screen !== 'playing') return;
   309→    this.keys.add(e.code);
   310→    if (e.code === 'KeyE') { e.preventDefault(); this.openInventory(false); return; }
   311→    if (e.code.startsWith('Digit')) {
   312→      const n = parseInt(e.code.slice(5), 10);
   313→      if (n >= 1 && n <= 9) { this.player.selected = n - 1; this.syncHUD(); this.updateHandMesh(); }
   314→    }
   315→    if (e.code === 'KeyQ') this.dropSelected();
   316→    if (e.code === 'Space') e.preventDefault();
   317→  };

   323→  private onMouseMove = (e: MouseEvent): void => {
   324→    if (document.pointerLockElement !== this.canvas || useGameStore.getState().screen !== 'playing') return;
   325→    const sens = 0.0022 * this.settings.sensitivity;
   326→    const mx = Math.max(-60, Math.min(60, e.movementX));
   327→    const my = Math.max(-60, Math.min(60, e.movementY));
   328→    this.player.yaw -= mx * sens;
   329→    this.player.pitch -= my * sens;
   330→    this.player.pitch = Math.max(-Math.PI / 2 + 0.01, Math.min(Math.PI / 2 - 0.01, this.player.pitch));
   331→  };

   394→  // ── interaction ────────────────────────────────────────────────────────────
   395→  private updateTarget(): void {
   396→    const eye = new THREE.Vector3(this.player.x, this.player.eyeY(), this.player.z);
   397→    const dir = this.player.forwardVector();
   398→    this.target = raycast(this.world, eye.x, eye.y, eye.z, dir.x, dir.y, dir.z, 4.5);
   399→    if (this.target) {
   400→      this.highlight.visible = true;
   401→      this.highlight.position.set(this.target.x + 0.5, this.target.y + 0.5, this.target.z + 0.5);
   402→    } else {
   403→      this.highlight.visible = false;
   404→    }

   407→  private mineTick(dt: number): void {
   408→    const heldSlot = this.player.hotbar[this.player.selected];
   409→    const heldTool = heldSlot && heldSlot.count > 0 && isItemId(heldSlot.blockId)
   410→      ? getToolDef(heldSlot.blockId)
   411→      : undefined;
   412→
   413→    // mob attack takes priority over mining
   414→    if (this.mining && this.attackCooldown <= 0) {
   415→      const eye = new THREE.Vector3(this.player.x, this.player.eyeY(), this.player.z);
   416→      const dir = this.player.forwardVector();
   417→      const hit = this.mobs.raycastMob(eye.x, eye.y, eye.z, dir.x, dir.y, dir.z, 3.4);
   418→      if (hit) {
   419→        this.attackCooldown = 0.42;
   420→        this.startSwing();
   421→        const kx = hit.mob.x - this.player.x;
   422→        const kz = hit.mob.z - this.player.z;
   423→        this.mobs.hurtMob(hit.mob, heldTool ? heldTool.dmg : 2, kx, kz);
   424→        this.particles.hurt(hit.mob.x, hit.mob.y, hit.mob.z);
   425→        this.damageTool(heldTool && heldTool.type !== 'sword' ? 2 : 1);
   426→        this.crackMesh.visible = false;
   427→        return;
   428→      }
   429→    }
   430→    if (!this.mining || !this.target) { this.crackMesh.visible = false; return; }
   431→    // target changed? reset progress
   432→    const t = this.target;
   433→    if (!this.mineTarget || this.mineTarget.x !== t.x || this.mineTarget.y !== t.y || this.mineTarget.z !== t.z) {
   434→      this.mineTarget = t;
   435→      this.mineProgress = 0;
   436→    }
   437→    const def = getBlockDef(t.id);
   438→    if (!def) { this.crackMesh.visible = false; return; }
   439→    const { time, harvest } = breakInfo(def, heldTool);
   440→    if (!Number.isFinite(time)) { this.crackMesh.visible = false; return; }
   441→    this.mineProgress += dt / time;
   442→
   443→    // dig sound + swing loop
   444→    this.stepTimer += dt;
   445→    if (this.stepTimer > 0.24) {
   446→      this.stepTimer = 0;
   447→      audio.dig((def.sound ?? 'stone') as MaterialSound);
   448→      this.startSwing();
   449→    }
   450→
   451→    if (this.mineProgress >= 1) {
   452→      // break!
   453→      const col = tileAvgColor(Array.isArray(def.tiles) ? def.tiles[2] : def.tiles);
   454→      this.particles.burstBlockBreak(t.x, t.y, t.z, col);
   455→      audio.breakBlock((def.sound ?? 'stone') as MaterialSound);
   456→      const dropId = def.drop === undefined ? t.id : def.drop;
   457→      if (harvest && dropId && dropId > 0) {
   458→        this.drops.spawn(dropId, t.x + 0.5, t.y + 0.3, t.z + 0.5, 1);
   459→      }
   460→      this.damageTool(1);
   461→      this.world.setBlock(t.x, t.y, t.z, BLOCK.AIR);
   462→      this.mineProgress = 0;
   463→      this.mineTarget = null;
   464→      this.crackMesh.visible = false;
   465→      return;
   466→    }
   467→    // crack overlay
   468→    const stage = Math.min(9, Math.floor(this.mineProgress * 10));
   469→    this.crackMesh.material = this.crackMats[stage];
   470→    this.crackMesh.visible = true;
   471→    this.crackMesh.position.set(t.x + 0.5, t.y + 0.5, t.z + 0.5);
   472→  }

   474→  private placeBlock(): void {
   475→    if (!this.target || this.placeCooldown > 0) return;
   476→    // right-click on crafting table opens the 3x3 grid (unless sneaking)
   477→    const targetId = this.world.getBlock(this.target.x, this.target.y, this.target.z);
   478→    if (targetId === BLOCK.CRAFTING_TABLE && !this.player.sneaking) {
   479→      this.openInventory(true);
   480→      return;
   481→    }
   482→    const slot = this.player.hotbar[this.player.selected];
   483→    if (!slot || slot.blockId === 0 || slot.count <= 0) return;
   484→    // eating food items
   485→    if (isItemId(slot.blockId)) {
   486→      const itemDef = getItemDef(slot.blockId);
   487→      if (itemDef?.food && this.player.hunger < 19.6 && this.eatCooldown <= 0) {
   488→        this.eatCooldown = 1.4;
   489→        this.player.hunger = Math.min(20, this.player.hunger + itemDef.food);
   490→        audio.eat();
   491→        window.setTimeout(() => audio.burp(), 700);
   492→        this.startSwing();
   493→        slot.count--;
   494→        if (slot.count <= 0) { slot.blockId = 0; slot.count = 0; }
   495→        this.syncHUD();
   496→        this.updateHandMesh();
   497→      }
   498→      return;
   499→    }
   500→    const bx = this.target.x + this.target.nx;
   501→    const by = this.target.y + this.target.ny;
   502→    const bz = this.target.z + this.target.nz;
   503→    if (by < 0 || by >= WORLD_HEIGHT) return;
   504→    const existing = this.world.getBlock(bx, by, bz);
   505→    if (existing !== BLOCK.AIR && !isLiquid(existing)) return;
   506→    // don't place inside player
   507→    if (aabbIntersectsBlock(this.player.entity, bx, by, bz)) return;
   508→    const def = getBlockDef(slot.blockId);
   509→    this.world.setBlock(bx, by, bz, slot.blockId);
   510→    audio.place((def?.sound ?? 'stone') as MaterialSound);
   511→    this.placeCooldown = 0.22;
   512→    this.startSwing();
   513→    slot.count--;
   514→    if (slot.count <= 0) { slot.blockId = 0; slot.count = 0; }
   515→    this.syncHUD();
   516→    this.updateHandMesh();
   517→  }

   649→  private clickSlot(list: HotbarSlot[], idx: number, button: 'left' | 'right'): void {
   650→    const slot = list[idx];
   651→    const cur = this.cursor;
   652→    if (button === 'right') {
   653→      if (!cur) {
   654→        if (isEmptySlot(slot)) return;
   655→        const half = Math.ceil(slot.count / 2);
   656→        this.cursor = { ...slot, count: half };
   657→        slot.count -= half;
   658→        if (slot.count <= 0) list[idx] = emptySlot();
   659→      } else {
   660→        if (isEmptySlot(slot)) {
   661→          list[idx] = { blockId: cur.blockId, count: 1, dur: cur.dur };
   662→          cur.count--;
   663→        } else if (slot.blockId === cur.blockId && slot.count < maxStack(slot.blockId)) {
   664→          slot.count++;
   665→          cur.count--;
   666→        } else return;
   667→        if (cur.count <= 0) this.cursor = null;
   668→      }
   669→      return;
   670→    }
   671→    // left click
   672→    if (!cur) {
   673→      if (isEmptySlot(slot)) return;
   674→      this.cursor = { ...slot };
   675→      list[idx] = emptySlot();
   676→    } else if (isEmptySlot(slot)) {
   677→      list[idx] = { ...cur };
   678→      this.cursor = null;
   679→    } else if (slot.blockId === cur.blockId && !isToolItem(slot.blockId)) {
   680→      const max = maxStack(slot.blockId);
   681→      const take = Math.min(cur.count, max - slot.count);
   682→      slot.count += take;
   683→      cur.count -= take;
   684→      if (cur.count <= 0) this.cursor = null;
   685→    } else {
   686→      list[idx] = { ...cur };
   687→      this.cursor = { ...slot };
   688→    }
   689→  }

   691→  private shiftMove(area: 'hotbar' | 'main' | 'craft', list: HotbarSlot[], idx: number): void {
   692→    const slot = list[idx];
   693→    if (isEmptySlot(slot)) return;
   694→    let left: number;
   695→    if (area === 'hotbar') {
   696→      left = addToSlots(this.player.main, slot.blockId, slot.count, slot.dur);
   697→    } else {
   698→      // main/craft -> hotbar first, then main (craft items go home)
   699→      left = addToSlots(this.player.hotbar, slot.blockId, slot.count, slot.dur);
   700→      if (left > 0 && area === 'main') left = addToSlots(this.player.main, slot.blockId, left, slot.dur);
   701→    }
   702→    if (left <= 0) list[idx] = emptySlot();
   703→    else slot.count = left;
   704→  }

   706→  private updateCraftOut(): void {
   707→    const grid = this.craftGrid;
   708→    const size = this.invTable ? 3 : 2;
   709→    const ids = grid.map((s) => (isEmptySlot(s) ? 0 : s.blockId));
   710→    const res = matchRecipe(ids, size);
   711→    this.craftOut = res
   712→      ? { blockId: res.id, count: res.count, dur: freshDur(res.id) }
   713→      : null;
   714→  }

   716→  private takeCraftOutput(shift: boolean): void {
   717→    if (!this.craftOut) return;
   718→    const grid = this.craftGrid;
   719→    const consume = (): void => {
   720→      for (let i = 0; i < grid.length; i++) {
   721→        const s = grid[i];
   722→        if (isEmptySlot(s)) continue;
   723→        s.count--;
   724→        if (s.count <= 0) grid[i] = emptySlot();
   725→      }
   726→    };
   727→    if (!shift) {
   728→      const out = this.craftOut;
   729→      if (!this.cursor) {
   730→        this.cursor = { ...out };
   731→        consume();
   732→      } else if (this.cursor.blockId === out.blockId && !isToolItem(out.blockId) && this.cursor.count + out.count <= maxStack(out.blockId)) {
   733→        this.cursor.count += out.count;
   734→        consume();
   735→      } else return;
   736→    } else {
   737→      let guard = 0;
   738→      while (guard++ < 64) {
   739→        const size = this.invTable ? 3 : 2;
   740→        const res = matchRecipe(grid.map((s) => (isEmptySlot(s) ? 0 : s.blockId)), size);
   741→        if (!res) break;
   742→        const left = this.addToInventory(res.id, res.count, freshDur(res.id));
   743→        if (left > 0) {
   744→          const p = this.player.entity;
   745→          this.drops.spawn(res.id, p.x, p.y + 1, p.z, left);
   746→        }
   747→        consume();
   748→        audio.pop();
   749→      }
   750→    }
   751→    this.updateCraftOut();
   752→    this.syncInventory(true);
   753→    this.syncHUD(true);
   754→  }

   842→  // ── chunk streaming ────────────────────────────────────────────────────────
   843→  private streamChunks(): void {
   844→    const R = this.settings.renderDistance;
   845→    const pcx = Math.floor(this.player.entity.x / CHUNK_SIZE);
   846→    const pcz = Math.floor(this.player.entity.z / CHUNK_SIZE);
   847→
   848→    const t0 = performance.now();
   849→    // generate data (spiral by distance), budget ~6ms
   850→    outer:
   851→    for (let r = 0; r <= R + 1; r++) {
   852→      for (let dx = -r; dx <= r; dx++) {
   853→        for (let dz = -r; dz <= r; dz++) {
   854→          if (Math.max(Math.abs(dx), Math.abs(dz)) !== r) continue;
   855→          const key = chunkKey(pcx + dx, pcz + dz);
   856→          if (!this.world.chunks.has(key)) {
   857→            this.world.ensureChunk(pcx + dx, pcz + dz);
   858→            if (performance.now() - t0 > 6) break outer;
   859→          }
   860→        }
   861→      }
   862→    }
   863→
   864→    // mesh chunks with all 4 neighbors' data, budget ~7ms
   865→    const t1 = performance.now();
   866→    outer2:
   867→    for (let r = 0; r <= R; r++) {
   868→      for (let dx = -r; dx <= r; dx++) {
   869→        for (let dz = -r; dz <= r; dz++) {
   870→          if (Math.max(Math.abs(dx), Math.abs(dz)) !== r) continue;
   871→          const chunk = this.world.getChunk(pcx + dx, pcz + dz);
   872→          if (chunk && chunk.hasData && chunk.needsMesh) {
   873→            const n = this.world.getChunk(pcx + dx + 1, pcz + dz);
   874→            const s = this.world.getChunk(pcx + dx - 1, pcz + dz);
   875→            const e = this.world.getChunk(pcx + dx, pcz + dz + 1);
   876→            const w = this.world.getChunk(pcx + dx, pcz + dz - 1);
   877→            if (n?.hasData && s?.hasData && e?.hasData && w?.hasData) {
   878→              this.world.buildMesh(chunk);
   879→              if (performance.now() - t1 > 7) break outer2;
   880→            }
   881→          }
   882→        }
   883→      }
   884→    }
   885→
   886→    // unload far chunks occasionally
   887→    if ((this.saveTimer % 5) < 0.016) {
   888→      for (const key of Array.from(this.world.chunks.keys())) {
   889→        const [cx, cz] = key.split(',').map(Number);
   890→        if (Math.max(Math.abs(cx - pcx), Math.abs(cz - pcz)) > R + 3) {
   891→          this.world.unloadChunk(cx, cz);
   892→        }
   893→      }
   894→    }
   895→  }

   920→  // ── main loop ──────────────────────────────────────────────────────────────
   921→  private loop = (): void => {
   922→    if (this.disposed) return;
   923→    this.raf = requestAnimationFrame(this.loop);
   924→    const now = performance.now();
   925→    let dt = (now - this.lastTime) / 1000;
   926→    this.lastTime = now;
   927→    if (dt > 0.25) dt = 0.25;
   928→
   929→    // fps
   930→    this.fpsAccum += dt;
   931→    this.fpsFrames++;
   932→    if (this.fpsAccum >= 0.5) {
   933→      this.fps = Math.round(this.fpsFrames / this.fpsAccum);
   934→      this.fpsAccum = 0;
   935→      this.fpsFrames = 0;
   936→    }
   937→
   938→    const screen = useGameStore.getState().screen;
   939→    const simulate = this.running && (screen === 'playing');
   940→
   941→    if (simulate && this.player && this.world) {
   942→      this.accumulator += dt;
   943→      const step = 1 / 60;
   944→      let steps = 0;
   945→      while (this.accumulator >= step && steps < 5) {
   946→        this.physicsStep(step);
   947→        this.accumulator -= step;
   948→        steps++;
   949→      }
   950→      this.frameUpdate(dt);
   951→    } else {
   952→      // still render (paused view behind menu)
   953→      if (this.player && this.world && this.sky) {
   954→        this.sky.update(0, this.camera, this.scene, 60, 130);
   955→        this.applySkyFog();
   956→      }
   957→    }
   958→
   959→    this.renderer.render(this.scene, this.camera);
   960→
   961→    // debug overlay ~4Hz
   962→    this.debugTimer += dt;
   963→    if (this.debugTimer > 0.25 && this.player && this.world) {
   964→      this.debugTimer = 0;
   965→      this.updateDebug();
   966→    }
   967→  };

   969→  private physicsStep(step: number): void {
   970→    const p = this.player;
   971→    // input direction
   972→    let forward = 0, strafe = 0;
   973→    if (this.keys.has('KeyW')) forward += 1;
   974→    if (this.keys.has('KeyS')) forward -= 1;
   975→    if (this.keys.has('KeyA')) strafe -= 1;
   976→    if (this.keys.has('KeyD')) strafe += 1;
   977→    const wishJump = this.keys.has('Space');
   978→    const wishSneak = this.keys.has('ShiftLeft') || this.keys.has('ShiftRight');
   979→    p.sprinting = (this.keys.has('ControlLeft') || this.keys.has('ControlRight')) && forward > 0;
   980→
   981→    const wasInWater = p.entity.inWater;
   982→    const wasOnGround = p.entity.onGround;
   983→    const prevVy = p.entity.vy;
   984→
   985→    p.moveInput({ forward, strafe }, this.world, step, wishJump, wishSneak);
   986→
   987→    // footsteps
   988→    if (p.entity.onGround && p.stepDistance > 2.1) {
   989→      p.stepDistance = 0;
   990→      const below = this.world.getBlock(Math.floor(p.x), Math.floor(p.y - 0.4), Math.floor(p.z));
   991→      const def = getBlockDef(below);
   992→      if (def) audio.step((def.sound ?? 'stone') as MaterialSound);
   993→    }
   994→    // splash
   995→    if (!wasInWater && p.entity.inWater && prevVy < -4) {
   996→      audio.splash();
   997→      this.particles.splash(p.x, p.y + 0.2, p.z);
   998→    }
   999→    // landing particles
  1000→    if (!wasOnGround && p.entity.onGround && this.prevVy < -7) {
  1001→      const below = this.world.getBlock(Math.floor(p.x), Math.floor(p.y - 0.4), Math.floor(p.z));
  1002→      const def = getBlockDef(below);
  1003→      if (def) {
  1004→        const col = tileAvgColor(Array.isArray(def.tiles) ? def.tiles[2] : def.tiles);
  1005→        this.particles.burstLand(p.x, p.y, p.z, col, Math.min(14, Math.floor(-this.prevVy / 2)));
  1006→      }
  1007→    }
  1008→    this.wasInWater = p.entity.inWater;
  1009→    this.wasOnGround = p.entity.onGround;
  1010→    this.prevVy = p.entity.vy;
  1011→
  1012→    // hurt cooldown decay
  1013→    if (p.hurtCooldown > 0) p.hurtCooldown -= step;
  1014→
  1015→    // ── hunger drain ──
  1016→    if (p.sprinting && (forward !== 0 || strafe !== 0)) {
  1017→      p.hunger = Math.max(0, p.hunger - 0.085 * step);
  1018→    } else if (forward !== 0 || strafe !== 0) {
  1019→      p.hunger = Math.max(0, p.hunger - 0.012 * step);
  1020→    } else {
  1021→      p.hunger = Math.max(0, p.hunger - 0.0015 * step);
  1022→    }
  1023→    if (wishJump && p.onGround) p.hunger = Math.max(0, p.hunger - 0.05);
  1024→
  1025→    // void damage
  1026→    if (p.y < -8) {
  1027→      p.damage(4);
  1028→      p.entity.vy = 0;
  1029→      p.entity.y = -8;
  1030→    }
  1031→
  1032→    // death
  1033→    if (p.dead) {
  1034→      useGameStore.getState().setScreen('dead');
  1035→      if (document.pointerLockElement === this.canvas) document.exitPointerLock();
  1036→    }
  1037→  }

  1039→  private frameUpdate(dt: number): void {
  1040→    const p = this.player;
  1041→
  1042→    // camera
  1043→    p.applyCamera(this.settings.fov, 8, dt);
  1044→
  1045→    // cooldowns
  1046→    this.placeCooldown -= dt;
  1047→    this.attackCooldown -= dt;
  1048→    this.eatCooldown -= dt;
  1049→
  1050→    // hunger regen / starve
  1051→    this.hungerRegenTimer += dt;
  1052→    if (this.hungerRegenTimer > 2) {
  1053→      this.hungerRegenTimer = 0;
  1054→      if (p.hunger >= 18 && p.health < p.maxHealth) {
  1055→        p.heal(1);
  1056→        p.hunger = Math.max(0, p.hunger - 0.4);
  1057→      }
  1058→    }
  1059→    this.starveTimer += dt;
  1060→    if (this.starveTimer > 3) {
  1061→      this.starveTimer = 0;
  1062→      if (p.hunger <= 0 && p.health > 2) {
  1063→        p.damage(1);
  1064→        audio.hurt();
  1065→      }
  1066→    }
  1067→
  1068→    // interaction
  1069→    this.updateTarget();
  1070→    this.mineTick(dt);
  1071→
  1072→    // drops + particles
  1073→    this.drops.update(dt, p.entity, (stack) => {
  1074→      const picked = this.tryPickup(stack);
  1075→      if (picked) {
  1076→        const name = isItemId(stack.blockId)
  1077→          ? (getItemDef(stack.blockId)?.name ?? 'Item')
  1078→          : (getBlockDef(stack.blockId)?.name ?? 'Block');
  1079→        this.showToast(name + ' ×' + stack.count);
  1080→      }
  1081→      return picked;
  1082→    });
  1083→    this.particles.update(dt);
  1084→
  1085→    // mobs
  1086→    if (this.mobs) {
  1087→      const eyeY = p.eyeY();
  1088→      this.mobs.update(dt, {
  1089→        x: p.x, y: p.y, z: p.z,
  1090→        vx: 0, vy: 0, vz: 0,
  1091→        width: 0.6, height: 1.8,
  1092→        onGround: p.onGround, inWater: p.inWater,
  1093→        eyeY: () => eyeY,
  1094→      }, this.sky.sunLevel, {
  1095→        damagePlayer: (amount, fx, fz) => {
  1096→          if (amount <= 0) return;
  1097→          p.damage(amount);
  1098→          audio.hurt();
  1099→          // knockback away from source
  1100→          const kx = p.x - fx;
  1101→          const kz = p.z - fz;
  1102→          const len = Math.hypot(kx, kz) || 1;
  1103→          p.entity.vx += (kx / len) * 6.5;
  1104→          p.entity.vz += (kz / len) * 6.5;
  1105→          p.entity.vy = Math.max(p.entity.vy, 4.2);
  1106→        },
  1107→        spawnDrop: (itemId, dx, dy, dz) => {
  1108→          this.drops.spawn(itemId, dx, dy, dz, 1);
  1109→        },
  1110→        explodeParticles: (ex, ey, ez) => {
  1111→          for (let i = 0; i < 3; i++) {
  1112→            this.particles.burstBlockBreak(ex - 0.5 + i * 0.5, ey, ez, [0.35, 0.33, 0.3]);
  1113→          }
  1114→          this.particles.burstLand(ex, ey, ez, [0.2, 0.2, 0.2], 20);
  1115→        },
  1116→        deathParticles: (dx2, dy2, dz2) => {
  1117→          this.particles.burstLand(dx2, dy2, dz2, [0.85, 0.25, 0.25], 10);
  1118→        },
  1119→        fireParticle: (fx2, fy2, fz2) => {
  1120→          this.particles.spawnParticle(fx2, fy2, fz2, (Math.random() - 0.5) * 0.6, 1.6 + Math.random(), (Math.random() - 0.5) * 0.6, [1, 0.45, 0.1], 0.09, 0.5, -2);
  1121→        },
  1122→        playerX: p.x,
  1123→        playerY: p.y,
  1124→        playerZ: p.z,
  1125→      });
  1126→    }
  1127→
  1128→    // sky + fog + lighting uniforms
  1129→    this.sky.update(dt, this.camera, this.scene, 60, 130);
  1130→    this.applySkyFog();
  1131→
  1132→    // hand animation
  1133→    this.animateHand(dt);
  1134→
  1135→    // streaming + autosave
  1136→    this.streamChunks();
  1137→    this.saveTimer += dt;
  1138→    if (this.saveTimer > 20) {
  1139→      this.saveTimer = 0;
  1140→      this.saveGame();
  1141→    }
  1142→
  1143→    // underwater check for HUD
  1144→    const camBlock = this.world.getBlock(Math.floor(this.camera.position.x), Math.floor(this.camera.position.y), Math.floor(this.camera.position.z));
  1145→    const underwater = camBlock === BLOCK.WATER;
  1146→    const st = useGameStore.getState();
  1147→    if (st.hud.underwater !== underwater) st.setHud({ underwater });
  1148→
  1149→    this.syncHUD();
  1150→  }

  1152→  private applySkyFog(): void {
  1153→    if (!this.sky) return;
  1154→    const mats = [this.world?.getMaterial('opaque'), this.world?.getMaterial('cutout'), this.world?.getMaterial('water')];
  1155→    const fog = this.scene.fog as THREE.Fog | null;
  1156→    const underwater = this.world && this.camera.position && isLiquid(this.world.getBlock(
  1157→      Math.floor(this.camera.position.x), Math.floor(this.camera.position.y), Math.floor(this.camera.position.z)
  1158→    ));
  1159→    for (const m of mats) {
  1160→      if (!m) continue;
  1161→      m.uniforms.uSunLevel.value = this.sky.sunLevel;
  1162→      m.uniforms.uTime.value = performance.now() / 1000;
  1163→      if (underwater) {
  1164→        m.uniforms.uFogColor.value.setRGB(0.09, 0.24, 0.55);
  1165→        m.uniforms.uFogNear.value = 4;
  1166→        m.uniforms.uFogFar.value = 26;
  1167→      } else if (fog) {
  1168→        (m.uniforms.uFogColor.value as THREE.Color).copy(fog.color);
  1169→        m.uniforms.uFogNear.value = fog.near;
  1170→        m.uniforms.uFogFar.value = fog.far;
  1171→      }
  1172→    }
  1173→    // entity lights follow sun
  1174→    this.ambient.intensity = 0.25 + this.sky.sunLevel * 0.75;
  1175→    this.sunLight.intensity = 0.15 + this.sky.sunLevel * 0.8;
  1176→    const angle = ((this.sky.time / DAY_LENGTH) - 0.25) * Math.PI * 2;
  1177→    this.sunLight.position.set(Math.cos(angle), Math.max(0.2, Math.sin(angle)), 0.3).normalize().multiplyScalar(50);
  1178→  }

  1180→  private updateDebug(): void {

  1185→    // yaw=0 looks toward -Z (north); positive yaw (mouse left) rotates toward -X (west)
  1186→    const dirs = ['north', 'north-west', 'west', 'south-west', 'south', 'south-east', 'east', 'north-east'];
  1187→    const facing = dirs[Math.round(yawDeg / 45) % 8];

  1193→      biome: this.world.terrain.biomeAt(Math.floor(p.x), Math.floor(p.z)),

  1196→      chunks: this.world.chunks.size,
