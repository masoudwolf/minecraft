   108→// ─── Model builders (vanilla textures + MC box-UV mapping) ───────────────────
   109→/** Build a box mesh textured from a vanilla skin part via standard MC box UVs. */
   110→function boxPart(p: MobSkinPart, w: number, h: number, d: number, tag: string): THREE.Mesh {
   111→  const geo = new THREE.BoxGeometry(w, h, d);
   112→  boxUV(geo, p.lay, p.texW, p.texH);
   113→  const mesh = new THREE.Mesh(geo, new THREE.MeshLambertMaterial({ map: p.tex }));
   114→  mesh.userData.part = tag;
   115→  return mesh;
   116→}
   117→
   118→/** Collect a mesh's material(s) into the parts list (per-instance tint/light). */
   119→function collectMats(list: THREE.MeshLambertMaterial[], mesh: THREE.Mesh): void {
   120→  const m = mesh.material as THREE.MeshLambertMaterial | THREE.MeshLambertMaterial[];
   121→  if (Array.isArray(m)) list.push(...m);
   122→  else list.push(m);
   123→}

   125→/**
   126→ * Leg with a proper HIP PIVOT (Minecraft-style limb rigging): the pivot Group
   127→ * sits at the hip joint (top of the leg) and the leg box hangs BELOW it, so
   128→ * animating pivot.rotation.x swings the leg from the top — like MC. `furPart`
   129→ * optionally adds an inflated fleece box (sheep) that rides the swing; like
   130→ * vanilla, the fleece leg wraps only the UPPER HALF of the leg.
   131→ */
   132→function legPivot(p: MobSkinPart, w: number, h: number, d: number, hipX: number, hipY: number, hipZ: number, side = 0, furPart?: { part: MobSkinPart; inflate: number }): THREE.Group {
   133→  const pivot = new THREE.Group();
   134→  pivot.position.set(hipX, hipY, hipZ);
   135→  pivot.userData.side = side;
   136→  const leg = boxPart(p, w, h, d, 'limb');
   137→  leg.position.y = -h / 2; // hang from the hip
   138→  pivot.add(leg);
   139→  if (furPart) {
   140→    const f = furPart.inflate;
   141→    const furH = h * 0.5 + f * 2;
   142→    const fur = boxPart(furPart.part, w + f * 2, furH, d + f * 2, 'fur');
   143→    leg.add(fur);
   144→    fur.position.y = h / 2 - furH / 2 + f * 0.5; // top-aligned wrap
   145→  }
   146→  return pivot;
   147→}

   149→function makeShadow(radius: number, scene: THREE.Scene): THREE.Mesh {
   150→  const geo = new THREE.CircleGeometry(radius, 12);
   151→  const mat = new THREE.MeshBasicMaterial({ color: 0x000000, transparent: true, opacity: 0.28, depthWrite: false });
   152→  const mesh = new THREE.Mesh(geo, mat);
   153→  mesh.rotation.x = -Math.PI / 2;
   154→  mesh.renderOrder = 1;
   155→  scene.add(mesh);
   156→  return mesh;
   157→}

   159→function quadruped(skins: MobSkins, opts: {
   160→  bodyW: number; bodyH: number; bodyD: number; bodyY: number;
   161→  legW: number; legH: number;
   162→  headS: number; headY: number; headZ: number;
   163→  shadowR: number;
   164→  fur?: { head: MobSkinPart; body: MobSkinPart; limb: MobSkinPart; inflate: number };
   165→}): MobParts {
   166→  const group = new THREE.Group();
   167→  const mats: THREE.MeshLambertMaterial[] = [];
   168→  const body = boxPart(skins.body, opts.bodyW, opts.bodyH, opts.bodyD, 'body');
   169→  body.position.y = opts.bodyY;
   170→  group.add(body);
   171→  if (opts.fur) {
   172→    // fleece layer as a CHILD of the body — rides along, inflates by ~MC 1.75u
   173→    const f = opts.fur.inflate;
   174→    const furBody = boxPart(opts.fur.body, opts.bodyW + f * 2, opts.bodyH + f * 2, opts.bodyD + f * 2, 'fur');
   175→    body.add(furBody);
   176→    furBody.position.set(0, 0, 0);
   177→  }
   178→  const head = boxPart(skins.head, opts.headS, opts.headS, opts.headS, 'head');
   179→  head.position.set(0, opts.headY, opts.headZ);
   180→  group.add(head);
   181→  if (opts.fur) {
   182→    const f = opts.fur.inflate * 0.6;
   183→    const furHead = boxPart(opts.fur.head, opts.headS + f * 2, opts.headS + f * 2, opts.headS + f * 2, 'fur');
   184→    head.add(furHead);
   185→    furHead.position.set(0, 0, 0);
   186→  }
   187→  const legs: THREE.Mesh[] = [];
   188→  const lx = opts.bodyW / 2 - opts.legW / 2;
   189→  const lz = opts.bodyD / 2 - opts.legW / 2;
   190→  const hipY = opts.bodyY - opts.bodyH / 2; // hips at the body's underside
   191→  for (const [sx, sz] of [[-1, -1], [1, -1], [-1, 1], [1, 1]]) {
   192→    const pivot = legPivot(skins.limb, opts.legW, opts.legH, opts.legW, sx * lx, hipY, sz * lz, sx, opts.fur ? { part: opts.fur.limb, inflate: opts.fur.inflate * 0.5 } : undefined);
   193→    group.add(pivot);
   194→    legs.push(pivot as unknown as THREE.Mesh);
   195→  }
   196→  for (const mesh of [body, head]) collectMats(mats, mesh);
   197→  for (const pv of legs) for (const child of (pv as unknown as THREE.Group).children) collectMats(mats, child as THREE.Mesh);
   198→  return { group, head, legs, arms: [], materials: mats, shadow: null as unknown as THREE.Mesh };
   199→}

   201→function humanoid(skins: MobSkins, thin = false): MobParts {
   202→  const lw = thin ? 0.16 : 0.25;
   203→  const aw = thin ? 0.14 : 0.22;
   204→  const group = new THREE.Group();
   205→  const mats: THREE.MeshLambertMaterial[] = [];
   206→  const body = boxPart(skins.body, 0.5, 0.72, 0.26, 'body');
   207→  body.position.y = 1.1;
   208→  group.add(body);
   209→  const head = boxPart(skins.head, 0.5, 0.5, 0.5, 'head');
   210→  head.position.y = 1.72;
   211→  group.add(head);
   212→  const legs: THREE.Mesh[] = [];
   213→  for (const sx of [-1, 1]) {
   214→    const pivot = legPivot(skins.limb, lw, 0.74, lw, sx * 0.125, 0.74, 0, sx);
   215→    group.add(pivot);
   216→    legs.push(pivot as unknown as THREE.Mesh);
   217→  }
   218→  const arms: THREE.Mesh[] = [];
   219→  const armPart = skins.limb2 ?? skins.limb;
   220→  for (const sx of [-1, 1]) {
   221→    const pivot = new THREE.Group();
   222→    pivot.position.set(sx * (0.25 + aw / 2), 1.42, 0);
   223→    const arm = boxPart(armPart, aw, 0.7, aw, 'limb');
   224→    arm.position.y = -0.32;
   225→    pivot.add(arm);
   226→    group.add(pivot);
   227→    arms.push(arm);
   228→    (arm as unknown as { pivot: THREE.Group }).pivot = pivot;
   229→  }
   230→  for (const mesh of [body, head]) collectMats(mats, mesh);
   231→  for (const pv of legs) for (const child of (pv as unknown as THREE.Group).children) collectMats(mats, child as THREE.Mesh);
   232→  for (const arm of arms) collectMats(mats, arm);
   233→  return { group, head, legs, arms, materials: mats, shadow: null as unknown as THREE.Mesh };
   234→}

   241→  if (!def) return null;

   243→  return def.builder(skins);
   244→}

   549→// ─── Ray vs AABB (slab method) ───────────────────────────────────────────────
   550→export interface MobHit { mob: Mob; dist: number }
   551→
   552→function rayAABB(ox: number, oy: number, oz: number, dx: number, dy: number, dz: number, e: AABBEntity, maxDist: number): number | null {
   553→  const half = e.width / 2;
   554→  const minX = e.x - half, maxX = e.x + half;
   555→  const minY = e.y, maxY = e.y + e.height;
   556→  const minZ = e.z - half, maxZ = e.z + half;
   557→  let tmin = 0;
   558→  let tmax = maxDist;
   559→  for (const [o, d, min, max] of [[ox, dx, minX, maxX], [oy, dy, minY, maxY], [oz, dz, minZ, maxZ]] as const) {
   560→    if (Math.abs(d) < 1e-9) {
   561→      if (o < min || o > max) return null;
   562→    } else {
   563→      let t1 = (min - o) / d;
   564→      let t2 = (max - o) / d;
   565→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp; }
   566→      tmin = Math.max(tmin, t1);
   567→      tmax = Math.min(tmax, t2);
   568→      if (tmin > tmax) return null;
   569→    }
   570→  }
   571→  return tmin;
   572→}

   734→  private trySpawnMob(playerX: number, playerY: number, playerZ: number, sunLevel: number, passiveCount: number, hostileCount: number): void {
   735→    const angle = Math.random() * Math.PI * 2;
   736→    const dist = 16 + Math.random() * 26;
   737→    const x = Math.floor(playerX + Math.cos(angle) * dist) + 0.5;
   738→    const z = Math.floor(playerZ + Math.sin(angle) * dist) + 0.5;
   739→    const cx = Math.floor(x);
   740→    const cz = Math.floor(z);
   741→    // find surface
   742→    let sy = -1;
   743→    for (let y = 95; y > 2; y--) {
   744→      const id = this.world.getBlock(cx, y, cz);
   745→      if (id !== 0 && !isWaterId(id)) {
   746→        sy = y;
   747→        break;
   748→      }
   749→    }
   750→    if (sy < 3) return;
   751→    const above = this.world.getBlock(cx, sy + 1, cz);
   752→    const above2 = this.world.getBlock(cx, sy + 2, cz);
   753→    if (above !== 0 || above2 !== 0) return;
   754→    const groundId = this.world.getBlock(cx, sy, cz);
   755→    if (isWaterId(groundId)) return;
   756→
   757→    const l = this.world.getLight(cx, sy + 1, cz);
   758→    if (l < 0) return;
   759→    const skyL = l >> 4;
   760→    const blkL = l & 15;
   761→    const effLight = Math.max(blkL, skyL * sunLevel);
   762→
   763→    // villagers live on village grounds (planks / cobblestone); mooshrooms on mycelium
   764→    if (groundId === BLOCK.PLANKS || groundId === BLOCK.COBBLESTONE) {
   765→      // swamp planks/cobble = witch hut grounds: keep a witch around (day or night)
   766→      if (this.world.biomeAt?.(cx, cz) === 'swamp') {
   767→        const witchCount = this.mobs.filter((m2) => m2.type === 'witch').length;
   768→        if (witchCount < 1 && Math.random() < 0.3) {
   769→          this.spawn('witch', x, sy + 1, z);
   770→        }
   771→        return;
   772→      }
   773→      // iron golem: at most one patrolling each village chunk roll
   774→      const golemCount = this.mobs.filter((m2) => m2.type === 'golem').length;
   775→      if (golemCount < 1 && Math.random() < 0.22 && skyL >= 9 && sunLevel > 0.55) {
   776→        this.spawn('golem', x, sy + 1, z);
   777→        return;
   778→      }
   779→      const villagerCount = this.mobs.filter((m) => m.type === 'villager').length;
   780→      if (villagerCount < 5 && skyL >= 9 && sunLevel > 0.55) {
   781→        const herd = 1 + Math.floor(Math.random() * 2);
   782→        for (let i = 0; i < herd; i++) this.spawn('villager', x + (Math.random() - 0.5) * 2, sy + 1, z + (Math.random() - 0.5) * 2);
   783→      }
   784→      return;
   785→    }
   786→    if (groundId === BLOCK.MYCELIUM) {
   787→      const passiveNow = passiveCount + this.mobs.filter((m) => m.type === 'mooshroom').length;
   788→      if (passiveNow < 12 && skyL >= 9 && sunLevel > 0.55) {
   789→        const herd = 1 + Math.floor(Math.random() * 2);
   790→        for (let i = 0; i < herd; i++) this.spawn('mooshroom', x + (Math.random() - 0.5) * 3, sy + 1, z + (Math.random() - 0.5) * 3);
   791→      }
   792→      return;
   793→    }
   794→
   795→    const wantHostile = hostileCount < 12 && (effLight < 6);
   796→    const wantPassive = passiveCount < 10 && skyL >= 9 && sunLevel > 0.55 && (groundId === BLOCK.GRASS || groundId === BLOCK.SNOW_GRASS);
   797→
   798→    if (wantHostile && (!wantPassive || Math.random() < 0.65)) {
   799→      const roll = Math.random();
   800→      const isSwamp = this.world.biomeAt?.(cx, cz) === 'swamp';
   801→      const type: MobType = roll < 0.32 ? 'zombie' : roll < 0.58 ? 'skeleton' : roll < 0.74 ? 'creeper' : roll < 0.86 ? 'spider' : roll < 0.95 && isSwamp ? 'witch' : 'enderman';
   802→      this.spawn(type, x, sy + 1, z);
   803→    } else if (wantPassive) {
   804→      const roll = Math.random();
   805→      const type: MobType = roll < 0.3 ? 'pig' : roll < 0.55 ? 'sheep' : roll < 0.8 ? 'cow' : 'chicken';
   806→      // spawn small herd for passive
   807→      const herd = 1 + Math.floor(Math.random() * 3);
   808→      for (let i = 0; i < herd; i++) {
   809→        this.spawn(type, x + (Math.random() - 0.5) * 3, sy + 1, z + (Math.random() - 0.5) * 3, type === 'sheep' ? pickSheepVariant() : '');
   810→      }
   811→    }
   812→  }

   814→  hurtMob(mob: Mob, dmg: number, kx: number, kz: number, cb?: MobCallbacks): boolean {
   815→    if (mob.dead) return false;
   816→    mob.health -= dmg;
   817→    mob.hurtT = 0.4;
   818→    const len = Math.hypot(kx, kz) || 1;
   819→    mob.vx += (kx / len) * 7;
   820→    mob.vz += (kz / len) * 7;
   821→    mob.vy = Math.max(mob.vy, 4.4);
   822→    audio.mobHurt(mob.def.sound);
   823→    if (mob.def.hostile) {
   824→      mob.state = 'chase';
   825→      if (mob.type === 'spider' || mob.type === 'enderman') mob.provoked = true;
   826→    } else if (mob.type === 'golem') {
   827→      // golems never flee — they stoically keep patrolling
   828→      mob.state = 'idle';
   829→    } else {
   830→      mob.state = 'flee';
   831→      mob.stateTimer = 4;
   832→      mob.wanderX = mob.x + (mob.x - kx) * 10;
   833→      mob.wanderZ = mob.z + (mob.z - kz) * 10;
   834→    }
   835→    // endermen often warp away when hurt (MC behavior)
   836→    const ecb = cb ?? this.lastCb;
   837→    if (mob.type === 'enderman' && !mob.dead && ecb && Math.random() < 0.55) {
   838→      this.teleportNear(mob, mob.x, mob.z, 7, 15, ecb);
   839→    }
   840→    if (mob.health <= 0) {
   841→      mob.dead = true;
   842→      mob.deathT = 0.45;
   843→      if (mob.type === 'creeper' && mob.fuse >= 0) mob.fuse = -1;
   844→      return true;
   845→    }
   846→    return false;
   847→  }

   849→  /** enderman teleport: find a valid surface spot within [minR,maxR] of (cx,cz) */
   850→  private teleportNear(m: Mob, cx: number, cz: number, minR: number, maxR: number, cb: MobCallbacks): boolean {
   851→    const ox = m.x, oy = m.y, oz = m.z;
   852→    for (let i = 0; i < 14; i++) {
   853→      const a = Math.random() * Math.PI * 2;
   854→      const r = minR + Math.random() * (maxR - minR);
   855→      const x = Math.floor(cx + Math.cos(a) * r);
   856→      const z = Math.floor(cz + Math.sin(a) * r);
   857→      let sy = -1;
   858→      for (let y = 95; y > 2; y--) {
   859→        const id = this.world.getBlock(x, y, z);
   860→        if (id !== 0 && !isWaterId(id)) { sy = y; break; }
   861→      }
   862→      if (sy < 3) continue;
   863→      // head room for full height
   864→      const top = Math.ceil(m.def.height);
   865→      let clear = true;
   866→      for (let yy = sy + 1; yy <= sy + top; yy++) {
   867→        if (this.world.getBlock(x, yy, z) !== 0) { clear = false; break; }
   868→      }
   869→      if (!clear) continue;
   870→      m.x = x + 0.5; m.y = sy + 1; m.z = z + 0.5;
   871→      m.vx = 0; m.vy = 0; m.vz = 0;
   872→      m.state = 'idle'; m.stateTimer = 1;
   873→      cb.teleportParticles(ox, oy + m.height * 0.4, oz);
   874→      cb.teleportParticles(m.x, m.y + m.height * 0.4, m.z);
   875→      audio.enderTeleport();
   876→      return true;
   877→    }
   878→    return false;
   879→  }

   881→  /** effective light (0..15) at a mob's position */
   882→  private lightAt(m: Mob, sunLevel: number): number {
   883→    const l = this.world.getLight(Math.floor(m.x), Math.floor(m.y + 0.5), Math.floor(m.z));
   884→    if (l < 0) return 15;
   885→    return Math.max(l & 15, (l >> 4) * sunLevel);
   886→  }

   888→  /** creeper explosion */
   889→  private explode(x: number, y: number, z: number, cb: MobCallbacks): void {
   890→    audio.boom();
   891→    const R = 2.6;
   892→    for (let bx = Math.floor(x - R); bx <= Math.floor(x + R); bx++)
   893→      for (let by = Math.floor(y - R); by <= Math.floor(y + R); by++)
   894→        for (let bz = Math.floor(z - R); bz <= Math.floor(z + R); bz++) {
   895→          const d = Math.hypot(bx + 0.5 - x, by + 0.5 - y, bz + 0.5 - z);
   896→          if (d > R) continue;
   897→          const id = this.world.getBlock(bx, by, bz);
   898→          if (id !== 0 && id !== BLOCK.BEDROCK && id !== BLOCK.WATER) {
   899→            // chain reaction: TNT in the blast primes instead of vanishing
   900→            if (id === BLOCK.TNT) {
   901→              cb.igniteTnt?.(bx, by, bz);
   902→              continue;
   903→            }
   904→            this.world.setBlock(bx, by, bz, 0);
   905→          }
   906→        }
   907→    // damage nearby mobs (chain) + player
   908→    for (const m of this.mobs) {
   909→      if (m.dead) continue;
   910→      const d = Math.hypot(m.x - x, m.y - y, m.z - z);
   911→      if (d < R * 2 && m.type !== 'creeper') {
   912→        this.hurtMob(m, Math.max(1, Math.round(14 * (1 - d / (R * 2)))), m.x - x, m.z - z, cb);
   913→      }
   914→    }
   915→    const pd = Math.hypot(cb.playerX - x, cb.playerY - y, cb.playerZ - z);
   916→    if (pd < R * 2) {
   917→      cb.damagePlayer(Math.max(1, Math.round(16 * (1 - pd / (R * 2)))), x, z);
   918→    }
   919→    cb.explodeParticles(x, y, z);
   920→  }

   930→      for (const m of this.mobs) {
   931→        if (m.def.hostile) hostile++; else passive++;
   932→      }
   933→      if (this.mobs.length < 22 && Math.random() < 0.5) {
   934→        this.trySpawnMob(player.x, player.y, player.z, sunLevel, passive, hostile);

   938→    // ── mobs ──

   942→
   943→      // despawn far / invalid: NaN or below-world positions never heal on their
   944→      // own and NaN breaks the distance check (NaN > 64 === false), so mobs that
   945→      // glitch out would otherwise persist forever
   946→      if (
   947→        distToPlayer > 64 ||
   948→        !Number.isFinite(m.x) || !Number.isFinite(m.y) || !Number.isFinite(m.z) ||
   949→        m.y < -20
   950→      ) {
   951→        this.scene.remove(m.group);
   952→        this.scene.remove(m.parts.shadow);
   953→        this.mobs.splice(i, 1);

   955→      }
   956→
   957→      // death animation

   959→        m.deathT -= dt;

   961→        this.setTint(m, 1, 0.35, 0.35);
   962→        if (m.deathT <= 0) {
   963→          for (const drop of m.def.drops) {
   964→            const n = drop.min + Math.floor(Math.random() * (drop.max - drop.min + 1));
   965→            // sheep drop wool matching their color variant
   966→            const dropId = m.type === 'sheep' && drop.id === BLOCK.WOOL ? sheepWoolId(m.variant) : drop.id;
   967→            for (let d = 0; d < n; d++) {
   968→              cb.spawnDrop(dropId, m.x, m.y + 0.4, m.z);
   969→            }
   970→          }
   971→          cb.spawnXP(m.x, m.y + 0.4, m.z, m.def.hostile ? 5 : 1 + Math.floor(Math.random() * 3));
   972→          cb.deathParticles(m.x, m.y + m.height / 2, m.z);
   973→          this.scene.remove(m.group);
   974→          this.scene.remove(m.parts.shadow);
   975→          this.mobs.splice(i, 1);
   976→        }
   977→        continue;
   978→      }
   979→
   980→      // hurt tint decay
   981→      if (m.hurtT > 0) {
   982→        m.hurtT -= dt;
   983→        this.setTint(m, 1, 0.35, 0.35);
   984→        if (m.hurtT <= 0) this.setTint(m, 1, 1, 1);
   985→      }
   986→

   989→      if (m.ambientCd <= 0 && distToPlayer < 18) {
   990→        m.ambientCd = 5 + Math.random() * 9;
