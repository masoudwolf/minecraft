   194→type TierName = keyof typeof TIER_STATS;

   196→function makeTool(type: ToolDef['type'], tierName: TierName): ToolDef {
   197→  const t = TIER_STATS[tierName];
   198→  const dmg =
   199→    type === 'sword'
   200→      ? t.swordDmg
   201→      : type === 'axe'
   202→        ? t.swordDmg - 1
   203→        : type === 'pickaxe'
   204→          ? Math.max(2, t.swordDmg - 2)
   205→          : Math.max(2, t.swordDmg - 2);
   206→  return { type, tier: t.tier, speed: t.speed, dur: t.dur, dmg };
   207→}

   209→// ─── pixel-art helpers ────────────────────────────────────────────────────────
   210→function noiseRect(ctx: CanvasRenderingContext2D, colors: string[], x0: number, y0: number, w: number, h: number, seed = 1): void {
   211→  let s = seed;
   212→  const rnd = (): number => {
   213→    s = (s * 16807) % 2147483647;
   214→    return s / 2147483647;
   215→  };
   216→  for (let y = 0; y < h; y++)
   217→    for (let x = 0; x < w; x++) {
   218→      ctx.fillStyle = colors[Math.floor(rnd() * colors.length)];
   219→      ctx.fillRect(x0 + x, y0 + y, 1, 1);
   220→    }
   221→}
   222→
   223→function px(ctx: CanvasRenderingContext2D, x: number, y: number, color: string, w = 1, h = 1): void {
   224→  ctx.fillStyle = color;
   225→  ctx.fillRect(x, y, w, h);
   226→}

   245→/** diagonal stick handle from bottom-left to (hx,hy), 2px thick */
   246→function drawHandle(ctx: CanvasRenderingContext2D, hx: number, hy: number, len: number): void {
   247→  for (let i = 0; i < len; i++) {
   248→    px(ctx, hx - i, hy + i, HANDLE, 2, 1);
   249→    px(ctx, hx - i, hy + i + 1, HANDLE_DARK, 1, 1);
   250→  }
   251→}

   253→function drawPickaxe(ctx: CanvasRenderingContext2D, tier: TierName): void {
   254→  const [main, dark, light] = TIER_COLORS[tier];
   255→  drawHandle(ctx, 9, 5, 9); // handle from (9,5) down-left to (1,13)
   256→  // head: arc across the top-right
   257→  px(ctx, 5, 2, main, 7, 2);
   258→  px(ctx, 4, 4, main, 2, 2);
   259→  px(ctx, 12, 4, main, 2, 2);
   260→  px(ctx, 3, 6, main, 2, 2);

   263→  px(ctx, 4, 4, light, 1, 1);

   297→function drawSword(ctx: CanvasRenderingContext2D, tier: TierName): void {
   298→  const [main, dark, light] = TIER_COLORS[tier];
   299→  // blade diagonal (4,11)->(13,2), 2px wide
   300→  for (let i = 0; i < 9; i++) {
   301→    px(ctx, 4 + i, 11 - i, main, 2, 1);
   302→    if (i % 2 === 0) px(ctx, 4 + i, 10 - i, light, 1, 1);
   303→  }
   304→  px(ctx, 13, 2, main, 2, 2);
   305→  px(ctx, 14, 2, dark, 1, 1);
   306→  // guard (perpendicular)
   307→  px(ctx, 3, 9, dark, 1, 2);
   308→  px(ctx, 4, 11, dark, 2, 1);
   309→  px(ctx, 5, 12, dark, 1, 2);
   310→  // handle
   311→  px(ctx, 3, 12, HANDLE, 2, 1);
   312→  px(ctx, 2, 13, HANDLE, 2, 1);
   313→  px(ctx, 1, 14, HANDLE_DARK, 2, 1);
   314→}

   682→// ─── mining model ─────────────────────────────────────────────────────────────
   683→export interface BreakInfo {
   684→  /** seconds to break */
   685→  time: number;
   686→  /** will the block drop its item */
   687→  harvest: boolean;
   688→}
   689→
   690→export function breakInfo(def: BlockDef, tool?: ToolDef): BreakInfo {
   691→  if (!Number.isFinite(def.hardness)) return { time: Infinity, harvest: false };
   692→  const minTier = def.minTier ?? 0;
   693→  let speed = 1;
   694→  let harvest = def.tool ? minTier === 0 : true;
   695→  if (def.tool && tool && tool.type === def.tool) {
   696→    speed = tool.speed;
   697→    harvest = tool.tier >= minTier;
   698→  }
   699→  const base = harvest ? 1.5 : 5;
   700→  return { time: Math.max(0.05, (def.hardness * base) / speed), harvest };
   701→}

   703→// ─── Icon rendering (cached dataURLs) ────────────────────────────────────────
   704→const iconCache = new Map<number, string>();
   705→
   706→export function getItemIcon(itemId: number): string {
   707→  const cached = iconCache.get(itemId);
   708→  if (cached) return cached;
   709→  const def = ITEMS[itemId];
   710→  if (!def) return '';
   711→  const c = document.createElement('canvas');
   712→  c.width = 16; c.height = 16;
   713→  const ctx = c.getContext('2d')!;
   714→  ctx.imageSmoothingEnabled = false;
   715→  def.icon(ctx);
   716→  // scale up 4x for crisp display
   717→  const big = document.createElement('canvas');
   718→  big.width = 64; big.height = 64;
   719→  const bctx = big.getContext('2d')!;
   720→  bctx.imageSmoothingEnabled = false;
   721→  bctx.drawImage(c, 0, 0, 64, 64);
   722→  const url = big.toDataURL();
   723→  iconCache.set(itemId, url);
   724→  return url;
   725→}
