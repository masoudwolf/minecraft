    50→  // ── OWNED sun-shadow pass ────────────────────────────────────────────────
    51→  // three r186 renders shadow maps into a DepthTexture with compareFunction
    52→  // (sampler2DShadow) plus a color attachment holding INVERTED byte depth —
    53→  // neither matches the RGBA-packed sampler this pack was built on, and the
    54→  // compare-sampler variant wedges software GL. So the pack renders its OWN
    55→  // depth pass into a color RT (32-bit packed depth, plain sampler2D).

    57→  // DEPTH MATERIALS: per-mesh instead of a single scene.overrideMaterial, so
    58→  // cutout casters (tall grass, flowers, wheat, sugarcane, leaves) sample their
    59→  // real texture alpha in the shadow map — a plain override turned every cross
    60→  // plant into a SOLID X-shaped quad occluder = the "grass squashed flat on the
    61→  // ground" shadow blobs the user reported.

   126→  // ── settings ────────────────────────────────────────────────────────────────
   127→  applySettings(settings: Settings): void {
   128→    // work on a COPY — never mutate the zustand store object (the software-GL
   129→    // downgrade used to leak into saved settings and fight the UI toggles)
   130→    const gfx: GfxSettings = { ...settings.gfx };

   132→      // software GL: keep the fancy look but drop heavy features automatically
   133→      gfx.renderScale = Math.min(gfx.renderScale, 0.66);
   134→      gfx.shadows = Math.min(gfx.shadows, 1);
   135→      gfx.waterQuality = Math.min(gfx.waterQuality, 0);
   136→      gfx.cloudQuality = Math.min(gfx.cloudQuality, 0);
   137→      gfx.grassDensity = Math.min(gfx.grassDensity, 0.22);
   138→    }
   139→    this.gfx = gfx;
   140→
   141→    // internal resolution scale
   142→    const base = Math.min(window.devicePixelRatio, 2);
   143→    this.renderer.setPixelRatio(base * CLAMP(gfx.renderScale, 0.4, 1));
   144→    this.resize();
   145→
   146→    // post-processing
   147→    if (gfx.postfx) {
   148→      if (!this.postfx) {
   149→        this.postfx = new PostFX(this.renderer, this.scene, this.camera, {
   150→          bloom: gfx.bloom, godRays: gfx.godRays, grade: true, fxaa: gfx.fxaa,
   151→        });
   152→      }
   153→      this.postfx.setBloomEnabled(gfx.bloom);
   154→      this.postfx.setGodRaysEnabled(gfx.godRays);
   155→      this.postfx.setFxaaEnabled(gfx.fxaa);
   156→      this.postfx.setGrade(gfx.exposure, gfx.saturation, gfx.contrast, gfx.vignette);
   157→    }

   159→    // sky + clouds
   160→    if (!this.atmosphere) this.atmosphere = new AtmosphereSky(this.scene);
   161→    if (!this.clouds) this.clouds = new VolumetricClouds(this.scene);
   162→    this.atmosphere.mesh.visible = true;
   163→    this.clouds.mesh.visible = gfx.volumetricClouds;
   164→
   165→    // shadows
   166→    this.setShadowQuality(gfx.shadows);
   167→
   168→    // water quality
   169→    if (this.waterMat) {
   170→      this.waterMat.uniforms.uWaterQ.value = gfx.waterQuality;
   171→      this.waterMat.uniforms.uHasRefl.value = gfx.waterQuality >= 1 ? 1 : 0;
   172→    }
   173→    if (gfx.waterQuality >= 1) this.ensureReflRT(gfx.waterQuality);
   174→

   176→    if (this.grass.density !== gfx.grassDensity || !this.grass.enabled) {
   177→      this.grass.enabled = gfx.grassDensity > 0.02;
   178→      if (this.world) this.grass.setDensity(gfx.grassDensity, this.world);
   179→      else this.grass.density = gfx.grassDensity;
   180→    }
   181→  }

   183→  // ── shadows ─────────────────────────────────────────────────────────────────
   184→  setShadowQuality(q: number): void {
   185→    const sizes = [0, 1024, 2048, 4096];
   186→    shadowState.realShadows = q > 0; // mobs/engine read this to hide blob circles
   187→    if (q <= 0) {
   188→      if (this.shadowRT) { this.shadowRT.dispose(); this.shadowRT = null; }
   189→      this.renderer.shadowMap.enabled = false;
   190→      this.setShadowStrength(0);
   191→      this.maxTorchShadows = 0;
   192→      this.refreshChunkShadowFlags();
   193→      return;
   194→    }
   195→    const size = sizes[q] ?? 2048;
   196→    // three's own shadow pipeline stays OFF (no casting light) — shadowMap.enabled
   197→    // only drives the vanilla blob-shadow suppression + receiver flag semantics;
   198→    // the REAL depth map is our own pass (runShadowPass).
   199→    this.renderer.shadowMap.enabled = true;
   200→    this.renderer.shadowMap.type = THREE.PCFShadowMap;
   201→    this.sunLight.castShadow = false;
   202→    if (!this.shadowRT) {
   203→      this.shadowRT = new THREE.WebGLRenderTarget(size, size, {
   204→        minFilter: THREE.NearestFilter,
   205→        magFilter: THREE.NearestFilter,
   206→        depthBuffer: true,
   207→      });
   208→    } else if (this.shadowRT.width !== size) {
   209→      this.shadowRT.setSize(size, size);
   210→    }
   211→    // torch cube shadows: 1 source on High, 2 on Ultra (each = 6 tiny renders).
   212→    // Software GL (llvmpipe) freezes intermittently inside the cube passes —
   213→    // real GPUs get the feature, the sandbox falls back to unshadowed torches.
   214→    this.maxTorchShadows = this.isSoftware ? 0 : (q >= 3 ? 2 : 1);
   215→    while (this.cubeRTs.length < this.maxTorchShadows) {
   216→      const rt = new THREE.WebGLCubeRenderTarget(384, {
   217→        // LinearFilter: free bilinear softening of the cube penumbra (the
   218→        // shader adds a tangent-frame 4-tap PCF on top of this)
   219→        minFilter: THREE.LinearFilter,
   220→        magFilter: THREE.LinearFilter,
   221→        generateMipmaps: false,
   222→      });
   223→      this.cubeRTs.push(rt);
   224→      this.cubeCams.push(new THREE.CubeCamera(0.1, 24, rt));
   225→    }
   226→    const half = CLAMP((this.gfx?.renderScale ?? 4) * 0 + 56, 48, 96);
   227→    const cam = this.shadowCam;
   228→    cam.left = -half; cam.right = half; cam.top = half; cam.bottom = -half;
   229→    cam.near = 1; cam.far = 320;
   230→    cam.updateProjectionMatrix();
   231→    this.setShadowStrength(1);
   232→    if (this.scene.getObjectById(this.sunLight.target.id) == null) this.scene.add(this.sunLight.target);
   233→    this.refreshChunkShadowFlags();
   234→  }

   236→  /** re-apply caster/receiver flags to ALREADY-BUILT chunk meshes (they were
   237→   *  flagged at mesh time — toggling shadows later leaves stale flags and the
   238→   *  world would stop casting shadows until every chunk remeshed) */
   239→  private refreshChunkShadowFlags(): void {
   240→    if (!this.world) return;
   241→    const shadowsOn = this.renderer.shadowMap.enabled;
   242→    for (const chunk of this.world.chunks.values()) {
   243→      const m = chunk.meshes;
   244→      if (!m) continue;
   245→      if (m.opaque) { m.opaque.castShadow = shadowsOn; m.opaque.receiveShadow = shadowsOn; }
   246→      if (m.cutout) { m.cutout.castShadow = shadowsOn; m.cutout.receiveShadow = shadowsOn; }
   247→      if (m.water) { m.water.castShadow = false; m.water.receiveShadow = true; }
   248→    }
   249→  }

   251→  /** true when the object (or any descendant ≤5 levels down) wants to cast —
   252→   *  covers top-level chunk meshes and entity Groups with mesh children
   253→   *  (player model nests Group→pivot→Mesh at depth 2-3; depth 5 is headroom) */
   254→  private isShadowCaster(o: THREE.Object3D, depth = 0): boolean {
   255→    if (o.castShadow === true) return true;
   256→    if (depth >= 5) return false;
   257→    for (const c of o.children) {
   258→      if (this.isShadowCaster(c, depth + 1)) return true;
   259→    }
   260→    return false;
   261→  }

   263→  /** alpha-aware depth material for a source material: cutout casters keep
   264→   *  their texture alpha in the shadow map (cross plants / leaves / hats),
   265→   *  everything else uses the plain packed-depth material */
   266→  private depthMatFor(src: THREE.Material): THREE.MeshDepthMaterial {
   267→    let d = this.depthFor.get(src);
   268→    if (!d) {
   269→      const lm = src as THREE.MeshLambertMaterial;
   270→      if (lm.alphaTest > 0 && lm.map) {
   271→        d = new THREE.MeshDepthMaterial({
   272→          depthPacking: THREE.RGBADepthPacking,
   273→          side: THREE.DoubleSide,
   274→          map: lm.map,
   275→          alphaTest: lm.alphaTest,
   276→        });
   277→      } else {
   278→        d = this.depthMatPlain;
   279→      }
   280→      this.depthFor.set(src, d);
   281→    }
   282→    return d;
   283→  }

   285→  private voxelDepthSeeded = false;
   286→  /** the voxel cutout material is a ShaderMaterial (no .map property) — seed
   287→   *  its depth variant with the ATLAS texture + its alphaTest explicitly, so
   288→   *  tall grass / flowers / wheat / leaves cut real holes in the shadow map
   289→   *  instead of occluding as solid quads (the "grass squashed flat" blobs) */
   290→  private seedVoxelDepthMats(): void {
   291→    if (this.voxelDepthSeeded || !this.world) return;
   292→    const cut = this.world.getMaterial('cutout');
   293→    if (cut) {
   294→      const atlas = cut.uniforms.uAtlas?.value as THREE.Texture | undefined;
   295→      if (atlas) {
   296→        this.depthFor.set(cut, new THREE.MeshDepthMaterial({
   297→          depthPacking: THREE.RGBADepthPacking,
   298→          side: THREE.DoubleSide,
   299→          map: atlas,
   300→          alphaTest: (cut.uniforms.uAlphaTest?.value as number) ?? 0.5,
   301→        }));
   302→        this.voxelDepthSeeded = true;
   303→      }
   304→    }
   305→  }

   307→  /** walk the visible part of the scene under `root`, swapping every
   308→   *  castShadow mesh to its depth material (recorded for restore) */
   309→  private swapToDepthMaterials(root: THREE.Object3D, parentVisible: boolean): void {
   310→    const visible = parentVisible && root.visible;
   311→    if (!visible) return;
   312→    if (root instanceof THREE.Mesh && root.castShadow === true) {
   313→      this.matSwaps.push({ mesh: root, mat: root.material });
   314→      if (Array.isArray(root.material)) {
   315→        root.material = root.material.map((mm) => this.depthMatFor(mm));
   316→      } else {
   317→        root.material = this.depthMatFor(root.material);
   318→      }
   319→    }
   320→    for (const c of root.children) this.swapToDepthMaterials(c, visible);
   321→  }
   322→
   323→  /** stash + toggle top-level visibility so only casters render, then swap
   324→   *  caster meshes to depth materials. Returns a restore closure. */
   325→  private beginCasterSession(): () => void {
   326→    this.visSwaps.length = 0;
   327→    this.matSwaps.length = 0;
   328→    const list = this.scene.children;
   329→    for (let i = 0; i < list.length; i++) {
   330→      const o = list[i];
   331→      this.visSwaps.push({ o, v: o.visible });
   332→      o.visible = o.visible !== false && this.isShadowCaster(o);
   333→    }
   334→    this.swapToDepthMaterials(this.scene, true);
   335→    const restore = (): void => {
   336→      for (const s of this.matSwaps) s.mesh.material = s.mat;
   337→      for (const s of this.visSwaps) s.o.visible = s.v;
   338→    };
   339→    return restore;
   340→  }

   342→  /** render the scene from the sun into our RGBA-packed depth RT */
   343→  private runShadowPass(): void {
   344→    if (!this.shadowRT) return;
   345→    this.seedVoxelDepthMats();
   346→    const cam = this.shadowCam;
   347→    const t = this.shadowTarget;
   348→    const d = this.shadowLightDir;
   349→    const h = d.y > 0.18 ? d.y : 0.18;
   350→    cam.position.set(t.x + d.x * 90, t.y + h * 90, t.z + d.z * 90);
   351→    cam.up.set(0, 1, 0);
   352→    cam.lookAt(t);
   353→    cam.updateMatrixWorld();
   354→
   355→    const prevRT = this.renderer.getRenderTarget();
   356→    this.renderer.getClearColor(this.tmpClearColor);
   357→    const prevClearAlpha = this.renderer.getClearAlpha();
   358→
   359→    const restore = this.beginCasterSession();
   360→    this.renderer.setRenderTarget(this.shadowRT);
   361→    this.renderer.setClearColor(SHADOW_CLEAR, 0);
   362→    this.renderer.clear();
   363→    this.renderer.render(this.scene, cam);
   364→
   365→    // torch / dynamic point-light cube depth maps (same caster session — the
   366→    // material swaps and visibility stashes stay valid for every face)
   367→    if (this.maxTorchShadows > 0 && this.torchSources.length > 0) {
   368→      for (let i = 0; i < this.torchSources.length && i < this.maxTorchShadows; i++) {
   369→        const src = this.torchSources[i];
   370→        const cube = this.cubeCams[i];
   371→        if (!cube) continue;
   372→        cube.position.copy(src.pos);
   373→        cube.updateMatrixWorld();
   374→        // the 6 face cameras are CubeCamera's children — near/far live THERE
   375→        // (the CubeCamera wrapper has no updateProjectionMatrix of its own)
   376→        for (const face of cube.children) {
   377→          const fc = face as THREE.PerspectiveCamera;
   378→          if (fc.isPerspectiveCamera) {
   379→            fc.near = 0.1;
   380→            fc.far = src.far;
   381→            fc.updateProjectionMatrix();
   382→          }
   383→        }
   384→        this.renderer.setRenderTarget(null);
   385→        this.renderer.setClearColor(SHADOW_CLEAR, 0);
   386→        cube.update(this.renderer, this.scene);
   387→      }
   388→    }
   389→
   390→    restore();
   391→    this.renderer.setClearColor(this.tmpClearColor, prevClearAlpha);
   392→    this.renderer.setRenderTarget(prevRT);
   393→
   394→    // texture matrix: raw proj*view — gfxShadow adds the NDC bias itself
   395→    this.shadowMatrix.multiplyMatrices(cam.projectionMatrix, cam.matrixWorldInverse);
   396→    const sz = this.shadowRT.width;
   397→    const texel = new THREE.Vector2(1 / sz, 1 / sz);
   398→    for (const m of this.voxelMats()) {
   399→      m.uniforms.uShadowMap.value = this.shadowRT.texture;
   400→      m.uniforms.uShadowMatrix.value.copy(this.shadowMatrix);
   401→      (m.uniforms.uShadowTexel.value as THREE.Vector2).set(1 / sz, 1 / sz);
   402→    }
   403→    this.grass.setShadowUniforms(this.shadowRT.texture, this.shadowMatrix, texel);
   404→    // torch cube uniforms → opaque + cutout (+ grass mat via GrassManager)
   405→    for (let i = 0; i < 2; i++) {
   406→      const src = this.torchSources[i];
   407→      const active = i < this.maxTorchShadows && !!src && !!this.cubeRTs[i];
   408→      const tex = active ? this.cubeRTs[i].texture : null;
   409→      for (const m of this.voxelMats()) {
   410→        if (!m.uniforms.uTorchCount) continue;
   411→        (m.uniforms[`uTorchPos${i}`].value as THREE.Vector3).copy(src ? src.pos : this.shadowTarget);
   412→        m.uniforms[`uTorchMap${i}`].value = tex;
   413→        m.uniforms[`uTorchFar${i}`].value = src ? src.far : 24;
   414→        m.uniforms[`uTorchRange${i}`].value = src ? src.range : 0;
   415→      }
   416→      this.grass.setTorchUniforms(i, src ? src.pos : null, tex, src ? src.far : 24, src ? src.range : 0);
   417→    }
   418→    for (const m of this.voxelMats()) {
   419→      if (m.uniforms.uTorchCount) m.uniforms.uTorchCount.value = this.maxTorchShadows > 0 && this.torchSources.length > 0 ? Math.min(this.maxTorchShadows, this.torchSources.length) : 0;
   420→    }
   421→    this.grass.setTorchCount(this.torchSources.length > 0 ? Math.min(this.maxTorchShadows, this.torchSources.length) : 0);
   422→  }
   423→
   424→  private setShadowStrength(s: number): void {
   425→    for (const m of this.voxelMats()) {
   426→      if (m.uniforms.uShadowStrength) m.uniforms.uShadowStrength.value = s;
   427→    }
   428→  }

   430→  /** all voxel materials that participate in sun shadows (terrain + water) */
   431→  private voxelMats(): THREE.ShaderMaterial[] {
   432→    if (!this.world) return [];
   433→    return [
   434→      this.world.getMaterial('opaque'),

   436→      this.world.getMaterial('water'),

   440→  // ── chunk hooks ─────────────────────────────────────────────────────────────
   441→  onChunkMeshed(chunk: Chunk, world: World): void {
   442→    if (this.gfx && this.gfx.grassDensity > 0.02) this.grass.updateChunk(chunk, world);
   443→    const m = chunk.meshes;
   444→    if (!m) return;
   445→    const shadowsOn = this.renderer.shadowMap.enabled;
   446→    if (m.opaque) { m.opaque.castShadow = shadowsOn; m.opaque.receiveShadow = shadowsOn; }
   447→    if (m.cutout) { m.cutout.castShadow = shadowsOn; m.cutout.receiveShadow = shadowsOn; }
   448→    if (m.water) { m.water.castShadow = false; m.water.receiveShadow = true; }
   449→  }

   451→  // ── per-frame update ────────────────────────────────────────────────────────
   452→  update(dt: number, camera: THREE.PerspectiveCamera, sky: SkySystem, playerX: number, playerY: number, playerZ: number, underwater: boolean): void {
   453→    if (!this.gfx || !this.atmosphere) return;
   454→    const gfx = this.gfx;
   455→
   456→    // sun direction (same formula as SkySystem)
   457→    const dayFrac = sky.time / DAY_LENGTH;
   458→    const angle = (dayFrac - 0.25) * Math.PI * 2;
   459→    const sunDir = this.tmpV1.set(Math.cos(angle), Math.sin(angle), 0.18).normalize();
   460→    const sunHeight = Math.sin(angle);
   461→    const dayAmount = CLAMP((sunHeight + 0.12) / 0.32, 0, 1);
   462→    const sunsetAmount = CLAMP(1 - Math.abs(sunHeight) / 0.28, 0, 1) * (dayAmount > 0.02 ? 1 : 0.35);
   463→    this.lastSunDir.copy(sunDir);
   464→
   465→    // cloud coverage: base scattered + weather
   466→    const cover = CLAMP(0.22 + sky.weatherDarkness * 1.1, 0, 1);
   467→    const storm = sky.weatherDarkness;
   468→
   469→    // sun color: warm at low sun, white at noon, moon-dim at night
   470→    const lightDir = this.tmpV4.copy(sunDir); // the ACTUAL light source (moon at night)
   471→    const sunCol = this.tmpV2;
   472→    const warmth = sunsetAmount;
   473→    sunCol.set(
   474→      1.0,
   475→      0.92 - warmth * 0.28,
   476→      0.82 - warmth * 0.55,
   477→    ).multiplyScalar(0.14 + 0.86 * dayAmount * (1 - storm * 0.55));
   478→    if (dayAmount < 0.08) {
   479→      // moonlight — dim cool directional light from the moon's position
   480→      sunCol.set(0.35, 0.42, 0.62).multiplyScalar(0.28);
   481→      lightDir.multiplyScalar(-1);
   482→    }
   483→    this.lastSunColor.setRGB(sunCol.x, sunCol.y, sunCol.z);
   484→
   485→    this.atmosphere.update(camera, sunDir, dayAmount, sunsetAmount, storm, sky.lightningFlash, cover);
   486→    if (this.clouds?.mesh.visible) {
   487→      this.clouds.update(camera, dt, lightDir, this.lastSunColor, dayAmount, cover, storm);
   488→    }
   489→
   490→    // fog snapshot for grass
   491→    if (this.scene.fog) {
   492→      const fog = this.scene.fog as THREE.Fog;
   493→      this.lastFogColor.copy(fog.color);
   494→      this.lastFogNear = fog.near;
   495→      this.lastFogFar = fog.far;
   496→    }
   497→    this.grass.update(performance.now() / 1000, sky.sunLevel, this.lastFogColor, this.lastFogNear, this.lastFogFar);
   498→
   499→    // voxel sun/shadow uniforms — sun dir + strength; the MAP itself is filled
   500→    // by runShadowPass() every frame (terrain opaque/cutout AND water share it)
   501→    const shadowOn = (gfx.shadows ?? 0) > 0 && this.shadowRT !== null;
   502→    for (const m of this.voxelMats()) {
   503→      if (m.uniforms.uSunDirW) m.uniforms.uSunDirW.value.copy(lightDir);
   504→      if (m.uniforms.uSunColorW) m.uniforms.uSunColorW.value.copy(this.lastSunColor);
   505→      if (m.uniforms.uShadowStrength) m.uniforms.uShadowStrength.value = shadowOn ? 1 : 0;
   506→    }
   507→    this.grass.setShadowStrength(shadowOn ? 1 : 0, lightDir);
   508→
   509→    // shadow target follows the player (snapped to block grid — no shimmer)
   510→    if (shadowOn) {
   511→      const px = Math.round(playerX);
   512→      const py = Math.round(playerY);
   513→      const pz = Math.round(playerZ);
   514→      this.shadowTarget.set(px, py, pz);
   515→      this.shadowLightDir.copy(lightDir).normalize();
   516→      // nearest torch/dynamic-light sources for the cube shadow slots
   517→      this.torchTimer -= dt;
   518→      if (this.torchTimer <= 0) {
   519→        this.torchTimer = 0.3;
   520→        this.pickTorchSources(playerX, playerY, playerZ);
   521→      }
   522→    }
   523→    // sun light still follows the player for entity Lambert shading
   524→    if (this.renderer.shadowMap.enabled) {
   525→      const px = Math.round(playerX);
   526→      const py = Math.round(playerY);
   527→      const pz = Math.round(playerZ);
   528→      this.sunLight.target.position.set(px, py, pz);
   529→      this.sunLight.target.updateMatrixWorld();
   530→      const h = lightDir.y > 0.18 ? lightDir.y : 0.18;
   531→      this.sunLight.position.set(px + lightDir.x * 90, py + h * 90, pz + lightDir.z * 90);
   532→    }
   533→
   534→    this.underwaterCam = underwater;
   535→
   536→    // entity sweeps (cheap, throttled)
   537→    this.sweepTimer += dt;
   538→    if (this.sweepTimer > 0.5) {
   539→      this.sweepTimer = 0;
   540→      this.sweepEntities();
   541→    }
   542→    this.grassSweepTimer += dt;
   543→    if (this.grassSweepTimer > 2) {
   544→      this.grassSweepTimer = 0;
   545→      if (this.world) this.grass.sweep(this.world);
   546→    }
   547→
   548→    // postfx per-frame uniforms
   549→    if (this.postfx) {
   550→      // sun screen position for god rays
   551→      let gx = -1, gy = -1, strength = 0;
   552→      if (gfx.godRays && !underwater) {
   553→        const sd = this.tmpV3.copy(sunDir).multiplyScalar(600).add(camera.position);
   554→        sd.project(camera);
   555→        if (sd.z < 1) {
   556→          gx = (sd.x + 1) / 2;
   557→          gy = (sd.y + 1) / 2;
   558→          // wide window: shafts still work indoors when the sun sits well
   559→          // off-screen behind a window/door (light keeps streaming through
   560→          // the bright opening pixels toward the sun direction)
   561→          const onScreen = gx > -0.55 && gx < 1.55 && gy > -0.55 && gy < 1.55;
   562→          const dayBoost = CLAMP(sunHeight * 4 + 0.35, 0, 1); // strongest near horizon
   563→          strength = onScreen ? gfx.godRaysStrength * dayBoost * (1 - storm * 0.85) * (1 - cover * 0.55) : 0;
   564→        }
   565→      }
   566→      const samples = gfx.preset === 'ultra' || gfx.preset === 'high' ? 48 : gfx.preset === 'medium' ? 36 : 24;
   567→      this.postfx.setSunScreen(gx, gy, strength, samples);
   568→      this.postfx.setUnderwater(underwater ? 1 : 0, performance.now() / 1000);
   569→    }
   570→  }
   571→
   572→  // ── entity shadow flags + blob-shadow suppression ──────────────────────────
   573→  // RECURSIVE walk: the old version only inspected TOP-LEVEL scene children,
   574→  // so entity Groups (player model, mob models, boats, drops) were never
   575→  // flagged castShadow and nothing but terrain ever rendered into the depth
   576→  // map — mobs/player kept their vanilla blob circles and cast no light-based
   577→  // shadow at all (user report). Now every MeshLambertMaterial mesh anywhere
   578→  // gets flagged, and scene PointLights are collected as torch-shadow sources.
   579→  private sweepEntities(): void {
   580→    const shadowsOn = this.renderer.shadowMap.enabled;
   581→    this.scenePointLights.length = 0;
   582→    const walk = (o: THREE.Object3D): void => {
   583→      // the camera subtree holds the FIRST-PERSON HAND — it must never cast
   584→      // (it lives at the eye position; a depth-pass entry would paint a
   585→      // floating box shadow right under the player)
   586→      if (o === this.camera) return;
   587→      if (o instanceof THREE.PointLight && o.visible && this.scenePointLights.length < 8) {
   588→        this.scenePointLights.push(o);
   589→      }
   590→      if (o instanceof THREE.Mesh) {
   591→        const lm = o.material as THREE.Material | THREE.Material[];
   592→        const first = Array.isArray(lm) ? lm[0] : lm;
   593→        // blob shadows (vanilla-style dark circles) → hidden while real
   594→        // shadows render (mobs.ts also consults shadowState every frame)
   595→        if (first && (first as THREE.MeshBasicMaterial).isMeshBasicMaterial) {
   596→          const geo = o.geometry;
   597→          if (geo instanceof THREE.CircleGeometry && (first as THREE.MeshBasicMaterial).color.getHex() === 0x000000) {
   598→            o.visible = !shadowsOn && o.visible !== false;
   599→            return;
   600→          }
   601→        }
   602→        if (first && (first as THREE.MeshLambertMaterial).isMeshLambertMaterial && !o.userData.__gfxShadow) {
   603→          o.userData.__gfxShadow = true;
   604→          o.castShadow = true;
   605→          o.receiveShadow = true;
   606→        }
   607→      }
   608→      for (const c of o.children) walk(c);
   609→    };
   610→    for (const o of this.scene.children) walk(o);
   611→  }
   612→
   613→  /** nearest torch/point-light sources for the cube-shadow slots */
   614→  private pickTorchSources(px: number, py: number, pz: number): void {
   615→    this.torchSources.length = 0;
   616→    if (this.maxTorchShadows <= 0 || !this.world) return;
   617→    const cands: Array<{ pos: THREE.Vector3; range: number; far: number; d2: number }> = [];
   618→    // block torches (chunk.torches is rebuilt by the mesher — always current)
   619→    for (const chunk of this.world.chunks.values()) {
   620→      const arr = chunk.torches;
   621→      if (!arr || arr.length === 0) continue;
   622→      // cheap chunk-level reject
   623→      if (Math.abs(chunk.cx * 16 + 8 - px) > 40 || Math.abs(chunk.cz * 16 + 8 - pz) > 40) continue;
   624→      for (const t of arr) {
   625→        const dx = t[0] - px, dy = t[1] - py, dz = t[2] - pz;
   626→        const d2 = dx * dx + dy * dy + dz * dz;
   627→        if (d2 < 26 * 26) cands.push({ pos: new THREE.Vector3(t[0], t[1], t[2]), range: 14, far: 17, d2 });
   628→      }
   629→    }
   630→    // dynamic scene point lights (knight sword glow, future lanterns…)
   631→    for (const l of this.scenePointLights) {
   632→      const p = l.getWorldPosition(new THREE.Vector3());
   633→      const d2 = p.distanceToSquared(this.tmpV1.set(px, py, pz));
   634→      const dist = l.distance > 0 ? l.distance : 14;
   635→      if (d2 < 26 * 26) cands.push({ pos: p, range: Math.min(dist, 20), far: Math.min(dist, 20) + 3, d2 });
   636→    }
   637→    cands.sort((a, b) => a.d2 - b.d2);
   638→    for (let i = 0; i < cands.length && i < this.maxTorchShadows; i++) {
   639→      this.torchSources.push(cands[i]);
   640→    }
   641→  }
   642→
   643→  // ── reflection RT ───────────────────────────────────────────────────────────
   644→  private ensureReflRT(quality: number): void {
   645→    const w = Math.floor(window.innerWidth * (quality >= 2 ? 0.75 : 0.5));
   646→    const h = Math.floor(window.innerHeight * (quality >= 2 ? 0.75 : 0.5));
   647→    if (!this.reflRT) {
   648→      this.reflRT = new THREE.WebGLRenderTarget(w, h, {
   649→        type: THREE.HalfFloatType,
   650→        depthBuffer: true,
   651→      });
   652→    } else if (this.reflRT.width !== w || this.reflRT.height !== h) {
   653→      this.reflRT.setSize(w, h);
   654→    }
   655→  }
   656→
   657→  private renderReflection(): void {
   658→    if (!this.world || !this.reflRT || !this.waterMat) return;
   659→    const cam = this.camera;
   660→
   661→    // hide all water meshes
   662→    for (const chunk of this.world.chunks.values()) {
   663→      if (chunk.meshes?.water) chunk.meshes.water.visible = false;
   664→    }
   665→
   666→    // mirror camera about the horizontal plane y = WATER_PLANE_Y
   667→    const n = this.tmpV1.set(0, 1, 0);
   668→    const pos = this.tmpV2.copy(cam.position);
   669→    const refl = this.tmpV3.copy(pos).addScaledVector(n, -2 * (pos.y - WATER_PLANE_Y));
   670→    this.mirrorCam.position.copy(refl);
   671→    const dir = cam.getWorldDirection(new THREE.Vector3());
   672→    dir.y = -dir.y; // reflect view direction about the horizontal plane
   673→    const target = refl.clone().add(dir);
   674→    this.mirrorCam.up.set(0, 1, 0);
   675→    this.mirrorCam.up.y = -1; // reflected up vector keeps texture orientation
   676→    this.mirrorCam.lookAt(target);
   677→    this.mirrorCam.fov = cam.fov;
   678→    this.mirrorCam.aspect = cam.aspect;
   679→    this.mirrorCam.near = cam.near;
   680→    this.mirrorCam.far = cam.far;
   681→    this.mirrorCam.updateProjectionMatrix();
   682→    this.mirrorCam.updateMatrixWorld();
   683→
   684→    // oblique near-plane clipping — everything below the water plane is cut
   685→    this.tmpPlane.setFromNormalAndCoplanarPoint(n, new THREE.Vector3(0, WATER_PLANE_Y, 0));
   686→    this.tmpPlane.applyMatrix4(this.mirrorCam.matrixWorldInverse);
   687→    const clip = this.tmpPlane.normal.clone().multiplyScalar(-1);
   688→    const constant = -this.tmpPlane.constant;
   689→    const clipPlane = new THREE.Vector4(clip.x, clip.y, clip.z, constant);
   690→    const pm = this.mirrorCam.projectionMatrix;
   691→    this.tmpQ.x = (Math.sign(clipPlane.x) + pm.elements[8]) / pm.elements[0];
   692→    this.tmpQ.y = (Math.sign(clipPlane.y) + pm.elements[9]) / pm.elements[5];
   693→    this.tmpQ.z = -1;
   694→    this.tmpQ.w = (1 + pm.elements[10]) / pm.elements[14];
   695→    clipPlane.multiplyScalar(2 / clipPlane.dot(this.tmpQ));
   696→    pm.elements[2] = clipPlane.x;
   697→    pm.elements[6] = clipPlane.y;
   698→    pm.elements[10] = clipPlane.z + 1;
   699→    pm.elements[14] = clipPlane.w;
   700→
   701→    // texture matrix: NDC-bias * proj * view
   702→    this.reflMatrix.set(
   703→      0.5, 0, 0, 0.5,
   704→      0, 0.5, 0, 0.5,
   705→      0, 0, 0.5, 0.5,
   706→      0, 0, 0, 1,
   707→    );
   708→    this.reflMatrix.multiply(pm);
   709→    this.reflMatrix.multiply(this.mirrorCam.matrixWorldInverse);
   710→    this.waterMat.uniforms.uReflMatrix.value.copy(this.reflMatrix);
   711→    this.waterMat.uniforms.uReflMap.value = this.reflRT.texture;
   712→
   713→    const prevRT = this.renderer.getRenderTarget();
   714→    this.renderer.setRenderTarget(this.reflRT);
   715→    this.renderer.clear();
   716→    this.renderer.render(this.scene, this.mirrorCam);
   717→    this.renderer.setRenderTarget(prevRT);
   718→
   719→    // restore water visibility
   720→    for (const chunk of this.world.chunks.values()) {
   721→      if (chunk.meshes?.water) chunk.meshes.water.visible = true;
   722→    }
   723→  }
   724→
   725→  // ── render ──────────────────────────────────────────────────────────────────
   726→  render(_dt: number): void {
   727→    if (this.gfx && this.gfx.shadows > 0 && this.shadowRT) {
   728→      this.runShadowPass();
   729→    }
   730→    const useComposer = this.gfx?.postfx === true && this.postfx !== null;
   731→    if (useComposer && this.gfx!.waterQuality >= 1 && !this.underwaterCam) {
   732→      this.renderReflection();
   733→    }
   734→    // truthful triangle stats: composer passes would reset info per pass
   735→    this.renderer.info.autoReset = false;
   736→    this.renderer.info.reset();
   737→    if (useComposer) {
   738→      this.postfx!.render(_dt);
   739→    } else {
   740→      this.renderer.render(this.scene, this.camera);
   741→    }
   742→    this.renderer.info.autoReset = true;
   743→  }
   744→
   745→  resize(): void {
   746→    const w = window.innerWidth;
   747→    const h = window.innerHeight;
   748→    this.postfx?.setSize(w, h);
   749→  }
   750→
   751→  dispose(): void {
   752→    this.atmosphere?.dispose();
   753→    this.clouds?.dispose();
   754→    this.grass.dispose();
   755→    this.postfx?.dispose();
   756→    this.reflRT?.dispose();
   757→    this.shadowRT?.dispose();
   758→    for (const rt of this.cubeRTs) rt.dispose();
   759→    this.cubeRTs.length = 0;
   760→    this.cubeCams.length = 0;
   761→    for (const d of this.depthFor.values()) if (d !== this.depthMatPlain) d.dispose();
   762→    this.depthFor.clear();
   763→    this.depthMatPlain.dispose();
   764→  }
