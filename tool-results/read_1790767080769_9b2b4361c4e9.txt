   174→  constructor(canvas: HTMLCanvasElement) {
   175→    this.canvas = canvas;

   179→    // swiftshader) reject 'high-performance' contexts — retry plain, then
   180→    // low-power without the perf caveat, before giving up
   181→    const attempts: THREE.WebGLRendererParameters[] = [
   182→      { canvas, antialias: false, powerPreference: 'high-performance' },
   183→      { canvas, antialias: false },
   184→      { canvas, antialias: false, powerPreference: 'low-power', failIfMajorPerformanceCaveat: false },
   185→    ];
   186→    let renderer: THREE.WebGLRenderer | null = null;
   187→    for (const opts of attempts) {
   188→      try {
   189→        renderer = new THREE.WebGLRenderer(opts);
   190→        break;
   191→      } catch {
   192→        // next fallback
   193→      }
   194→    }
   195→    if (!renderer) throw new Error('WebGL is not available in this browser');
   196→    this.renderer = renderer;
   197→    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
   198→    this.renderer.setSize(window.innerWidth, window.innerHeight);
   199→    this.renderer.outputColorSpace = THREE.SRGBColorSpace;
   200→
   201→    this.scene = new THREE.Scene();

   267→  /** create a persistent world in the DB, then play it */
   268→  async createWorld(name: string, gameMode: GameMode, seed?: number): Promise<void> {
   269→    const store = useGameStore.getState();
   270→    try {
   271→      const res = await fetch('/api/worlds', {
   272→        method: 'POST',
   273→        headers: { 'Content-Type': 'application/json' },
   274→        body: JSON.stringify({ name, gameMode, seed }),
   275→      });
   276→      if (!res.ok) throw new Error('create failed');
   277→      const { world } = (await res.json()) as { world: { id: string; name: string; seed: number } };
   278→      this.currentWorldId = world.id;
   279→      this.currentWorldName = world.name;
   280→      store.setCurrentWorld(world.id, world.name);
   281→      this.setupWorld(world.seed, null, gameMode);
   282→      await this.preloadSpawn();
   283→      this.enterPlaying();
   284→      this.saveGame(); // persist spawn baseline immediately
   285→    } catch (err) {
   286→      console.error('createWorld failed', err);
   287→      // offline fallback: play a local-only world
   288→      const s = seed ?? Math.floor(Math.random() * 2147483647);
   289→      this.currentWorldId = null;
   290→      this.currentWorldName = name;
   291→      store.setCurrentWorld(null, name);
   292→      this.setupWorld(s, null, gameMode);
   293→      await this.preloadSpawn();
   294→      this.enterPlaying();
   295→    }
   296→  }

   298→  /** load a DB world and play it */
   299→  async loadWorld(id: string): Promise<void> {
   300→    const store = useGameStore.getState();
   301→    try {
   302→      const res = await fetch(`/api/worlds/${id}`);
   303→      if (!res.ok) throw new Error('load failed');
   304→      const { world } = (await res.json()) as { world: { id: string; name: string; gameMode: GameMode; seed: number; time: number; data: string } };
   305→      let save: SaveData | null = null;
   306→      try { save = JSON.parse(world.data) as SaveData; } catch { save = null; }
   307→      if (save) save.gameMode = world.gameMode;
   308→      this.currentWorldId = world.id;
   309→      this.currentWorldName = world.name;
   310→      store.setCurrentWorld(world.id, world.name);
   311→      this.setupWorld(world.seed, save, world.gameMode);
   312→      if (save) this.sky.time = save.time ?? this.sky.time;
   313→      await this.preloadSpawn();
   314→      this.enterPlaying();
   315→    } catch (err) {
   316→      console.error('loadWorld failed', err);
   317→      this.showToast('Failed to load world');
   318→      useGameStore.getState().setScreen('menu');
   319→    }
   320→  }

   322→  /** one-time migration: legacy localStorage save → DB world (singleton-guarded) */
   323→  private migratePromise: Promise<void> | null = null;
   324→  async migrateLocalSave(): Promise<void> {
   325→    if (this.migratePromise) return this.migratePromise;
   326→    this.migratePromise = (async (): Promise<void> => {
   327→      try {
   328→        const MIGRATED_FLAG = 'voxelcraft.migrated';
   329→        const raw = localStorage.getItem(SAVE_KEY);
   330→        if (!raw || localStorage.getItem(MIGRATED_FLAG)) return;
   331→        localStorage.setItem(MIGRATED_FLAG, '1'); // set first — prevents duplicate worlds on race/retry
   332→        const save = JSON.parse(raw) as SaveData & { gameMode?: GameMode };
   333→        const res = await fetch('/api/worlds', {
   334→          method: 'POST',
   335→          headers: { 'Content-Type': 'application/json' },
   336→          body: JSON.stringify({
   337→            name: 'Migrated World',
   338→            gameMode: save.gameMode ?? 'survival',
   339→            seed: save.seed,
   340→            time: save.time,
   341→            data: raw,
   342→          }),
   343→        });
   344→        if (!res.ok) return;
   345→        localStorage.removeItem(SAVE_KEY);
   346→        console.log('[voxelcraft] migrated local save → DB world');
   347→      } catch { /* non-fatal */ }
   348→    })();
   349→    return this.migratePromise;
   350→  }

   396→  private setupWorld(seed: number, save: SaveData | null, gameMode: GameMode = 'survival'): void {
   397→    // clear previous world if any
   398→    if (this.world) {
   399→      for (const key of Array.from(this.world.chunks.keys())) {
   400→        const [cx, cz] = key.split(',').map(Number);
   401→        this.world.unloadChunk(cx, cz);
   402→      }
   403→    }
   404→    this.drops?.clear();
   405→    this.mobs?.clear();
   406→    this.boats?.clear();
   407→    this.ridingBoat = null;
   408→    this.clearPrimedTnt();
   409→    this.world = new World(seed, save?.edits);
   410→    this.scene.add(this.world.group);
   411→    this.sky = new SkySystem(this.scene, seed);
   412→    this.sky.time = save?.time ?? DAY_LENGTH * 0.3;
   413→    this.weather?.dispose();
   414→    this.weather = new WeatherSystem(this.scene, seed, (wx, wz) => this.world.terrain.biomeAt(wx, wz));
   415→    this.weather.onStrike = (sx, sy, sz) => this.spawnLightningBolt(sx, sy, sz);
   416→    this.clearLightningBolts();
   417→    this.drops = new DropManager(this.scene, this.world, getAtlas().texture);
   418→    this.particles = new ParticleSystem(this.scene);
   419→    this.mobs = new MobManager(this.scene, this.world);
   420→    this.boats = new BoatManager(this.scene, this.world);
   421→    if (save?.mobs && Array.isArray(save.mobs)) this.mobs.restore(save.mobs);
   422→    this.xpOrbs = new XPOrbManager(this.scene, this.world);
   423→    this.blockEnts = new BlockEntityManager(this.world);
   424→    this.player = new Player(this.camera);
   425→    this.player.gameMode = save?.gameMode ?? gameMode;
   426→    this.player.flying = save?.flying ?? false;
   427→    this.spawnPoint = save?.spawn ?? null;
   428→    if (this.spawnPoint) this.mobs.spawnGuard = { x: this.spawnPoint.x, z: this.spawnPoint.z, r: 20 };
   429→    if (save?.blockEntities) this.blockEnts.load(save.blockEntities);
   430→
   431→    // spawn position
   432→    let sx = 8, sz = 8;
   433→    this.world.ensureChunk(0, 0);
   434→    let sy = this.world.surfaceY(sx, sz);
   435→    let attempts = 0;
   436→    while ((sy <= 40 || sy === 0) && attempts < 24) {
   437→      sx += 12; sz += 5; attempts++;
   438→      this.world.ensureChunk(Math.floor(sx / CHUNK_SIZE), Math.floor(sz / CHUNK_SIZE));
   439→      sy = this.world.surfaceY(sx, sz);
   440→    }
   441→    if (save?.player) {
   442→      this.player.entity.x = save.player.x;
   443→      this.player.entity.y = save.player.y;
   444→      this.player.entity.z = save.player.z;
   445→      this.player.yaw = save.player.yaw;
   446→      this.player.pitch = save.player.pitch;
   447→      this.player.health = save.player.health;
   448→      this.player.hotbar = save.player.hotbar;
   449→      this.player.main = save.player.main && save.player.main.length === 27
   450→        ? save.player.main
   451→        : Array.from({ length: 27 }, () => ({ blockId: 0, count: 0 }));
   452→      this.player.armor = save.player.armor && save.player.armor.length === 4
   453→        ? save.player.armor
   454→        : [null, null, null, null];
   455→      this.player.selected = save.player.selected;
   456→      this.player.fallStartY = save.player.y;
   457→      this.player.level = save.player.level ?? 0;
   458→      this.player.xp = save.player.xp ?? 0;
   459→      if (save.achievements) this.achievements.restore(save.achievements);
   460→    } else {
   461→      this.player.entity.x = sx + 0.5;
   462→      this.player.entity.y = sy + 1.2;
   463→      this.player.entity.z = sz + 0.5;
   464→      this.player.fallStartY = this.player.entity.y;
   465→    }
   466→    this.updateHandMesh(true);
   467→    this.syncHUD(true);
   468→    this.applySkyFog();
   469→  }

   471→  private async preloadSpawn(): Promise<void> {
   472→    const store = useGameStore.getState();
   473→    store.setScreen('loading');
   474→    const pcx = Math.floor(this.player.entity.x / CHUNK_SIZE);
   475→    const pcz = Math.floor(this.player.entity.z / CHUNK_SIZE);
   476→    const dataR = 2;
   477→    const jobs: [number, number][] = [];
   478→    for (let dx = -dataR; dx <= dataR; dx++)
   479→      for (let dz = -dataR; dz <= dataR; dz++)
   480→        jobs.push([pcx + dx, pcz + dz]);
   481→    jobs.sort((a, b) => (Math.abs(a[0] - pcx) + Math.abs(a[1] - pcz)) - (Math.abs(b[0] - pcx) + Math.abs(b[1] - pcz)));
   482→
   483→    for (let i = 0; i < jobs.length; i++) {
   484→      this.world.ensureChunk(jobs[i][0], jobs[i][1]);
   485→      if (i % 3 === 0) {
   486→        const p = i / jobs.length;
   487→        useGameStore.getState().setHud({ loadingProgress: p, loadingLabel: 'Generating world…' });
   488→        await this.nextFrame();
   489→      }
   490→    }
   491→    // mesh immediate area
   492→    for (let dx = -1; dx <= 1; dx++) {
   493→      for (let dz = -1; dz <= 1; dz++) {
   494→        const c = this.world.getChunk(pcx + dx, pcz + dz);
   495→        if (c) this.world.buildMesh(c);
   496→      }
   497→    }
   498→    useGameStore.getState().setHud({ loadingProgress: 1, loadingLabel: 'Building terrain…' });
   499→    await this.nextFrame();
   500→  }

   596→  private onKeyDown = (e: KeyboardEvent): void => {
   597→    // OS key auto-repeat: a HELD key fires keydown ~30x/sec. Held movement keys
   598→    // are already in this.keys (added on the first press), so re-processing
   599→    // repeats would only re-trigger ONE-SHOT actions — critically the creative
   600→    // flight double-tap, which then toggled flight on/off while Space was held.
   601→    if (e.repeat) {
   602→      if (e.code === 'Space' || e.code === 'F3' || e.code === 'F5') e.preventDefault();
   603→      return;
   604→    }
   605→    if (e.code === 'F3') { e.preventDefault(); useGameStore.getState().toggleDebug(); return; }
   606→    const st = useGameStore.getState();
   607→    if (e.code === 'F5') {
   608→      if (st.screen === 'playing') {
   609→        e.preventDefault(); // don't reload the page mid-game
   610→        this.cameraMode = ((this.cameraMode + 1) % 3) as 0 | 1 | 2;
   611→        useGameStore.getState().updateSettings({ cameraMode: this.cameraMode }); // persist F5 preference
   612→        audio.click();
   613→      }
   614→      return;
   615→    }
   616→    if (st.inv.open && st.screen === 'playing') {
   617→      if (e.code === 'KeyE' || e.code === 'Escape') { e.preventDefault(); this.closeInventory(); return; }
   618→      if (e.code.startsWith('Digit')) {
   619→        const n = parseInt(e.code.slice(5), 10);
   620→        if (n >= 1 && n <= 9 && this.invHover) this.invHotbarSwap(this.invHover.area, this.invHover.idx, n - 1);
   621→      }
   622→      return;
   623→    }
   624→    // trade panel open: Escape closes it
   625→    if (st.tradeOpen && st.screen === 'playing') {
   626→      if (e.code === 'Escape' || e.code === 'KeyE') { e.preventDefault(); this.closeTrade(); return; }
   627→      return;
   628→    }
   629→    // enchanting panel open: Escape closes it
   630→    if (st.enchantOpen && st.screen === 'playing') {
   631→      if (e.code === 'Escape' || e.code === 'KeyE') { e.preventDefault(); this.closeEnchant(); return; }
   632→      return;
   633→    }
   634→    if (st.screen !== 'playing') return;
   635→    this.keys.add(e.code);
   636→    if (e.code === 'F4') {
   637→      e.preventDefault();
   638→      const next = !st.creatorOpen;
   639→      useGameStore.getState().setCreatorOpen(next);
   640→      // unlock the cursor while the panel is open (clicks must reach the UI)
   641→      if (next && document.pointerLockElement === this.canvas) document.exitPointerLock();
   642→      else if (!next) this.requestLock();
   643→      return;
   644→    }
   645→    if (e.code === 'KeyE') { e.preventDefault(); this.openInventory(false); return; }
   646→    if (e.code.startsWith('Digit')) {
   647→      const n = parseInt(e.code.slice(5), 10);
   648→      if (n >= 1 && n <= 9) { this.player.selected = n - 1; this.syncHUD(); this.updateHandMesh(); }
   649→    }
   650→    if (e.code === 'KeyQ') this.dropSelected();
   651→    // creative: double-tap space toggles flight (fresh presses only — repeats
   652→    // are filtered above). Tap is consumed so a held key or a stray third tap
   653→    // can't chain-toggle flight back and forth.
   654→    if (e.code === 'Space' && this.player.isCreative) {
   655→      const now = performance.now();
   656→      if (now - this.lastSpaceTap < 280) {
   657→        this.player.flying = !this.player.flying;
   658→        if (this.player.flying) {
   659→          // small liftoff impulse: guarantees the player rises even if Space is
   660→          // released instantly (landing-cancel only fires while vy <= 0)
   661→          this.player.entity.vy = Math.max(this.player.entity.vy, 3.4);
   662→        }
   663→        audio.click();
   664→        this.syncHUD(true);
   665→        this.lastSpaceTap = 0;
   666→      } else {
   667→        this.lastSpaceTap = now;
   668→      }
   669→    }
   670→    if (e.code === 'Space') e.preventDefault();
   671→  };

   922→  /** apply bone meal to a block; returns true when consumed */
   923→  private fertilize(x: number, y: number, z: number, id: number): boolean {
   924→    const greens: [number, number][] = [[0.5, 0.95, 0.4], [0.65, 1, 0.5], [0.4, 0.85, 0.35]] as unknown as [number, number][];
   925→    void greens;
   926→    // grass: sprout tall grass + flowers around (MC-like scatter)
   927→    if (id === BLOCK.GRASS) {
   928→      let planted = 0;
   929→      for (let i = 0; i < 14 && planted < 5; i++) {
   930→        const bx = x + Math.floor(Math.random() * 5) - 2;
   931→        const bz = z + Math.floor(Math.random() * 5) - 2;
   932→        const by = y + 1;
   933→        if (this.world.getBlock(bx, by, bz) !== BLOCK.AIR) continue;
   934→        if (this.world.getBlock(bx, y, bz) !== BLOCK.GRASS) continue;
   935→        const r = Math.random();
   936→        const block = r < 0.72 ? BLOCK.TALL_GRASS : r < 0.86 ? BLOCK.FLOWER_RED : BLOCK.FLOWER_YELLOW;
   937→        this.world.setBlock(bx, by, bz, block);
   938→        planted++;
   939→      }
   940→      if (planted > 0) {
   941→        this.fertilizeFx(x, y + 1, z);
   942→        this.achievements.unlock('gardener');
   943→        return true;
   944→      }
   945→      return false;
   946→    }
   947→    // mycelium: sprout small mushrooms
   948→    if (id === BLOCK.MYCELIUM) {
   949→      let planted = 0;
   950→      for (let i = 0; i < 12 && planted < 4; i++) {
   951→        const bx = x + Math.floor(Math.random() * 5) - 2;
   952→        const bz = z + Math.floor(Math.random() * 5) - 2;
   953→        const by = y + 1;
   954→        if (this.world.getBlock(bx, by, bz) !== BLOCK.AIR) continue;
   955→        if (this.world.getBlock(bx, y, bz) !== BLOCK.MYCELIUM) continue;
   956→        this.world.setBlock(bx, by, bz, Math.random() < 0.5 ? BLOCK.MUSHROOM_RED : BLOCK.MUSHROOM_BROWN);
   957→        planted++;
   958→      }
   959→      if (planted > 0) {
   960→        this.fertilizeFx(x, y + 1, z);
   961→        this.achievements.unlock('gardener');
   962→        return true;
   963→      }
   964→      return false;
   965→    }
   966→    // small mushroom: chance to grow a giant one
   967→    if (id === BLOCK.MUSHROOM_RED || id === BLOCK.MUSHROOM_BROWN) {
   968→      if (Math.random() < 0.55 && this.growGiantMushroom(x, y, z, id)) {
   969→        this.fertilizeFx(x, y, z);
   970→        this.achievements.unlock('gardener');
   971→        return true;
   972→      }
   973→      this.showToast('The mushroom refuses to grow');
   974→      return false;
   975→    }
   976→    // wheat crop: advance 1-2 stages
   977→    if (isWheatCrop(id)) {
   978→      if (id === BLOCK.WHEAT_STAGE3) {
   979→        this.showToast('The wheat is already ripe');
   980→        return false;
   981→      }
   982→      const next = Math.min(BLOCK.WHEAT_STAGE3, nextWheatStage(id) + (Math.random() < 0.35 ? 1 : 0));
   983→      this.world.setBlock(x, y, z, next);
   984→      this.fertilizeFx(x, y + 0.5, z);
   985→      return true;
   986→    }
   987→    // sapling: instant tree
   988→    if (isSapling(id)) {
   989→      if (this.growSaplingTree(x, y, z, id)) {
   990→        this.fertilizeFx(x, y + 1, z);
   991→        this.achievements.unlock('gardener');
   992→        return true;
   993→      }
   994→      this.showToast('Not enough room to grow');
   995→      return false;
   996→    }
   997→    this.showToast('Bone meal has no effect here');
   998→    return false;
   999→  }

  1001→  /** grow a full tree from a sapling at (x,y,z); returns false when blocked */
  1002→  private growSaplingTree(x: number, y: number, z: number, saplingId: number): boolean {
  1003→    const below = this.world.getBlock(x, y - 1, z);
  1004→    if (below !== BLOCK.GRASS && below !== BLOCK.DIRT && below !== BLOCK.SNOW_GRASS && below !== BLOCK.FARMLAND) return false;
  1005→    const isSpruce = saplingId === BLOCK.SPRUCE_SAPLING;
  1006→    const height = isSpruce ? 6 + Math.floor(Math.random() * 3) : 4 + Math.floor(Math.random() * 3);
  1007→    // trunk column must be clear
  1008→    for (let dy = 1; dy <= height + 1; dy++) {
  1009→      if (y + dy >= WORLD_HEIGHT) return false;
  1010→      const cell = this.world.getBlock(x, y + dy, z);
  1011→      if (cell !== BLOCK.AIR && !isLiquid(cell)) return false;
  1012→    }
  1013→    const log = isSpruce ? BLOCK.SPRUCE_LOG : BLOCK.LOG;
  1014→    const leaf = isSpruce ? BLOCK.SPRUCE_LEAVES : BLOCK.LEAVES;
  1015→    const topY = y + height;
  1016→    // trunk (replaces the sapling)
  1017→    this.world.setBlock(x, y, z, log);
  1018→    for (let dy = 1; dy <= height; dy++) this.world.setBlock(x, y + dy, z, log);
  1019→    if (isSpruce) {
  1020→      // conical spruce: alternating ring radii + tip
  1021→      this.setIfAir(x, topY + 1, z, leaf);
  1022→      for (let ly = topY; ly >= y + 2; ly--) {
  1023→        const layer = topY - ly;
  1024→        const r = layer % 2 === 0 ? 1 : 2;
  1025→        for (let dx = -r; dx <= r; dx++)
  1026→          for (let dz = -r; dz <= r; dz++) {
  1027→            if (dx === 0 && dz === 0) continue;
  1028→            if (Math.abs(dx) === r && Math.abs(dz) === r && r === 2) continue;
  1029→            this.setIfAir(x + dx, ly, z + dz, leaf);
  1030→          }
  1031→      }
  1032→    } else {
  1033→      // oak canopy: two wide layers + two small layers (terrain.placeTree pattern)
  1034→      for (let dy = -2; dy <= 1; dy++) {
  1035→        const r = dy <= -1 ? 2 : 1;
  1036→        for (let dx = -r; dx <= r; dx++)
  1037→          for (let dz = -r; dz <= r; dz++) {
  1038→            if (dx === 0 && dz === 0 && dy <= 0) continue;
  1039→            this.setIfAir(x + dx, topY + dy, z + dz, leaf);
  1040→          }
  1041→      }
  1042→    }
  1043→    return true;
  1044→  }

  1057→  /** grow a giant mushroom from a small one (bonemeal); returns false when blocked */
  1058→  private growGiantMushroom(x: number, y: number, z: number, smallId: number): boolean {
  1059→    const below = this.world.getBlock(x, y - 1, z);
  1060→    if (below !== BLOCK.MYCELIUM && below !== BLOCK.DIRT && below !== BLOCK.GRASS) return false;
  1061→    const capId = smallId === BLOCK.MUSHROOM_BROWN ? BLOCK.MUSHROOM_BROWN_CAP : BLOCK.MUSHROOM_RED_CAP;
  1062→    const height = 4 + Math.floor(Math.random() * 3); // 4..6
  1063→    // clearance check
  1064→    for (let dy = 1; dy <= height + 1; dy++) {
  1065→      if (y + dy >= WORLD_HEIGHT) return false;
  1066→      const cell = this.world.getBlock(x, y + dy, z);
  1067→      if (cell !== BLOCK.AIR && !isLiquid(cell)) return false;
  1068→    }
  1069→    // stem (replaces the small mushroom at the base)
  1070→    this.world.setBlock(x, y, z, BLOCK.MUSHROOM_STEM);
  1071→    for (let dy = 1; dy <= height; dy++) this.world.setBlock(x, y + dy, z, BLOCK.MUSHROOM_STEM);
  1072→    const topY = y + height;
  1073→    if (smallId === BLOCK.MUSHROOM_RED) {
  1074→      for (let dx = -2; dx <= 2; dx++)
  1075→        for (let dz = -2; dz <= 2; dz++) {
  1076→          if (Math.abs(dx) === 2 && Math.abs(dz) === 2) continue;
  1077→          this.setIfAir(x + dx, topY, z + dz, capId);
  1078→        }
  1079→      for (let dx = -1; dx <= 1; dx++)
  1080→        for (let dz = -1; dz <= 1; dz++)
  1081→          this.setIfAir(x + dx, topY + 1, z + dz, capId);
  1082→    } else {
  1083→      for (let dx = -2; dx <= 2; dx++)
  1084→        for (let dz = -2; dz <= 2; dz++) {
  1085→          if (Math.abs(dx) === 2 && Math.abs(dz) === 2) continue;
  1086→          this.setIfAir(x + dx, topY - 1, z + dz, capId);
  1087→        }
  1088→      for (let dx = -1; dx <= 1; dx++)
  1089→        for (let dz = -1; dz <= 1; dz++)
  1090→          this.setIfAir(x + dx, topY, z + dz, capId);
  1091→      this.setIfAir(x, topY + 1, z, capId);
  1092→    }
  1093→    return true;
  1094→  }

  1203→  private updateFishing(dt: number): void {
  1204→    const f = this.fishing;
  1205→    if (!f || !this.bobberMesh || !this.fishLine) return;
  1206→    const held = this.player.hotbar[this.player.selected];
  1207→    if (!held || held.count <= 0 || !isRodItem(held.blockId) || this.player.dead) { this.endFishing(); return; }
  1208→    const dist = Math.hypot(f.x - this.player.x, f.z - this.player.z);
  1209→    if (dist > 26) { this.endFishing(); return; }
  1210→
  1211→    if (f.state === 'fly') {
  1212→      // projectile arc toward the aim point
  1213→      f.vy -= 18 * dt;
  1214→      f.x += f.vx * dt; f.y += f.vy * dt; f.z += f.vz * dt;
  1215→      const bx = Math.floor(f.x), by = Math.floor(f.y), bz = Math.floor(f.z);
  1216→      const bid = this.world.getBlock(bx, by, bz);
  1217→      if (isWaterId(bid)) {
  1218→        f.state = 'float';
  1219→        f.y = by + 0.9;
  1220→        f.vx = 0; f.vy = 0; f.vz = 0;
  1221→        const lure = lureFactor(held.ench);
  1222→        const rain = this.weather.raining ? 0.75 : 1; // MC: rain bites faster
  1223→        f.nextBite = (5 + Math.random() * 14) * lure * rain;
  1224→        audio.rodSplash(dist);
  1225→        this.particles.burstLand(f.x, by + 1, f.z, [0.55, 0.7, 0.95], 8);
  1226→      } else if (bid !== BLOCK.AIR) {
  1227→        // snagged on land: the line snaps back (no cost)
  1228→        this.endFishing();
  1229→        return;
  1230→      }
  1231→    } else {
  1232→      // floating: bob gently; bite window after the wait
  1233→      if (f.state === 'bite') {
  1234→        f.biteT -= dt;
  1235→        if (f.biteT <= 0) {
  1236→          f.state = 'float';
  1237→          f.nextBite = 4 + Math.random() * 10; // missed — fish may return
  1238→        }
  1239→      } else {
  1240→        f.nextBite -= dt;
  1241→        if (f.nextBite <= 0) {
  1242→          f.state = 'bite';
  1243→          f.biteT = 1.4;
  1244→          audio.fishBite();
  1245→          this.particles.burstLand(f.x, f.y, f.z, [0.55, 0.7, 0.95], 5);
  1246→        }
  1247→      }
  1248→    }
  1249→
  1250→    // visuals: bobber (dips while biting) + line from the rod hand
  1251→    this.bobberMesh.position.set(f.x, f.y + (f.state === 'bite' ? -0.12 : Math.sin(performance.now() / 300) * 0.02), f.z);
  1252→    this.bobberMesh.rotation.y += dt * 2;
  1253→    const p = this.player;
  1254→    const hand = new THREE.Vector3(p.x, p.eyeY() - 0.32, p.z);
  1255→    hand.x += -Math.cos(p.yaw) * 0.34 - Math.sin(p.yaw) * 0.24;
  1256→    hand.z += Math.sin(p.yaw) * 0.34 - Math.cos(p.yaw) * 0.24;
  1257→    const posAttr = this.fishLine.geometry.getAttribute('position') as THREE.BufferAttribute;
  1258→    posAttr.setXYZ(0, hand.x, hand.y, hand.z);
  1259→    posAttr.setXYZ(1, this.bobberMesh.position.x, this.bobberMesh.position.y, this.bobberMesh.position.z);
  1260→    posAttr.needsUpdate = true;
  1261→  }

  1263→  /** fish on! loot roll (fish / junk / treasure, Luck of the Sea shifts treasure) */

  1269→    const roll = Math.random();
  1270→    const treasureShare = 0.10 + luck;
  1271→    const junkShare = 0.18;
  1272→    let loot: { id: number; count: number };
  1273→    if (roll < treasureShare) {
  1274→      const t = Math.random();

  1279→    } else {
  1280→      loot = Math.random() < 0.65 ? { id: ITEM.RAW_COD, count: 1 } : { id: ITEM.RAW_SALMON, count: 1 };
  1281→    }
  1282→    const left = this.addToInventory(loot.id, loot.count);
  1283→    if (left > 0) this.drops.spawn(loot.id, this.player.x, this.player.y + 1, this.player.z, left);

  1287→    if (loot.id === ITEM.RAW_COD || loot.id === ITEM.RAW_SALMON) this.addXP(1 + Math.floor(Math.random() * 3));
  1288→    // rod durability (1 per catch, MC-style; Unbreaking works)
  1289→    if (held && !this.player.isCreative) {
  1290→      const rdef = getRodDef(held.blockId);
  1291→      if (rdef && !unbreakingKeep(held.ench?.unbreaking ?? 0)) {
  1292→        held.dur = (held.dur ?? rdef.dur) - 1;
  1293→        if (held.dur <= 0) {
  1294→          this.player.hotbar[this.player.selected] = { blockId: 0, count: 0 };
  1295→          audio.breakBlock('glass');
  1296→          this.showToast('Your fishing rod broke!');
  1297→          this.updateHandMesh(true);
  1298→        }
  1299→      }
  1300→    }
  1301→    this.endFishing();
  1302→    this.syncHUD();
  1303→    this.syncInventory();
  1304→    this.updateHandMesh();
  1305→  }

  1600→      for (const s of slots) {

  1657→  private mineTick(dt: number): void {

  1660→      ? getToolDef(heldSlot.blockId)

  1662→
  1663→    // mob attack takes priority over mining (boats too)
  1664→    if (this.mining && this.attackCooldown <= 0) {
  1665→      const eye = new THREE.Vector3(this.player.x, this.player.eyeY(), this.player.z);

  1669→      const boatHit = this.ridingBoat ? null : this.boats.raycast(eye.x, eye.y, eye.z, dir.x, dir.y, dir.z, 3.4);
  1670→      if (boatHit && (!hit || boatHit.dist < hit.dist)) {

  1673→        this.boats.remove(boatHit.boat);

  1677→        this.crackMesh.visible = false;
  1678→        return;
  1679→      }
  1680→      if (hit) {
  1681→        this.attackCooldown = 0.42;
  1682→        this.startSwing();
  1683→        const kx = hit.mob.x - this.player.x;
  1684→        const kz = hit.mob.z - this.player.z;
  1685→        // Sharpness: +1 dmg per level (sword/axe)
  1686→        const melee = (heldTool ? heldTool.dmg : 2) + (heldTool ? sharpnessBonus(heldSlot?.ench) : 0);
  1687→        const killed = this.mobs.hurtMob(hit.mob, melee, kx, kz, this.mobCb ?? undefined);
  1688→        if (killed && hit.mob.def.hostile) this.achievements.unlock('monsterHunter');
  1689→        this.particles.hurt(hit.mob.x, hit.mob.y, hit.mob.z);
  1690→        this.damageTool(heldTool && heldTool.type !== 'sword' ? 2 : 1);
  1691→        this.crackMesh.visible = false;
  1692→        return;
  1693→      }
  1694→    }
  1695→    if (!this.mining || !this.target) { this.crackMesh.visible = false; return; }
  1696→    const t = this.target;
  1697→    // ── creative: instant break, no drops, no XP, no tool wear ──
  1698→    if (this.player.isCreative) {
  1699→      const cdef = getBlockDef(t.id);
  1700→      if (!cdef) return;
  1701→      const col = tileAvgColor(Array.isArray(cdef.tiles) ? cdef.tiles[2] : cdef.tiles);
  1702→      this.particles.burstBlockBreak(t.x, t.y, t.z, col);
  1703→      audio.breakBlock((cdef.sound ?? 'stone') as MaterialSound);
  1704→      if (containerOf(t.id)) this.blockEnts.destroy(t.x, t.y, t.z);
  1705→      this.world.setBlock(t.x, t.y, t.z, BLOCK.AIR);
  1706→      // pop unsupported blocks above (torch, flowers, bed) + plant stacks
  1707→      let py = t.y + 1;
  1708→      let guard = 0;
  1709→      while (py < WORLD_HEIGHT && guard++ < 96) {
  1710→        const aboveId = this.world.getBlock(t.x, py, t.z);
  1711→        const aboveDef = getBlockDef(aboveId);
  1712→        const isStack = aboveId === BLOCK.SUGARCANE || aboveId === BLOCK.CACTUS;
  1713→        if (!aboveDef?.needsGround && !isStack) break;
  1714→        if (containerOf(aboveId)) this.blockEnts.destroy(t.x, py, t.z);
  1715→        this.world.setBlock(t.x, py, t.z, BLOCK.AIR);
  1716→        py++;
  1717→      }
  1718→      this.startSwing();
  1719→      this.crackMesh.visible = false;
  1720→      return;
  1721→    }
  1722→    // target changed? reset progress
  1723→    if (!this.mineTarget || this.mineTarget.x !== t.x || this.mineTarget.y !== t.y || this.mineTarget.z !== t.z) {
  1724→      this.mineTarget = t;
  1725→      this.mineProgress = 0;
  1726→    }
  1727→    const def = getBlockDef(t.id);
  1728→    if (!def) { this.crackMesh.visible = false; return; }
  1729→    const bi = breakInfo(def, heldTool);
  1730→    // Efficiency: faster mining, but only when the held tool is the block's
  1731→    // matching class (MC behavior — a sword's Efficiency never digs faster)
  1732→    const effF = heldTool && heldSlot?.ench && heldTool.type === def.tool ? efficiencyFactor(heldSlot.ench) : 1;
  1733→    const time = this.instantBreak ? 0.04 : bi.time * effF; // cheat: instant break
  1734→    const harvest = bi.harvest;
  1735→    if (!Number.isFinite(time)) { this.crackMesh.visible = false; return; }
  1736→    this.mineProgress += dt / time;
  1737→
  1738→    // dig sound + swing loop
  1739→    this.stepTimer += dt;
  1740→    if (this.stepTimer > 0.24) {
  1741→      this.stepTimer = 0;
  1742→      audio.dig((def.sound ?? 'stone') as MaterialSound);
  1743→      this.startSwing();
  1744→    }
  1745→
  1746→    if (this.mineProgress >= 1) {
  1747→      // break!
  1748→      const col = tileAvgColor(Array.isArray(def.tiles) ? def.tiles[2] : def.tiles);
  1749→      this.particles.burstBlockBreak(t.x, t.y, t.z, col);
  1750→      audio.breakBlock((def.sound ?? 'stone') as MaterialSound);
  1751→      const dropId = def.drop === undefined ? t.id : def.drop;
  1752→      // gravel has a 12% chance to drop flint (MC-style, used for arrows)
  1753→      let actualDrop = harvest && t.id === BLOCK.GRAVEL && Math.random() < 0.12 ? ITEM.FLINT : dropId;
  1754→      // leaves drop saplings (~8%, MC-style) — the tree regrowth loop
  1755→      if (harvest && t.id === BLOCK.LEAVES && Math.random() < 0.08) actualDrop = BLOCK.OAK_SAPLING;
  1756→      if (harvest && t.id === BLOCK.SPRUCE_LEAVES && Math.random() < 0.08) actualDrop = BLOCK.SPRUCE_SAPLING;
  1757→      // tall grass drops wheat seeds (~20%, MC-style foraging)
