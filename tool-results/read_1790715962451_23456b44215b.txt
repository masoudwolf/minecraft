   108→// ─── Model builders (vanilla textures + MC box-UV mapping) ───────────────────
   109→/** Build a box mesh textured from a vanilla skin part via standard MC box UVs.
   110→ *  Cutout (alphaTest) so vanilla transparent texels (ribcage gaps, fleece face)
   111→ *  render as holes instead of black; DoubleSide so thin parts show both sides. */
   112→function boxPart(p: MobSkinPart, w: number, h: number, d: number, tag: string, opts?: BoxUVOptions): THREE.Mesh {
   113→  const geo = new THREE.BoxGeometry(w, h, d);
   114→  boxUV(geo, p.lay, p.texW, p.texH, opts);
   115→  const mesh = new THREE.Mesh(geo, new THREE.MeshLambertMaterial({ map: p.tex, alphaTest: 0.35, side: THREE.DoubleSide }));
   116→  mesh.userData.part = tag;
   117→  return mesh;
   118→}

   120→/** Collect a mesh's material(s) into the parts list (per-instance tint/light). */
   121→function collectMats(list: THREE.MeshLambertMaterial[], mesh: THREE.Mesh): void {
   122→  const m = mesh.material as THREE.MeshLambertMaterial | THREE.MeshLambertMaterial[];
   123→  if (Array.isArray(m)) list.push(...m);
   124→  else list.push(m);
   125→}

   127→/**
   128→ * Leg with a proper HIP PIVOT (Minecraft-style limb rigging): the pivot Group
   129→ * sits at the hip joint (top of the leg) and the leg box hangs BELOW it, so
   130→ * animating pivot.rotation.x swings the leg from the top — like MC. `furPart`
   131→ * optionally adds an inflated fleece box (sheep) that rides the swing; like
   132→ * vanilla, the fleece leg wraps only the UPPER HALF of the leg.
   133→ */
   134→function legPivot(p: MobSkinPart, w: number, h: number, d: number, hipX: number, hipY: number, hipZ: number, side = 0, furPart?: { part: MobSkinPart; inflate: number }): THREE.Group {
   135→  const pivot = new THREE.Group();
   136→  pivot.position.set(hipX, hipY, hipZ);
   137→  pivot.userData.side = side;
   138→  const leg = boxPart(p, w, h, d, 'limb');
   139→  leg.position.y = -h / 2; // hang from the hip
   140→  pivot.add(leg);
   141→  if (furPart) {
   142→    const f = furPart.inflate;
   143→    const furH = h * 0.5 + f * 2;
   144→    const fur = boxPart(furPart.part, w + f * 2, furH, d + f * 2, 'fur');
   145→    leg.add(fur);
   146→    fur.position.y = h / 2 - furH / 2 + f * 0.5; // top-aligned wrap
   147→  }
   148→  return pivot;
   149→}

   151→const solidTexCache = new Map<string, THREE.Texture>();
   152→/** Tiny solid-color texture — lets plain-colored parts (bow, stick arms) ride
   153→ *  the shared map+white-color material pipeline (world-light tinting safe). */
   154→function solidTex(hex: string): THREE.Texture {
   155→  const hit = solidTexCache.get(hex);
   156→  if (hit) return hit;
   157→  const c = document.createElement('canvas');
   158→  c.width = 2; c.height = 2;
   159→  const ctx = c.getContext('2d')!;
   160→  ctx.fillStyle = hex;
   161→  ctx.fillRect(0, 0, 2, 2);
   162→  const t = new THREE.CanvasTexture(c);
   163→  t.magFilter = THREE.NearestFilter;
   164→  t.minFilter = THREE.NearestFilter;
   165→  t.generateMipmaps = false;
   166→  t.colorSpace = THREE.SRGBColorSpace;
   167→  solidTexCache.set(hex, t);
   168→  return t;
   169→}
   170→
   171→function cutoutMat(hex: string): THREE.MeshLambertMaterial {
   172→  return new THREE.MeshLambertMaterial({ map: solidTex(hex), alphaTest: 0.35, side: THREE.DoubleSide });
   173→}

   175→/** Simple boxy Minecraft-style bow: curved limbs + string. Shoots toward the
   176→ *  belly side (-X of the group); attach with rotation.y = π/2 to shoot +Z. */
   177→function buildBow(): THREE.Group {
   178→  const g = new THREE.Group();
   179→  const wood = cutoutMat('#6b4d2a');
   180→  const woodLight = cutoutMat('#8a683c');
   181→  const str = cutoutMat('#e8e8e8');
   182→  const grip = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.2, 0.06), wood);
   183→  g.add(grip);
   184→  const upper = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.24, 0.05), woodLight);
   185→  upper.position.set(0.05, 0.2, 0);
   186→  upper.rotation.z = 0.5;
   187→  g.add(upper);
   188→  const lower = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.24, 0.05), woodLight);
   189→  lower.position.set(0.05, -0.2, 0);
   190→  lower.rotation.z = -0.5;
   191→  g.add(lower);
   192→  const string = new THREE.Mesh(new THREE.BoxGeometry(0.015, 0.5, 0.015), str);
   193→  string.position.set(0.1, 0, 0);
   194→  g.add(string);
   195→  return g;
   196→}

   198→function makeShadow(radius: number, scene: THREE.Scene): THREE.Mesh {
   199→  const geo = new THREE.CircleGeometry(radius, 12);
   200→  const mat = new THREE.MeshBasicMaterial({ color: 0x000000, transparent: true, opacity: 0.28, depthWrite: false });
   201→  const mesh = new THREE.Mesh(geo, mat);
   202→  mesh.rotation.x = -Math.PI / 2;
   203→  mesh.renderOrder = 1;
   204→  scene.add(mesh);
   205→  return mesh;
   206→}

   208→function quadruped(skins: MobSkins, opts: {
   209→  headY: number; headZ: number; headW: number; headH: number; headD: number;
   210→  bodyY: number; bodyZ: number;
   211→  legW: number; legH: number; legX: number; legZHind: number; legZFront: number;
   212→  shadowR: number;
   213→  fur?: { head: MobSkinPart; body: MobSkinPart; limb: MobSkinPart; inflate: number };
   214→  udder?: MobSkinPart;
   215→  horns?: MobSkinPart;
   216→}): MobParts {
   217→  const px = 1 / 16;
   218→  const group = new THREE.Group();
   219→  const mats: THREE.MeshLambertMaterial[] = [];
   220→  // VANILLA body construction (CowModel/PigModel/SheepModel): the body box is
   221→  // built VERTICAL (lay.w × lay.h × lay.d) with its UV cross laid out for that
   222→  // orientation, then rotated 90° about X. Reproducing the rotation puts every
   223→  // texel where vanilla puts it: the front region becomes the BELLY (udder
   224→  // lands rear-under), the back region becomes the spine, top/bottom regions
   225→  // become the chest/rump caps. Building it horizontal was the bug that
   226→  // rendered the cow's udder on its chest under the head.
   227→  const bl = skins.body.lay;
   228→  const body = boxPart(skins.body, bl.w * px, bl.h * px, bl.d * px, 'body');
   229→  body.rotation.x = Math.PI / 2;
   230→  body.position.set(0, opts.bodyY, opts.bodyZ);
   231→  group.add(body);
   232→  if (opts.fur) {
   233→    // fleece body: vanilla SheepFurModel inflate 1.75/side (+3.5px every axis).
   234→    // NO extra rotation on the child — the parent `body` mesh is ALREADY
   235→    // rotated π/2 about X, and the fleece is the SAME box (same pivot, same
   236→    // center) only inflated. Adding another π/2 here made the total rotation π,
   237→    // which pointed the fleece's 19.5px-long axis VERTICALLY (wool sticking up
   238→    // off the body — user-reported bug). Child at (0,0,0) rides the parent's
   239→    // rotation and centers exactly on the skin body, vanilla-style.
   240→    const fl = opts.fur.body.lay;
   241→    const furBody = boxPart(opts.fur.body, (fl.w + 3.5) * px, (fl.h + 3.5) * px, (fl.d + 3.5) * px, 'fur');
   242→    body.add(furBody);
   243→    collectMats(mats, furBody);
   244→  }
   245→  if (opts.udder) {
   246→    // vanilla CowModel udder: a 4×6×1 box ON the body part's local front face;
   247→    // after the 90° rotation it hangs under the REAR of the belly (vanilla)
   248→    const ul = opts.udder.lay;
   249→    const udder = boxPart(opts.udder, ul.w * px, ul.h * px, ul.d * px, 'udder');
   250→    udder.position.set(0, -6 * px, 5.5 * px);
   251→    body.add(udder);
   252→    collectMats(mats, udder);
   253→  }
   254→  const head = boxPart(skins.head, opts.headW, opts.headH, opts.headD, 'head');
   255→  head.position.set(0, opts.headY, opts.headZ);
   256→  group.add(head);
   257→  if (opts.fur) {
   258→    // fleece head: 6×6×6 (+0.6px/side) covering the BACK 6px of the 8px-deep
   259→    // skin head — the skin's front 2px face plate PROTRUDES past the wool
   260→    // (vanilla SheepFurModel geometry), so the eyes/muzzle show without any
   261→    // transparency tricks
   262→    const fl = opts.fur.head.lay;
   263→    const furHead = boxPart(opts.fur.head, (fl.w + 1.2) * px, (fl.h + 1.2) * px, (fl.d + 1.2) * px, 'fur');
   264→    furHead.position.set(0, 0, -1 * px);
   265→    head.add(furHead);
   266→    collectMats(mats, furHead);
   267→  }
   268→  if (opts.horns) {
   269→    // vanilla cow horns: 1×3×1 boxes at the head's upper corners
   270→    for (const sx of [-1, 1]) {
   271→      const horn = boxPart(opts.horns, 1 * px, 3 * px, 1 * px, 'horn');
   272→      horn.position.set(sx * 4.5 * px, 3.5 * px, 0.5 * px);
   273→      head.add(horn);
   274→      collectMats(mats, horn);
   275→    }
   276→  }
   277→  const legs: THREE.Mesh[] = [];
   278→  for (const [sx, sz] of [[-1, -1], [1, -1], [-1, 1], [1, 1]]) {
   279→    // vanilla hip: the pivot IS the vanilla leg pivot (body-bottom height) and
   280→    // the leg hangs exactly to the ground — no gap, swings from the top
   281→    const z = sz === -1 ? opts.legZFront : opts.legZHind;
   282→    const pivot = legPivot(skins.limb, opts.legW, opts.legH, opts.legW, sx * opts.legX, opts.legH, z, sx,
   283→      opts.fur ? { part: opts.fur.limb, inflate: opts.fur.inflate } : undefined);
   284→    group.add(pivot);
   285→    legs.push(pivot as unknown as THREE.Mesh);
   286→  }
   287→  for (const mesh of [body, head]) collectMats(mats, mesh);
   288→  for (const pv of legs) for (const child of (pv as unknown as THREE.Group).children) collectMats(mats, child as THREE.Mesh);
   289→  return { group, head, legs, arms: [], materials: mats, shadow: null as unknown as THREE.Mesh };
   290→}

   292→function humanoid(skins: MobSkins, thin = false): MobParts {
   293→  const lw = thin ? 0.16 : 0.25;
   294→  const aw = thin ? 0.14 : 0.22;
   295→  const group = new THREE.Group();
   296→  const mats: THREE.MeshLambertMaterial[] = [];
   297→  const body = boxPart(skins.body, 0.5, 0.72, 0.26, 'body');
   298→  body.position.y = 1.1;
   299→  group.add(body);
   300→  const head = boxPart(skins.head, 0.5, 0.5, 0.5, 'head');
   301→  head.position.y = 1.72;
   302→  group.add(head);
   303→  const legs: THREE.Mesh[] = [];
   304→  for (const sx of [-1, 1]) {
   305→    const pivot = legPivot(skins.limb, lw, 0.74, lw, sx * 0.125, 0.74, 0, sx);
   306→    group.add(pivot);
   307→    legs.push(pivot as unknown as THREE.Mesh);
   308→  }
   309→  const arms: THREE.Mesh[] = [];
   310→  const armPart = skins.limb2 ?? skins.limb;
   311→  for (const sx of [-1, 1]) {
   312→    const pivot = new THREE.Group();
   313→    pivot.position.set(sx * (0.25 + aw / 2), 1.42, 0);
   314→    const arm = boxPart(armPart, aw, 0.7, aw, 'limb');
   315→    arm.position.y = -0.32;
   316→    pivot.add(arm);
   317→    group.add(pivot);
   318→    arms.push(arm);
   319→    (arm as unknown as { limbPivot: THREE.Group }).limbPivot = pivot;
   320→  }
   321→  for (const mesh of [body, head]) collectMats(mats, mesh);
   322→  for (const pv of legs) for (const child of (pv as unknown as THREE.Group).children) collectMats(mats, child as THREE.Mesh);
   323→  for (const arm of arms) collectMats(mats, arm);
   324→  return { group, head, legs, arms, materials: mats, shadow: null as unknown as THREE.Mesh };
   325→}

   329→export function buildMobModel(type: string, variant?: string): MobParts | null {
   330→  const base = (type === 'mooshroom_brown' ? 'mooshroom' : type) as MobType;
   331→  const def = MOB_DEFS[base];
   332→  if (!def) return null;
   333→  const skins = getMobSkins(variant ? `${type}:${variant}` : type);
   334→  const parts = def.builder(skins);
   335→  return parts;
   336→}

   783→        pivot.rotation.z = -sx * 1.0;
   784→        const arm = boxPart(s.armStick!, 0.75, 0.125, 0.125, 'arm');
   785→        arm.position.set(sx * 0.3125, -0.0625, 0);
   786→        pivot.add(arm);
   787→        group.add(pivot);
   788→        arms.push(arm);
   789→        (arm as unknown as { limbPivot: THREE.Group }).limbPivot = pivot;
   790→      }
   791→      for (const mesh of [lower, body, head]) collectMats(mats, mesh);
   792→      for (const arm of arms) collectMats(mats, arm);
   793→      return { group, head, legs: [], arms, materials: mats, shadow: null as unknown as THREE.Mesh };
   794→    },
   795→  },
   796→};

   808→function rayAABB(ox: number, oy: number, oz: number, dx: number, dy: number, dz: number, e: AABBEntity, maxDist: number): number | null {
   809→  const half = e.width / 2;
   810→  const minX = e.x - half, maxX = e.x + half;
   811→  const minY = e.y, maxY = e.y + e.height;
   812→  const minZ = e.z - half, maxZ = e.z + half;
   813→  let tmin = 0;
   814→  let tmax = maxDist;
   815→  for (const [o, d, min, max] of [[ox, dx, minX, maxX], [oy, dy, minY, maxY], [oz, dz, minZ, maxZ]] as const) {
   816→    if (Math.abs(d) < 1e-9) {
   817→      if (o < min || o > max) return null;
   818→    } else {
   819→      let t1 = (min - o) / d;
   820→      let t2 = (max - o) / d;
   821→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp; }
   822→      tmin = Math.max(tmin, t1);
   823→      tmax = Math.min(tmax, t2);
   824→      if (tmin > tmax) return null;
   825→    }
   826→  }
   827→  return tmin;
   828→}

   990→  private trySpawnMob(playerX: number, playerY: number, playerZ: number, sunLevel: number, passiveCount: number, hostileCount: number): void {
   991→    const angle = Math.random() * Math.PI * 2;
   992→    const dist = 16 + Math.random() * 26;
   993→    const x = Math.floor(playerX + Math.cos(angle) * dist) + 0.5;
   994→    const z = Math.floor(playerZ + Math.sin(angle) * dist) + 0.5;
   995→    const cx = Math.floor(x);
   996→    const cz = Math.floor(z);
   997→    // find surface
   998→    let sy = -1;
   999→    for (let y = 95; y > 2; y--) {
  1000→      const id = this.world.getBlock(cx, y, cz);
  1001→      if (id !== 0 && !isWaterId(id)) {
  1002→        sy = y;
  1003→        break;
  1004→      }
  1005→    }
  1006→    if (sy < 3) return;
  1007→    const above = this.world.getBlock(cx, sy + 1, cz);
  1008→    const above2 = this.world.getBlock(cx, sy + 2, cz);
  1009→    if (above !== 0 || above2 !== 0) return;
  1010→    const groundId = this.world.getBlock(cx, sy, cz);
  1011→    if (isWaterId(groundId)) return;
  1012→
  1013→    const l = this.world.getLight(cx, sy + 1, cz);
  1014→    if (l < 0) return;
  1015→    const skyL = l >> 4;
  1016→    const blkL = l & 15;
  1017→    const effLight = Math.max(blkL, skyL * sunLevel);
  1018→
  1019→    // villagers live on village grounds (planks / cobblestone); mooshrooms on mycelium
  1020→    if (groundId === BLOCK.PLANKS || groundId === BLOCK.COBBLESTONE) {
  1021→      // swamp planks/cobble = witch hut grounds: keep a witch around (day or night)
  1022→      if (this.world.biomeAt?.(cx, cz) === 'swamp') {
  1023→        const witchCount = this.mobs.filter((m2) => m2.type === 'witch').length;
  1024→        if (witchCount < 1 && Math.random() < 0.3) {
  1025→          this.spawn('witch', x, sy + 1, z);
  1026→        }
  1027→        return;
  1028→      }
  1029→      // iron golem: at most one patrolling each village chunk roll
  1030→      const golemCount = this.mobs.filter((m2) => m2.type === 'golem').length;
  1031→      if (golemCount < 1 && Math.random() < 0.22 && skyL >= 9 && sunLevel > 0.55) {
  1032→        this.spawn('golem', x, sy + 1, z);
  1033→        return;
  1034→      }
  1035→      const villagerCount = this.mobs.filter((m) => m.type === 'villager').length;
  1036→      if (villagerCount < 5 && skyL >= 9 && sunLevel > 0.55) {
  1037→        const herd = 1 + Math.floor(Math.random() * 2);
  1038→        for (let i = 0; i < herd; i++) this.spawn('villager', x + (Math.random() - 0.5) * 2, sy + 1, z + (Math.random() - 0.5) * 2);
  1039→      }
  1040→      return;
  1041→    }
  1042→    if (groundId === BLOCK.MYCELIUM) {
  1043→      const passiveNow = passiveCount + this.mobs.filter((m) => m.type === 'mooshroom').length;
  1044→      if (passiveNow < 12 && skyL >= 9 && sunLevel > 0.55) {
  1045→        const herd = 1 + Math.floor(Math.random() * 2);
  1046→        for (let i = 0; i < herd; i++) this.spawn('mooshroom', x + (Math.random() - 0.5) * 3, sy + 1, z + (Math.random() - 0.5) * 3);
  1047→      }
  1048→      return;
  1049→    }
  1050→
  1051→    const wantHostile = hostileCount < 12 && (effLight < 6);
  1052→    const wantPassive = passiveCount < 10 && skyL >= 9 && sunLevel > 0.55 && (groundId === BLOCK.GRASS || groundId === BLOCK.SNOW_GRASS);
  1053→
  1054→    if (wantHostile && (!wantPassive || Math.random() < 0.65)) {
  1055→      const roll = Math.random();
  1056→      const isSwamp = this.world.biomeAt?.(cx, cz) === 'swamp';
  1057→      const type: MobType = roll < 0.32 ? 'zombie' : roll < 0.58 ? 'skeleton' : roll < 0.74 ? 'creeper' : roll < 0.86 ? 'spider' : roll < 0.95 && isSwamp ? 'witch' : 'enderman';
  1058→      this.spawn(type, x, sy + 1, z);
  1059→    } else if (wantPassive) {
  1060→      const roll = Math.random();
  1061→      // snowy ground → snow golems (pumpkin-headed, occasionally sheared)
  1062→      if (groundId === BLOCK.SNOW_GRASS && roll < 0.3) {
  1063→        this.spawn('snowgolem', x, sy + 1, z, Math.random() < 0.85 ? 'pumpkin' : 'plain');
  1064→        return;
  1065→      }
  1066→      const type: MobType = roll < 0.3 ? 'pig' : roll < 0.55 ? 'sheep' : roll < 0.8 ? 'cow' : 'chicken';
  1067→      // spawn small herd for passive
  1068→      const herd = 1 + Math.floor(Math.random() * 3);
  1069→      for (let i = 0; i < herd; i++) {
  1070→        this.spawn(type, x + (Math.random() - 0.5) * 3, sy + 1, z + (Math.random() - 0.5) * 3, type === 'sheep' ? pickSheepVariant() : '');
  1071→      }
  1072→    }
  1073→  }

  1075→  hurtMob(mob: Mob, dmg: number, kx: number, kz: number, cb?: MobCallbacks): boolean {
  1076→    if (mob.dead) return false;
  1077→    mob.health -= dmg;
  1078→    mob.hurtT = 0.4;
  1079→    const len = Math.hypot(kx, kz) || 1;
  1080→    mob.vx += (kx / len) * 7;
  1081→    mob.vz += (kz / len) * 7;
  1082→    mob.vy = Math.max(mob.vy, 4.4);
  1083→    audio.mobHurt(mob.def.sound);
  1084→    if (mob.def.hostile) {
  1085→      mob.state = 'chase';
  1086→      if (mob.type === 'spider' || mob.type === 'enderman') mob.provoked = true;
  1087→    } else if (mob.type === 'golem') {
  1088→      // golems never flee — they stoically keep patrolling
  1089→      mob.state = 'idle';
  1090→    } else {
  1091→      mob.state = 'flee';
  1092→      mob.stateTimer = 4;
  1093→      mob.wanderX = mob.x + (mob.x - kx) * 10;
  1094→      mob.wanderZ = mob.z + (mob.z - kz) * 10;
  1095→    }
  1096→    // endermen often warp away when hurt (MC behavior)
  1097→    const ecb = cb ?? this.lastCb;
  1098→    if (mob.type === 'enderman' && !mob.dead && ecb && Math.random() < 0.55) {
  1099→      this.teleportNear(mob, mob.x, mob.z, 7, 15, ecb);
  1100→    }
  1101→    if (mob.health <= 0) {
  1102→      mob.dead = true;
  1103→      mob.deathT = 0.45;
  1104→      if (mob.type === 'creeper' && mob.fuse >= 0) mob.fuse = -1;
  1105→      return true;
  1106→    }
  1107→    return false;
  1108→  }

  1110→  /** enderman teleport: find a valid surface spot within [minR,maxR] of (cx,cz) */
  1111→  private teleportNear(m: Mob, cx: number, cz: number, minR: number, maxR: number, cb: MobCallbacks): boolean {
  1112→    const ox = m.x, oy = m.y, oz = m.z;
  1113→    for (let i = 0; i < 14; i++) {
  1114→      const a = Math.random() * Math.PI * 2;
  1115→      const r = minR + Math.random() * (maxR - minR);
  1116→      const x = Math.floor(cx + Math.cos(a) * r);
  1117→      const z = Math.floor(cz + Math.sin(a) * r);
  1118→      let sy = -1;
  1119→      for (let y = 95; y > 2; y--) {
  1120→        const id = this.world.getBlock(x, y, z);
  1121→        if (id !== 0 && !isWaterId(id)) { sy = y; break; }
  1122→      }
  1123→      if (sy < 3) continue;
  1124→      // head room for full height
  1125→      const top = Math.ceil(m.def.height);
  1126→      let clear = true;
  1127→      for (let yy = sy + 1; yy <= sy + top; yy++) {
  1128→        if (this.world.getBlock(x, yy, z) !== 0) { clear = false; break; }
  1129→      }
  1130→      if (!clear) continue;
  1131→      m.x = x + 0.5; m.y = sy + 1; m.z = z + 0.5;
  1132→      m.vx = 0; m.vy = 0; m.vz = 0;
  1133→      m.state = 'idle'; m.stateTimer = 1;
  1134→      cb.teleportParticles(ox, oy + m.height * 0.4, oz);
  1135→      cb.teleportParticles(m.x, m.y + m.height * 0.4, m.z);
  1136→      audio.enderTeleport();
  1137→      return true;
  1138→    }
  1139→    return false;
  1140→  }

  1143→  private lightAt(m: Mob, sunLevel: number): number {

  1149→  /** creeper explosion */
  1150→  private explode(x: number, y: number, z: number, cb: MobCallbacks): void {
  1151→    audio.boom();
  1152→    const R = 2.6;
  1153→    for (let bx = Math.floor(x - R); bx <= Math.floor(x + R); bx++)
  1154→      for (let by = Math.floor(y - R); by <= Math.floor(y + R); by++)
  1155→        for (let bz = Math.floor(z - R); bz <= Math.floor(z + R); bz++) {
  1156→          const d = Math.hypot(bx + 0.5 - x, by + 0.5 - y, bz + 0.5 - z);
  1157→          if (d > R) continue;
  1158→          const id = this.world.getBlock(bx, by, bz);
  1159→          if (id !== 0 && id !== BLOCK.BEDROCK && id !== BLOCK.WATER) {
  1160→            // chain reaction: TNT in the blast primes instead of vanishing
  1161→            if (id === BLOCK.TNT) {
  1162→              cb.igniteTnt?.(bx, by, bz);
  1163→              continue;
  1164→            }
  1165→            this.world.setBlock(bx, by, bz, 0);
  1166→          }
  1167→        }
  1168→    // damage nearby mobs (chain) + player
  1169→    for (const m of this.mobs) {
  1170→      if (m.dead) continue;
  1171→      const d = Math.hypot(m.x - x, m.y - y, m.z - z);
  1172→      if (d < R * 2 && m.type !== 'creeper') {
  1173→        this.hurtMob(m, Math.max(1, Math.round(14 * (1 - d / (R * 2)))), m.x - x, m.z - z, cb);
  1174→      }
  1175→    }
  1176→    const pd = Math.hypot(cb.playerX - x, cb.playerY - y, cb.playerZ - z);
  1177→    if (pd < R * 2) {
  1178→      cb.damagePlayer(Math.max(1, Math.round(16 * (1 - pd / (R * 2)))), x, z);
  1179→    }
  1180→    cb.explodeParticles(x, y, z);
  1181→  }

  1183→  update(dt: number, player: AABBEntity & { eyeY(): number }, sunLevel: number, cb: MobCallbacks): void {
  1184→    this.time += dt;
  1185→    this.lastCb = cb;
  1186→    // ── spawning ──
  1187→    this.spawnTimer -= dt;
  1188→    if (this.spawnTimer <= 0) {
  1189→      this.spawnTimer = 1.6;
  1190→      let passive = 0, hostile = 0;
  1191→      for (const m of this.mobs) {

  1618→          // (slow-fall glide). side -1 left / +1 right → mirror the z rotation

  1620→          const flap = !m.onGround ? Math.sin(this.time * 26) * 0.85 : 0;
  1621→          pivot.rotation.z = side * flap;
  1622→          pivot.rotation.x = 0;

  1624→        }

  1626→          pivot.rotation.x = -Math.PI / 2 + Math.sin(m.walkPhase * 2.4) * 0.12;
  1627→          pivot.rotation.z = Math.sin(m.walkPhase * 1.2) * 0.06;

  1629→          // bow arm raises to aim while chasing; the other swings gently

  1632→            pivot.rotation.x = aiming ? -1.35 : Math.sin(m.walkPhase * 2.4) * 0.2;

  1640→          const raised = m.state === 'chase' ? -1.2 : 0;
  1641→          pivot.rotation.x = raised + Math.sin(m.walkPhase * 2.4 + (pivot === (m.parts.arms[0] as unknown as { limbPivot?: THREE.Group }).limbPivot ? 0 : Math.PI)) * 0.3;
  1642→          pivot.rotation.z = Math.sin(m.walkPhase * 1.2) * 0.05;
  1643→        } else if (m.type === 'enderman') {
  1644→          if (m.provoked) {
  1645→            pivot.rotation.x = -1.15 + Math.sin(m.walkPhase * 2.4) * 0.1;
  1646→            pivot.rotation.z = Math.sin(m.walkPhase * 1.2) * 0.04;
  1647→          } else {
  1648→            pivot.rotation.x = Math.sin(m.walkPhase * 2.4) * 0.35;
  1649→            pivot.rotation.z = 0;
  1650→          }
  1651→        }
  1652→      }
  1653→      // head bob
  1654→      m.parts.head.rotation.y = Math.sin(this.time * 0.7 + m.walkPhase) * 0.14;
  1655→      // enderman glows purple when provoked
  1656→      if (m.type === 'enderman' && m.hurtT <= 0) {
  1657→        if (m.provoked) this.setTint(m, 1, 0.72, 1.12);
  1658→        else this.setTint(m, 1, 1, 1);
  1659→      }
  1660→
  1661→      // ── world-light shading: entities obey voxel light (night realism) ──
  1662→      // same formula as the terrain shader: brightness = max(blockLight, skyLight × sunLevel).
  1663→      // Mobs go dark on the surface at night, stay visible near torches, and are
  1664→      // near-black in unlit caves — instead of glowing under scene lights.
  1665→      {
  1666→        const lb = this.world.getLightForMesh(Math.floor(m.x), Math.floor(m.y + m.height * 0.7), Math.floor(m.z));
  1667→        const target = Math.max(0.1, Math.max((lb & 15) / 15, ((lb >> 4) / 15) * sunLevel));
  1668→        m.lightF += (target - m.lightF) * Math.min(1, dt * 6);
  1669→        if (Math.abs(m.lightF - m.lastAppliedF) > 0.004) {
  1670→          const f = m.lightF;
  1671→          for (const mm of m.parts.materials) mm.color.setRGB(m.tintR * f, m.tintG * f, m.tintB * f);
  1672→          m.lastAppliedF = f;
  1673→        }
  1674→      }
  1675→
  1676→      // shadow
  1677→      m.parts.shadow.position.set(m.x, m.y + 0.03, m.z);
  1678→      const shadowScale = m.onGround ? 1 : Math.max(0.4, 1 - Math.min(1, Math.abs(m.vy) * 0.06));
  1679→      m.parts.shadow.scale.setScalar(shadowScale);
  1680→
  1681→      // burning visual
  1682→      if (m.burning && Math.random() < dt * 12) {
  1683→        cb.fireParticle(m.x + (Math.random() - 0.5) * 0.5, m.y + Math.random() * m.height, m.z + (Math.random() - 0.5) * 0.5);
  1684→      }
  1685→    }
  1686→
  1687→    // ── arrows ──
  1688→    for (let i = this.arrows.length - 1; i >= 0; i--) {
  1689→      const a = this.arrows[i];
  1690→      a.life -= dt;
  1691→      if (a.life <= 0 || a.stuck > (a.fromPlayer ? 45 : 0.8)) {
  1692→        this.scene.remove(a.mesh);
  1693→        this.arrows.splice(i, 1);
  1694→        continue;
  1695→      }
  1696→      if (a.stuck > 0) {
  1697→        a.stuck += dt;
  1698→        // stuck player arrows are pickupable: walk near them to collect
  1699→        if (a.fromPlayer && a.stuck > 0.8) {
  1700→          const dxp = a.x - player.x, dyp = a.y - (player.y + player.height * 0.5), dzp = a.z - player.z;
  1701→          if (Math.hypot(dxp, dyp, dzp) < 1.5) {
  1702→            cb.spawnDrop(ITEM.ARROW, a.x, a.y, a.z);
  1703→            this.scene.remove(a.mesh);
  1704→            this.arrows.splice(i, 1);
  1705→            continue;
  1706→          }
  1707→        }
  1708→        continue;
  1709→      }
  1710→      // substepped movement so fast arrows can't tunnel through mobs/blocks
  1711→      const speedLen = Math.hypot(a.vx, a.vy, a.vz);
  1712→      const steps = Math.max(1, Math.ceil((speedLen * dt) / 0.45));
  1713→      const sdt = dt / steps;
  1714→      let done = false;
  1715→      for (let s = 0; s < steps && !done; s++) {
  1716→        a.vy -= 18 * sdt;
  1717→        const nx = a.x + a.vx * sdt;
  1718→        const ny = a.y + a.vy * sdt;
  1719→        const nz = a.z + a.vz * sdt;
  1720→        // block hit
  1721→        const bid = this.world.getBlock(Math.floor(nx), Math.floor(ny), Math.floor(nz));
  1722→        if (bid !== 0) {
  1723→          a.stuck = 0.001;
  1724→          audio.arrowHit();
  1725→          done = true;
  1726→          break;
  1727→        }
  1728→        a.x = nx; a.y = ny; a.z = nz;
  1729→        if (a.fromPlayer) {
  1730→          // mob hit: point-in-expanded-AABB test per mob
  1731→          for (const m of this.mobs) {
  1732→            if (m.dead) continue;
  1733→            if (
  1734→              a.x > m.x - m.width / 2 - 0.15 && a.x < m.x + m.width / 2 + 0.15 &&
  1735→              a.y > m.y - 0.1 && a.y < m.y + m.height + 0.1 &&
  1736→              a.z > m.z - m.width / 2 - 0.15 && a.z < m.z + m.width / 2 + 0.15
  1737→            ) {

  1739→              const shooterDist = Math.hypot(a.x - cb.playerX, a.y - cb.playerY, a.z - cb.playerZ);

  1744→              done = true;
  1745→              break;
  1746→            }
  1747→          }
  1748→          if (done) break;
  1749→        } else {
  1750→          // player hit
  1751→          const px = player.x, py = player.y + player.height * 0.5, pz = player.z;
  1752→          if (Math.hypot(a.x - px, a.y - py, a.z - pz) < 0.75) {
  1753→            cb.damagePlayer(3, a.x - a.vx, a.z - a.vz);
  1754→            a.life = 0;
  1755→            done = true;
  1756→            break;
  1757→          }
  1758→        }
  1759→      }
  1760→      if (done) continue;
  1761→      a.mesh.position.set(a.x, a.y, a.z);
  1762→      a.mesh.lookAt(a.x + a.vx, a.y + a.vy, a.z + a.vz);
  1763→    }
  1764→
  1765→    // ── witch potions ──
  1766→    for (let i = this.potions.length - 1; i >= 0; i--) {
  1767→      const po = this.potions[i];
  1768→      po.life -= dt;
  1769→      po.vy -= 16 * dt;
  1770→      // substepped so fast potions can't tunnel through blocks
  1771→      const sp = Math.hypot(po.vx, po.vy, po.vz);
  1772→      const steps = Math.max(1, Math.ceil((sp * dt) / 0.4));
  1773→      const sdt = dt / steps;
  1774→      let shattered = false;
  1775→      for (let s = 0; s < steps && !shattered; s++) {
  1776→        po.x += po.vx * sdt;
  1777→        po.y += po.vy * sdt;
  1778→        po.z += po.vz * sdt;
  1779→        const bid = this.world.getBlock(Math.floor(po.x), Math.floor(po.y), Math.floor(po.z));
  1780→        if (bid !== 0 && !isWaterId(bid)) shattered = true;
  1781→      }
  1782→      const pd = Math.hypot(po.x - player.x, po.y - (player.y + 0.9), po.z - player.z);
  1783→      if (pd < 1.1) shattered = true; // direct hit
  1784→      if (shattered || po.life <= 0) {
  1785→        this.scene.remove(po.mesh);
  1786→        this.potions.splice(i, 1);
  1787→        if (shattered) {
  1788→          const shatterDist = Math.hypot(po.x - cb.playerX, po.y - cb.playerY, po.z - cb.playerZ);
  1789→          audio.potionShatter(shatterDist);
  1790→          cb.teleportParticles(po.x, po.y + 0.2, po.z); // purple magic splash
  1791→          // splash radius 2.6: damage + poison (MC splash lingering area)
  1792→          const pDist = Math.hypot(po.x - player.x, po.y - (player.y + 0.9), po.z - player.z);
  1793→          if (pDist < 2.6 && !cb.playerCreative) {
  1794→            cb.damagePlayer(Math.max(1, Math.round(4 * (1 - pDist / 2.6))), po.x, po.z);
  1795→            cb.poisonPlayer?.(4.5);
  1796→          }
  1797→        }
  1798→        continue;
  1799→      }
  1800→      po.mesh.position.set(po.x, po.y, po.z);
  1801→      po.mesh.rotation.x += dt * 7;
  1802→      po.mesh.rotation.z += dt * 4;
  1803→    }
