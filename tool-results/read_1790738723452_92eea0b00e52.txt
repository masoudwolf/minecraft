   209→function makeTool(type: ToolDef['type'], tierName: TierName): ToolDef {
   210→  const t = TIER_STATS[tierName];
   211→  const dmg =
   212→    type === 'sword'
   213→      ? t.swordDmg
   214→      : type === 'axe'
   215→        ? t.swordDmg - 1
   216→        : type === 'hoe'
   217→          ? Math.max(1, t.swordDmg - 3)
   218→          : Math.max(2, t.swordDmg - 2);
   219→  return { type, tier: t.tier, speed: t.speed, dur: t.dur, dmg };
   220→}

   222→// ─── pixel-art helpers ────────────────────────────────────────────────────────
   223→function noiseRect(ctx: CanvasRenderingContext2D, colors: string[], x0: number, y0: number, w: number, h: number, seed = 1): void {
   224→  let s = seed;
   225→  const rnd = (): number => {
   226→    s = (s * 16807) % 2147483647;
   227→    return s / 2147483647;
   228→  };
   229→  for (let y = 0; y < h; y++)
   230→    for (let x = 0; x < w; x++) {
   231→      ctx.fillStyle = colors[Math.floor(rnd() * colors.length)];
   232→      ctx.fillRect(x0 + x, y0 + y, 1, 1);
   233→    }
   234→}
   235→
   236→function px(ctx: CanvasRenderingContext2D, x: number, y: number, color: string, w = 1, h = 1): void {
   237→  ctx.fillStyle = color;
   238→  ctx.fillRect(x, y, w, h);
   239→}

   258→/** diagonal stick handle from bottom-left to (hx,hy), 2px thick */
   259→function drawHandle(ctx: CanvasRenderingContext2D, hx: number, hy: number, len: number): void {
   260→  for (let i = 0; i < len; i++) {
   261→    px(ctx, hx - i, hy + i, HANDLE, 2, 1);
   262→    px(ctx, hx - i, hy + i + 1, HANDLE_DARK, 1, 1);
   263→  }
   264→}

   792→// ─── mining model ─────────────────────────────────────────────────────────────
   793→export interface BreakInfo {
   794→  /** seconds to break */
   795→  time: number;
   796→  /** will the block drop its item */
   797→  harvest: boolean;
   798→}
   799→
   800→export function breakInfo(def: BlockDef, tool?: ToolDef): BreakInfo {
   801→  if (!Number.isFinite(def.hardness)) return { time: Infinity, harvest: false };
   802→  const minTier = def.minTier ?? 0;
   803→  let speed = 1;
   804→  let harvest = def.tool ? minTier === 0 : true;
   805→  if (def.tool && tool && tool.type === def.tool) {
   806→    speed = tool.speed;
   807→    harvest = tool.tier >= minTier;
   808→  }
   809→  const base = harvest ? 1.5 : 5;
   810→  return { time: Math.max(0.05, (def.hardness * base) / speed), harvest };
   811→}

   813→// ─── Icon rendering (cached dataURLs) ────────────────────────────────────────
   814→const iconCache = new Map<number, string>();
   815→
   816→export function getItemIcon(itemId: number): string {
   817→  const cached = iconCache.get(itemId);
   818→  if (cached) return cached;
   819→  const def = ITEMS[itemId];
   820→  if (!def) return '';
   821→  const c = document.createElement('canvas');
   822→  c.width = 16; c.height = 16;
   823→  const ctx = c.getContext('2d')!;
   824→  ctx.imageSmoothingEnabled = false;
   825→  def.icon(ctx);
   826→  // scale up 4x for crisp display
   827→  const big = document.createElement('canvas');
   828→  big.width = 64; big.height = 64;
   829→  const bctx = big.getContext('2d')!;
   830→  bctx.imageSmoothingEnabled = false;
   831→  bctx.drawImage(c, 0, 0, 64, 64);
   832→  const url = big.toDataURL();
   833→  iconCache.set(itemId, url);
   834→  return url;
   835→}
